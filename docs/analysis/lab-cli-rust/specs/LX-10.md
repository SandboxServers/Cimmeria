```rust
/// Runs `lab record`: arguments are checked first (exit 2), then the daemon is
/// probed (exit 3), then one tool call is made per action over its own MCP session.
pub fn run(args: &Args, ctx: &Ctx) -> u8 {
    let ep = Endpoint::discover();
    let env = Env {
        lab_home: paths::lab_home(),
        uat_root: paths::uat_root(),
        specs_dir: paths::default_specs_dir(),
        cwd: std::env::current_dir().unwrap_or_default(),
    };
    // Get-LabToken -and Get-LabStatus: a token, and a status that answers.
    let probe = || ep.token.as_deref().is_some_and(|t| !t.is_empty()) && status::fetch(&ep).is_some();
    // One MCP session per call: a command makes one call.
    let call = |name: &str, arguments: Value, timeout: Duration| -> Result<Value, McpError> {
        let mut client = McpClient::connect(&ep)?;
        let result = client.call_tool(name, arguments, timeout);
        client.close();
        result
    };
    let outcome = run_action(args, &env, &ctx.ui, probe, call);
    let out = render(&ctx.ui, args.json, &outcome);
    if out.to_stderr {
        anstream::eprintln!("{}", out.text);
    } else {
        anstream::println!("{}", out.text);
    }
    outcome.exit
}
```

## Step 2: `record/checks.rs` (ports `Test-RecordArguments`)

```rust
//! The `lab record` argument checks that need no daemon (`Test-RecordArguments`):
//! a usage error exits 2 before anything is asked of the daemon. The message
//! texts are the PowerShell command's, word for word, flag spellings included.

use std::sync::OnceLock;

use regex::Regex;

use super::Args;

/// The longest recording, in minutes.
pub const MAX_MINUTES: u32 = 30;
/// A section id.
pub const SECTION_PATTERN: &str = "^[a-z0-9][a-z0-9-]{0,47}$";
/// What `--character-expr` may hold.
pub const CHARACTER_EXPR_PATTERN: &str = "^[A-Za-z0-9 _${}]{1,64}$";
/// An instance name; it becomes part of the state file's name.
pub const INSTANCE_PATTERN: &str = "^[A-Za-z0-9_-]{1,32}$";

/// The usage text: four lines, in the PowerShell spelling.
pub const USAGE: &str = "usage: lab record start <section> [-Instance p2] [-Character <name>] [-MaxMinutes 30] [-SpecsDir <dir>] [-Json]\n       lab record status [-Instance p2] [-Json]\n       lab record stop [-Instance p2] [-CharacterExpr 'Px${run_id}'] [-Profile <name>] [-NoDraft] [-RecordingId <id>] [-Json]\n       lab record transform <recording id or folder> [-Section <id>] [-CharacterExpr ...] [-Profile ...] [-SpecsDir <dir>] [-Json]";

fn matches(cell: &'static OnceLock<Regex>, pattern: &str, text: &str) -> bool {
    cell.get_or_init(|| Regex::new(pattern).expect("record pattern"))
        .is_match(text)
}

/// True when `text` is a section id (case sensitive).
pub fn is_section(text: &str) -> bool {
    static RE: OnceLock<Regex> = OnceLock::new();
    matches(&RE, SECTION_PATTERN, text)
}

/// True when `text` is an instance name (case sensitive).
pub fn is_instance(text: &str) -> bool {
    static RE: OnceLock<Regex> = OnceLock::new();
    matches(&RE, INSTANCE_PATTERN, text)
}

/// True when `text` is a character expression (case sensitive).
pub fn is_character_expr(text: &str) -> bool {
    static RE: OnceLock<Regex> = OnceLock::new();
    matches(&RE, CHARACTER_EXPR_PATTERN, text)
}

/// `Some(text)` when an option was given: not absent and not an empty string.
pub fn given(value: &Option<String>) -> Option<&str> {
    value.as_deref().filter(|text| !text.is_empty())
}

/// The first problem with `args` that needs no daemon, or `None`.
pub fn check_arguments(args: &Args) -> Option<String> {
    let action = args.action.as_deref().unwrap_or("");
    if !["start", "status", "stop", "transform"].contains(&action) {
        return Some(USAGE.to_string());
    }
    let target = given(&args.target).unwrap_or("");
    match action {
        "start" => {
            if target.is_empty() {
                return Some(format!(
                    "start needs a section: lab record start <section>\n{USAGE}"
                ));
            }
            if !is_section(target) {
                return Some(format!(
                    "section: give a section id, {SECTION_PATTERN} (first-session); got '{target}'"
                ));
            }
        }
        "transform" => {
            if target.is_empty() {
                return Some(format!(
                    "transform needs a recording: lab record transform <recording id or folder>\n{USAGE}"
                ));
            }
        }
        _ => {
            if !target.is_empty() {
                return Some(format!("{action} takes no positional argument; got '{target}'"));
            }
        }
    }
    let instance = given(&args.instance).unwrap_or("");
    if !instance.is_empty() && !is_instance(instance) {
        return Some(format!(
            "-Instance: give an instance name like p2 ([A-Za-z0-9_-]); got '{instance}'"
        ));
    }
    // A string, so `-MaxMinutes abc` exits 2 with this message.
    let minutes_ok = match given(&args.max_minutes) {
        None => true,
        Some(m) => {
            m.len() <= 9
                && m.bytes().all(|b| b.is_ascii_digit())
                && m.parse::<u32>().is_ok_and(|n| (1..=MAX_MINUTES).contains(&n))
        }
    };
    if !minutes_ok {
        return Some(format!("-MaxMinutes must be 1 to {MAX_MINUTES}"));
    }
    let expr = given(&args.character_expr).unwrap_or("");
    if !expr.is_empty() && !is_character_expr(expr) {
        return Some("-CharacterExpr: give ^[A-Za-z0-9 _${}]{1,64}$ (Px${run_id})".to_string());
    }
    let section = given(&args.section).unwrap_or("");
    if !section.is_empty() && !is_section(section) {
        return Some(format!(
            "-Section: give a section id, {SECTION_PATTERN}; got '{section}'"
        ));
    }
    // (action the option belongs to, option name, whether it was given)
    let only_for = [
        ("start", "Character", given(&args.character).is_some()),
        ("start", "MaxMinutes", given(&args.max_minutes).is_some()),
        ("stop", "NoDraft", args.no_draft),
        ("stop", "RecordingId", given(&args.recording_id).is_some()),
        ("transform", "Section", given(&args.section).is_some()),
    ];
    for (owner, option, set) in only_for {
        if action != owner && set {
            return Some(format!("-{option} is for lab record {owner}"));
        }
    }
    let expr_or_profile = given(&args.character_expr).is_some() || given(&args.profile).is_some();
    if matches!(action, "start" | "status") && expr_or_profile {
        return Some("-CharacterExpr and -Profile are for stop and transform".to_string());
    }
    if matches!(action, "status" | "stop") && given(&args.specs_dir).is_some() {
        return Some("-SpecsDir is for start and transform".to_string());
    }
    if action == "transform" && !instance.is_empty() {
        return Some("transform runs on no instance; drop -Instance".to_string());
    }
    None
}
```

## Step 3: `record/state.rs` (ports `Get-RecordStatePath`, `Save-RecordState`, `Read-RecordState`)

State file format: one line of compact JSON, `{"recording_id":"<id>","instance":"<name>"}` plus `,"specs_dir":"<dir>"` last when the start sent a spec folder, then `\n`; UTF-8, no BOM. Path: `<lab home>\record-<instance>.json`.

```rust
//! The recording state file, `<lab home>\record-<instance>.json`: it names the
//! recording `lab record start` began, so `stop` needs no id.

use std::io;
use std::path::{Path, PathBuf};

use serde_json::Value;

use super::checks::is_instance;

/// What the state file holds. A file without `instance` or `specs_dir` still reads.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct RecordState {
    /// The recording the daemon is holding.
    pub recording_id: String,
    /// The instance it runs on.
    pub instance: Option<String>,
    /// The spec folder fixed at start, so the stop can print the draft's path.
    pub specs_dir: Option<String>,
}

/// `<lab_home>\record-<instance>.json`. The name is checked first because it
/// becomes part of a path: `Err("bad instance name '<x>'")` when it is not an
/// instance name.
pub fn state_path(lab_home: &Path, instance: &str) -> Result<PathBuf, String> {
    if !is_instance(instance) {
        return Err(format!("bad instance name '{instance}'"));
    }
    Ok(lab_home.join(format!("record-{instance}.json")))
}

fn json_string(text: &str) -> String {
    Value::String(text.to_owned()).to_string()
}

/// Writes the state file, creating the folder. `specs_dir` is empty when the
/// start sent none (then the key is left out).
pub fn save_state(
    path: &Path,
    recording_id: &str,
    instance: &str,
    specs_dir: &str,
) -> io::Result<()> {
    if let Some(dir) = path.parent() {
        std::fs::create_dir_all(dir)?;
    }
    let mut text = format!(
        "{{\"recording_id\":{},\"instance\":{}",
        json_string(recording_id),
        json_string(instance)
    );
    if !specs_dir.is_empty() {
        text.push_str(&format!(",\"specs_dir\":{}", json_string(specs_dir)));
    }
    text.push_str("}\n");
    std::fs::write(path, text)
}

/// The state, or `None` when there is no file, it cannot be read, or it has
/// no non-empty `recording_id`.
pub fn read_state(path: &Path) -> Option<RecordState> {
    let text = std::fs::read_to_string(path).ok()?;
    let value: Value = serde_json::from_str(text.trim_start_matches('\u{feff}')).ok()?;
    let object = value.as_object()?;
    let recording_id = object.get("recording_id")?.as_str()?.to_owned();
    if recording_id.is_empty() {
        return None;
    }
    let text_of = |key: &str| {
        object
            .get(key)
            .and_then(Value::as_str)
            .filter(|s| !s.is_empty())
            .map(str::to_owned)
    };
    Some(RecordState {
        recording_id,
        instance: text_of("instance"),
        specs_dir: text_of("specs_dir"),
    })
}
```

## Step 4: `record/json.rs` (ports `ConvertTo-RecordCompactJson`, `New-RecordFailure`, `ConvertFrom-RecordCallError`)

Key order: a failure is always `ok, exit, error`. A tool's answer keeps the keys in the order `serde_json` parsed them, which is alphabetical (the daemon writes them the same way, and `serde_json` here has no `preserve_order`), so `-Json` output equals what the PowerShell printed for the same daemon.

```rust
//! The `lab record` JSON line, the failure body and the exit-code mapping of a
//! failed tool call (D-LR16).

use serde_json::Value;

use crate::daemon::mcp::{hide_lease_ids, McpError};

/// A `--json` line stays under this many characters (a failure's error is cut to fit).
pub const BODY_CAP: usize = 400;

/// A JSON object as ordered pairs.
pub type Body = Vec<(String, Value)>;

/// `{ok:false, exit, error}`, with lease ids hidden in the error.
pub fn failure(exit: u8, message: &str) -> Body {
    vec![
        ("ok".to_owned(), Value::Bool(false)),
        ("exit".to_owned(), Value::from(exit)),
        ("error".to_owned(), Value::String(hide_lease_ids(message))),
    ]
}

/// A tool's answer as a body (an answer that is not an object gives an empty body).
pub fn body_from_reply(reply: &Value) -> Body {
    match reply.as_object() {
        Some(map) => map.iter().map(|(k, v)| (k.clone(), v.clone())).collect(),
        None => Vec::new(),
    }
}

/// One value of a body.
pub fn body_get<'a>(body: &'a Body, key: &str) -> Option<&'a Value> {
    body.iter().find(|(k, _)| k == key).map(|(_, v)| v)
}

/// The body's `error` text, or an empty string.
pub fn error_text(body: &Body) -> String {
    body_get(body, "error")
        .and_then(Value::as_str)
        .unwrap_or("")
        .to_owned()
}

fn join(pairs: &[(&str, Value)]) -> String {
    let items: Vec<String> = pairs
        .iter()
        .map(|(key, value)| format!("{}:{}", Value::String((*key).to_owned()), value))
        .collect();
    format!("{{{}}}", items.join(","))
}

/// One compact JSON line: null and empty-string top-level values dropped (zeros
/// and `false` kept). When the body has a string `error`, the error is cut until
/// the line is under [`BODY_CAP`] characters.
pub fn compact_json(body: &Body) -> String {
    let mut kept: Vec<(&str, Value)> = body
        .iter()
        .filter(|(_, v)| !(v.is_null() || v.as_str() == Some("")))
        .map(|(k, v)| (k.as_str(), v.clone()))
        .collect();
    let mut line = join(&kept);
    let error_at = kept.iter().position(|(k, v)| *k == "error" && v.is_string());
    if let Some(at) = error_at {
        let error = kept[at].1.as_str().unwrap_or("").to_owned();
        let mut keep = error.chars().count();
        while line.chars().count() >= BODY_CAP && keep > 0 {
            let over = line.chars().count() + 1 - BODY_CAP;
            keep = keep.saturating_sub(over.max(1));
            kept[at].1 = Value::String(error.chars().take(keep).collect());
            line = join(&kept);
        }
    }
    line
}

fn first_line(text: &str) -> &str {
    text.lines().next().unwrap_or("")
}

/// `{ok:false, exit, error}` text from a tool error, as `(exit, body)`. The
/// exit must be 1..=255; anything else becomes 1.
fn parse_refusal(text: &str) -> Option<(u8, Body)> {
    let value: Value = serde_json::from_str(text).ok()?;
    let map = value.as_object()?;
    if map.get("ok") != Some(&Value::Bool(false)) {
        return None;
    }
    let exit = map.get("exit")?;
    let error = map.get("error")?;
    let code = exit
        .as_i64()
        .or_else(|| exit.as_str().and_then(|s| s.trim().parse().ok()))?;
    let code = u8::try_from(code).ok().filter(|c| *c >= 1).unwrap_or(1);
    let message = match error {
        Value::String(s) => s.clone(),
        other => other.to_string(),
    };
    Some((code, failure(code, &message)))
}

/// A failed call as `(exit, body)`. A tool error whose text is the refusal JSON
/// keeps its body and its exit (the daemon answers 4 itself for a busy
/// instance); any other tool error is exit 1; an empty-reply protocol error
/// (`<tool> returned ...`) is exit 1; anything else means the daemon did not
/// answer, exit 3.
pub fn call_error(tool: &str, err: &McpError) -> (u8, Body) {
    let message = match err {
        McpError::Tool(text) => {
            let text = hide_lease_ids(text);
            return match parse_refusal(&text) {
                Some(parsed) => parsed,
                None => (1, failure(1, &text)),
            };
        }
        McpError::Protocol(text) => {
            let first = first_line(text);
            if first.starts_with(&format!("{tool} returned")) {
                return (1, failure(1, first));
            }
            format!("the lab daemon did not answer: {first}")
        }
        McpError::Unreachable(text) => {
            format!("the lab daemon did not answer: {}", first_line(text))
        }
        McpError::Http(code, text) => {
            format!("the lab daemon did not answer: HTTP {code}: {}", first_line(text))
        }
    };
    (3, failure(3, &message))
}
```

## Step 5: `record/action.rs` (ports `Invoke-RecordAction`, `Format-RecordDraftLine`, and `record.ps1`'s output rules)

Behaviour to keep (all in the code below): arguments checked first (exit 2); for `stop` the state file is read before the daemon is probed (exit 2 when there is none and no `--recording-id`); then the probe (exit 3: for `transform` the text is ``the transform runs in the lab daemon; start it with `lab start` `` with literal backticks, otherwise `the lab daemon is not answering (lab status; lab start)`); then exactly one tool call (timeouts: start 120 s, status 30 s, stop 180 s, transform 120 s). A failed stop keeps the state file; a good stop deletes it. Plain output is the PowerShell text. With `--json` the line goes to stdout whatever the exit; without it, exit 0 prints to stdout and any other exit to stderr. Lease ids are hidden in everything printed.

Deliberate small differences from the PowerShell: a refusal body whose `exit` is 0 or above 255 becomes exit 1 (the PowerShell would have treated exit 0 as success); a tool answer that is not a JSON object is treated as `{}`; a state file that cannot be deleted after a good stop gives exit 1 `lab record: <os error>` (the PowerShell threw to the same effect).

```rust

/// One tool call: name, arguments, timeout for that request.
pub type CallFn<'a> = dyn FnMut(&str, Value, Duration) -> Result<Value, McpError> + 'a;

/// What the actions need from the machine.
#[derive(Debug, Clone)]

    pub uat_root: PathBuf,

    pub cwd: PathBuf,

/// The result of one action.
#[derive(Debug, Clone)]
pub struct Outcome {
    /// The process exit code.
    pub exit: u8,
    /// What `--json` prints.
    pub body: Body,
    /// What a person gets (plain, or styled when the `Ui` is fancy); for a
    /// failure, the plain error text.

}

/// What to print, and where.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Rendered {
    /// The text, without its final newline.
    pub text: String,
    /// True for stderr.
    pub to_stderr: bool,
}

struct Job<'a> {
    args: &'a Args,
    env: &'a Env,
    ui: &'a Ui,
    instance: &'a str,
    specs: Option<String>,
}

fn refuse(exit: u8, message: &str) -> Outcome {
    let body = failure(exit, message);
    let text = error_text(&body);
    Outcome { exit, body, text }
}

fn tool(
    call: &mut CallFn<'_>,
    name: &str,
    sent: Map<String, Value>,
    secs: u64,
) -> Result<Value, Outcome> {
    call(name, Value::Object(sent), Duration::from_secs(secs)).map_err(|err| {
        let (exit, body) = call_error(name, &err);
        let text = error_text(&body);
        Outcome { exit, body, text }
    })
}

/// A field as text, the way PowerShell interpolates it: empty for a missing or null one.
fn field_text(reply: &Value, key: &str) -> String {
    match reply.get(key) {
        None | Some(Value::Null) => String::new(),
        Some(Value::String(s)) => s.clone(),
        Some(other) => other.to_string(),
    }
}

/// PowerShell truthiness of a field: missing, null, false, 0 and "" are false.
fn truthy(reply: &Value, key: &str) -> bool {
    match reply.get(key) {
        None | Some(Value::Null) => false,
        Some(Value::Bool(b)) => *b,
        Some(Value::String(s)) => !s.is_empty(),
        Some(Value::Number(n)) => n.as_f64().is_some_and(|f| f != 0.0),
        Some(_) => true,
    }
}

/// `75` is `1m 15s`, `5` is `5s`, `3700` is `1h 1m`.
pub fn format_age(secs: u64) -> String {
    if secs < 60 {
        format!("{secs}s")
    } else if secs < 3600 {
        format!("{}m {}s", secs / 60, secs % 60)
    } else {
        format!("{}h {}m", secs / 3600, secs % 3600 / 60)
    }
}

fn join_path(base: &str, parts: &[&str]) -> String {
    let mut path = PathBuf::from(base);
    for part in parts {
        path.push(part);
    }
    path.display().to_string()
}

/// Runs one action. `probe` says whether the daemon answers; `call` makes one
/// tool call. Nothing here prints.
pub fn run_action<P, F>(args: &Args, env: &Env, ui: &Ui, probe: P, mut call: F) -> Outcome
where
    P: Fn() -> bool,
    F: FnMut(&str, Value, Duration) -> Result<Value, McpError>,
{
    if let Some(bad) = check_arguments(args) {
        return refuse(2, &bad);
    }
    let action = args.action.as_deref().unwrap_or("");
    let instance = given(&args.instance).unwrap_or("");
    let specs = given(&args.specs_dir)
        .map(str::to_owned)
        .or_else(|| env.specs_dir.as_ref().map(|p| p.display().to_string()));

    // stop's state file is read before the daemon is asked: without it there is nothing to stop.
    let mut state: Option<RecordState> = None;
    let mut state_file: Option<PathBuf> = None;
    if action == "stop" {
        let name = if instance.is_empty() { "default" } else { instance };
        let path = match state_path(&env.lab_home, name) {
            Ok(path) => path,
            Err(message) => return refuse(1, &format!("lab record: {message}")),
        };
        state = read_state(&path);
        if state.is_none() && given(&args.recording_id).is_none() {
            return refuse(
                2,
                &format!(
                    "no recording started here for instance {name} ({}); lab record status shows one, -RecordingId <id> stops it",
                    path.display()
                ),
            );
        }
        state_file = Some(path);
    }
    if !probe() {
        let message = if action == "transform" {
            "the transform runs in the lab daemon; start it with `lab start`"
        } else {
            "the lab daemon is not answering (lab status; lab start)"
        };
        return refuse(3, message);
    }

    let mut sent = Map::new();
    if !instance.is_empty() {
        sent.insert("instance".to_owned(), Value::from(instance));
    }
    let job = Job {
        args,
        env,
        ui,
        instance,
        specs,
    };
    let call: &mut CallFn<'_> = &mut call;
    match action {
        "start" => start(&job, sent, call),
        "status" => status(&job, sent, call),
        "stop" => stop(&job, sent, state.as_ref(), state_file.as_deref(), call),
        _ => transform(&job, sent, call),
    }
}

fn start(job: &Job<'_>, mut sent: Map<String, Value>, call: &mut CallFn<'_>) -> Outcome {
    let args = job.args;
    let ui = job.ui;
    let section = given(&args.target).unwrap_or("").to_owned();
    sent.insert("section".to_owned(), Value::from(section.clone()));
    if let Some(character) = given(&args.character) {
        sent.insert("character".to_owned(), Value::from(character));
    }
    if let Some(minutes) = given(&args.max_minutes) {
        // check_arguments proved it is 1..=30.
        let minutes = minutes.parse::<u32>().unwrap_or(MAX_MINUTES);
        sent.insert("max_minutes".to_owned(), Value::from(minutes));
    }
    if let Some(specs) = &job.specs {
        sent.insert("specs_dir".to_owned(), Value::from(specs.clone()));
    }
    let reply = match tool(call, "lab_record_start", sent, 120) {
        Ok(reply) => reply,
        Err(outcome) => return outcome,
    };
    let id = field_text(&reply, "recording_id");
    let mut on = field_text(&reply, "instance");
    if on.is_empty() {
        on = if job.instance.is_empty() {
            "default".to_owned()
        } else {
            job.instance.to_owned()
        };
    }
    if !is_instance(&on) || id.is_empty() {
        return refuse(
            1,
            &format!(
                "lab_record_start answered without a usable recording_id or instance; lab record status -Instance {on} shows it"
            ),
        );
    }
    let stop_hint = if on == "default" {
        "lab record stop".to_owned()
    } else {
        format!("lab record stop -Instance {on}")
    };
    // The recording runs on the daemon whatever happens here: a state file
    // that cannot be written must still name it.
    let path = match state_path(&job.env.lab_home, &on) {
        Ok(path) => path,
        Err(message) => return refuse(1, &message),
    };
    if let Err(err) = save_state(&path, &id, &on, job.specs.as_deref().unwrap_or("")) {
        return refuse(
            1,
            &format!(
                "recording {id} started on {on}, but its state file could not be written ({err}); stop it with: {stop_hint} -RecordingId {id}"
            ),
        );
    }
    let text = if ui.fancy {
        let minutes = given(&args.max_minutes)
            .and_then(|m| m.parse::<u32>().ok())
            .unwrap_or(MAX_MINUTES);
        format!(
            "{} recording {on} · section {section} · stops by itself in {minutes}m\n{}",
            ui.paint(Tone::Ok, ui.mark(Mark::Running)),
            ui.paint(Tone::Dim, &format!("  stop with: {stop_hint}"))
        )
    } else {
        format!("recording {id} on {on} (section {section}); play, then: {stop_hint}")
    };
    Outcome {
        exit: 0,
        body: body_from_reply(&reply),
        text,
    }
}

fn status(job: &Job<'_>, sent: Map<String, Value>, call: &mut CallFn<'_>) -> Outcome {
    let ui = job.ui;
    let reply = match tool(call, "lab_record_status", sent, 30) {
        Ok(reply) => reply,
        Err(outcome) => return outcome,
    };
    let on = field_text(&reply, "instance");
    let id = field_text(&reply, "recording_id");
    let section = field_text(&reply, "section");
    let (age, ticks, anchors) = (
        field_text(&reply, "age_s"),
        field_text(&reply, "ticks"),
        field_text(&reply, "anchors"),
    );
    let running = truthy(&reply, "recording");
    let text = if !truthy(&reply, "recording_id") {
        if ui.fancy {
            ui.paint(Tone::Dim, &format!("{} no recording on {on}", ui.mark(Mark::Stopped)))
        } else {
            format!("no recording on {on}")
        }
    } else if ui.fancy {
        let head = if running {
            format!("{} recording {id}", ui.paint(Tone::Ok, ui.mark(Mark::Running)))
        } else {
            format!(
                "{} stopped by itself, waiting for lab record stop: {id}",
                ui.paint(Tone::Warn, ui.mark(Mark::Warn))
            )
        };
        let age_text = reply
            .get("age_s")
            .and_then(Value::as_u64)
            .map(format_age)
            .unwrap_or_default();
        let kv = |key: &str, value: &str| {
            format!("  {}  {value}", ui.paint(Tone::Dim, &format!("{key:<9}")))
        };
        [
            head,
            kv("instance", &on),
            kv("section", &section),
            kv("age", &age_text),
            kv("ticks", &ticks),
            kv("anchors", &anchors),
        ]
        .join("\n")
    } else {
        let what = if running {
            "recording"
        } else {
            "stopped by itself, waiting for lab record stop:"
        };
        format!("{what} {id} on {on} (section {section}): {age} s, {ticks} ticks, {anchors} anchors")
    };
    Outcome {
        exit: 0,
        body: body_from_reply(&reply),
        text,
    }
}

fn stop(
    job: &Job<'_>,
    mut sent: Map<String, Value>,
    state: Option<&RecordState>,
    state_file: Option<&Path>,
    call: &mut CallFn<'_>,
) -> Outcome {
    let args = job.args;
    let id = given(&args.recording_id)
        .map(str::to_owned)
        .or_else(|| state.map(|s| s.recording_id.clone()))
        .unwrap_or_default();
    sent.insert("recording_id".to_owned(), Value::from(id));
    let saved_on = state
        .and_then(|s| s.instance.as_deref())
        .filter(|_| job.instance.is_empty());
    if let Some(saved_on) = saved_on {
        sent.insert("instance".to_owned(), Value::from(saved_on));
    }
    if let Some(expr) = given(&args.character_expr) {
        sent.insert("character_expr".to_owned(), Value::from(expr));
    }
    if let Some(profile) = given(&args.profile) {
        sent.insert("profile".to_owned(), Value::from(profile));
    }
    if args.no_draft {
        sent.insert("no_draft".to_owned(), Value::Bool(true));
    }
    let reply = match tool(call, "lab_record_stop", sent, 180) {
        Ok(reply) => reply,
        Err(mut outcome) => {
            // The recording stays on a failure (LR-07): the file stays, so a retry needs no id.
            if matches!(outcome.exit, 1 | 3) && state.is_some() {
                outcome
                    .text
                    .push_str("\nthe recording is kept; retry with lab record stop");
                if !job.instance.is_empty() {
                    outcome.text.push_str(&format!(" -Instance {}", job.instance));
                }
            }
            return outcome;
        }
    };
    let removed = state_file
        .filter(|path| path.exists())
        .map(std::fs::remove_file)
        .transpose();
    if let Err(err) = removed {
        return refuse(1, &format!("lab record: {err}"));
    }
    // The draft went to the folder fixed at start; only the state file knows it.
    let draft_specs = state
        .and_then(|s| s.specs_dir.as_deref())
        .unwrap_or("");
    Outcome {
        exit: 0,
        text: draft_text(job.ui, &reply, draft_specs, &job.env.uat_root),
        body: body_from_reply(&reply),
    }
}

fn transform(job: &Job<'_>, mut sent: Map<String, Value>, call: &mut CallFn<'_>) -> Outcome {
    let args = job.args;
    let target = given(&args.target).unwrap_or("");
    // A folder given relative to here is sent in full; a bare id as it is.
    let path = Path::new(target);
    let rooted = path.is_absolute() || path.has_root();
    let full = job.env.cwd.join(path);
    let dir = if !rooted && full.is_dir() {
        std::path::absolute(&full).unwrap_or(full).display().to_string()
    } else {
        target.to_owned()
    };
    sent.insert("dir".to_owned(), Value::from(dir));
    if let Some(section) = given(&args.section) {
        sent.insert("section".to_owned(), Value::from(section));
    }
    if let Some(expr) = given(&args.character_expr) {
        sent.insert("character_expr".to_owned(), Value::from(expr));
    }
    if let Some(profile) = given(&args.profile) {
        sent.insert("profile".to_owned(), Value::from(profile));
    }
    if let Some(specs) = &job.specs {
        sent.insert("specs_dir".to_owned(), Value::from(specs.clone()));
    }
    let reply = match tool(call, "lab_record_transform", sent, 120) {
        Ok(reply) => reply,
        Err(outcome) => return outcome,
    };
    Outcome {
        exit: 0,
        text: draft_text(
            job.ui,
            &reply,
            job.specs.as_deref().unwrap_or(""),
            &job.env.uat_root,
        ),
        body: body_from_reply(&reply),
    }
}

struct DraftParts {
    draft: Option<String>,
    counts: String,
    stopped: String,
    raw: String,
    ended: String,
}

fn draft_parts(reply: &Value, specs_dir: &str, uat_root: &Path) -> DraftParts {
    let raw = field_text(reply, "raw");
    let uat = uat_root.display().to_string();
    let raw_path = if !uat.is_empty() && !raw.is_empty() {
        join_path(&uat, &["records", &raw, "record.jsonl"])
    } else {
        raw
    };
    let draft = field_text(reply, "draft");
    let draft_path = if draft.is_empty() {
        None
    } else if !specs_dir.is_empty() {
        Some(join_path(specs_dir, &["drafts", &draft]))
    } else {
        Some(format!("<specs>/drafts/{draft}"))
    };
    DraftParts {
        draft: draft_path,
        counts: format!(
            "{} row(s), {} step(s), {} TODO(s), {} dropped",
            field_text(reply, "rows"),
            field_text(reply, "steps"),
            field_text(reply, "todo"),
            field_text(reply, "dropped")
        ),
        stopped: format!("stopped; {} tick(s), no draft", field_text(reply, "ticks")),
        raw: raw_path,
        ended: field_text(reply, "ended"),
    }
}

/// A stop or transform answer for a person: the draft's and raw log's paths
/// (the tools answer with names only) and the counts. `specs_dir` is empty when
/// it is not known here, and the draft's path then starts `<specs>/drafts/`.
pub fn draft_text(ui: &Ui, reply: &Value, specs_dir: &str, uat_root: &Path) -> String {
    let parts = draft_parts(reply, specs_dir, uat_root);
    let mut lines: Vec<String> = Vec::new();
    if ui.fancy {
        let pass = ui.paint(Tone::Ok, ui.mark(Mark::Pass));
        match &parts.draft {
            Some(path) => {
                lines.push(format!("{pass} draft written: {}", ui.paint(Tone::Cmd, path)));
                lines.push(ui.paint(Tone::Dim, &format!("  {}", parts.counts)));
            }
            None => lines.push(format!("{pass} {}", parts.stopped)),
        }
        lines.push(ui.paint(Tone::Dim, &format!("  raw   {}", parts.raw)));
        if !parts.ended.is_empty() {
            lines.push(format!(
                "{} the recorder stopped early: {}",
                ui.paint(Tone::Warn, ui.mark(Mark::Warn)),
                parts.ended
            ));
        }
    } else {
        match &parts.draft {
            Some(path) => {
                lines.push(format!("draft {path}"));
                lines.push(format!("  {}", parts.counts));
            }
            None => lines.push(parts.stopped.clone()),
        }
        lines.push(format!("raw   {}", parts.raw));
        if !parts.ended.is_empty() {
            lines.push(format!("the recorder stopped early: {}", parts.ended));
        }
    }
    hide_lease_ids(&lines.join("\n"))
}

/// What the command prints for `outcome`: with `json`, the compact line on
/// stdout whatever the exit; otherwise the text on stdout for exit 0 and on
/// stderr (`error:` painted red when fancy) for any other exit.
pub fn render(ui: &Ui, json: bool, outcome: &Outcome) -> Rendered {
    if json {
        return Rendered {
            text: hide_lease_ids(&compact_json(&outcome.body)),
            to_stderr: false,
        };
    }
    let text = hide_lease_ids(&outcome.text);
    if outcome.exit == 0 {
        Rendered {
            text,
            to_stderr: false,
        }
    } else if ui.fancy {
        Rendered {
            text: format!("{} {text}", ui.paint(Tone::Fail, "error:")),
            to_stderr: true,
        }
    } else {
        Rendered {
            text,
            to_stderr: true,
        }
    }
}
```
