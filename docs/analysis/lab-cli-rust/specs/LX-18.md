**Text edit helper** (docs and PowerShell files, CRLF or LF, whichever the file has). Paste it at the top of a PowerShell call:

```powershell
function Edit-Text([string]$Path, [string[]]$Old, [string[]]$New) {
    $t = [IO.File]::ReadAllText($Path)
    $nl = if ($t.Contains("`r`n")) { "`r`n" } else { "`n" }
    $o = $Old -join $nl
    if (-not $t.Contains($o)) { throw "old text not found in $Path`n$o" }
    $i = $t.IndexOf($o)
    if ($t.IndexOf($o, $i + 1) -ge 0) { throw "old text is not unique in $Path" }
    [IO.File]::WriteAllText($Path, $t.Replace($o, ($New -join $nl)), [Text.UTF8Encoding]::new($false))
}
```

`$Old`/`$New` are arrays of lines (no line endings). An empty `$New` (`@()`) deletes the lines.

## Step A4 (F-LX7): one helper makes `--specs-dir` absolute

In `src/daemon/paths.rs`, after `pub fn default_specs_dir()`, add:

```rust
/// `dir` made absolute against `cwd`, as text. A path that is already rooted
/// (a drive letter or a UNC prefix with either slash, or a leading slash) is
/// returned as it is, on every platform; anything else is joined to `cwd`.
/// The folder need not exist. The daemon runs in another directory, so a
/// relative spec folder must be sent as the folder the user meant.
pub fn absolute_in(dir: &str, cwd: &Path) -> String {
    if dir.is_empty() || is_rooted(dir) {
        return dir.to_string();
    }
    cwd.join(dir).display().to_string()
}

/// [`absolute_in`] against this process's current directory (`dir` unchanged
/// when that cannot be read). Every `--specs-dir` goes through it before any
/// daemon call.
pub fn absolute_specs_dir(dir: &str) -> String {
    match std::env::current_dir() {
        Ok(cwd) => absolute_in(dir, &cwd),
        Err(_) => dir.to_string(),
    }
}

fn is_rooted(dir: &str) -> bool {
    let b = dir.as_bytes();
    let sep = |c: u8| c == b'\\' || c == b'/';
    Path::new(dir).is_absolute()
        || (b.len() >= 2 && sep(b[0]) && sep(b[1]))
        || (b.len() >= 3 && b[0].is_ascii_alphabetic() && b[1] == b':' && sep(b[2]))
}
```

/// The bytes that put the terminal back: show the cursor, leave the alternate screen.
fn leave_sequence() -> String {
    format!("{}{}", ansi(Show), ansi(LeaveAlternateScreen))
}

/// Puts the terminal back: raw mode off first (a write to a closed console must
/// not leave the parent shell without its line editing), then the cursor and
/// the screen. Safe to run twice.
pub fn restore_terminal<W: Write>(out: &mut W, raw: bool) {
    if raw {
        let _ = terminal::disable_raw_mode();
    }
    let _ = out.write_all(leave_sequence().as_bytes());
    let _ = out.flush();
}

/// A panic hook, as the panic machinery stores it.
type Hook = Box<dyn Fn(&PanicHookInfo<'_>) + Send + Sync + 'static>;

/// A hook that runs `restore` and then `previous`: the terminal is back on the
/// normal screen before the default hook prints, so the message stays visible
/// (printed on the alternate screen it would vanish with the screen).
fn hook_with(restore: impl Fn() + Send + Sync + 'static, previous: Hook) -> Hook {
    Box::new(move |info| {
        restore();
        previous(info);
    })
}

static LIVE: AtomicBool = AtomicBool::new(false);
static LIVE_RAW: AtomicBool = AtomicBool::new(false);
static HOOK: Once = Once::new();

/// While alive, a panic restores the terminal before it is reported. The hook
/// is installed once per process and does nothing while no live view runs.
pub struct PanicHookGuard;

impl PanicHookGuard {
    /// Arms the hook; `raw` says raw mode is on.
    pub fn install(raw: bool) -> Self {
        LIVE_RAW.store(raw, Ordering::SeqCst);
        LIVE.store(true, Ordering::SeqCst);
        HOOK.call_once(|| {
            let previous = panic::take_hook();
            panic::set_hook(hook_with(
                || {
                    if LIVE.load(Ordering::SeqCst) {
                        restore_terminal(&mut io::stdout(), LIVE_RAW.load(Ordering::SeqCst));
                    }
                },
                previous,
            ));
        });
        PanicHookGuard
    }
}

impl Drop for PanicHookGuard {
    fn drop(&mut self) {
        LIVE.store(false, Ordering::SeqCst);
    }
}

/// The `pwsh` arguments: `-NoProfile -File <script> <leading...>`, then the raw
/// tokens in PowerShell form (see [`to_powershell_tokens`]).
pub fn command_line(script: &Path, leading: &[&str], raw: &[OsString]) -> Vec<OsString> {
    let mut args = vec![

        OsString::from("-File"),

/// The tokens as PowerShell spells them: `--runs-per-lease` becomes
/// `-RunsPerLease`, `--json` becomes `-Json` and `--value=-x` becomes
/// `-Value:-x`. Anything else stays as typed, and so does every token after a
/// bare `--` (the `--` itself too).
pub fn to_powershell_tokens(raw: &[OsString]) -> Vec<OsString> {
    let mut after_separator = false;
    raw.iter()
        .map(|arg| {
            if after_separator {
                return arg.clone();
            }
            let Some(text) = arg.to_str() else {
                return arg.clone();
            };
            if text == "--" {
                after_separator = true;
                return arg.clone();
            }
            match powershell_token(text) {
                Some(converted) => OsString::from(converted),
                None => arg.clone(),
            }
        })
        .collect()
}

/// The PowerShell form of one `--long` token, or `None` when it is not one.
fn powershell_token(text: &str) -> Option<String> {
    let body = text.strip_prefix("--")?;
    let (name, value) = match body.split_once('=') {
        Some((name, value)) => (name, Some(value)),
        None => (body, None),
    };
    if !is_long_name(name) {
        return None;
    }
    Some(match value {
        Some(value) => format!("-{}:{value}", pascal_case(name)),
        None => format!("-{}", pascal_case(name)),
    })
}

/// `[a-z][a-z0-9-]*`: a kebab-case long flag name.
fn is_long_name(name: &str) -> bool {
    let mut chars = name.chars();
    chars.next().is_some_and(|c| c.is_ascii_lowercase())
        && chars.all(|c| c.is_ascii_lowercase() || c.is_ascii_digit() || c == '-')
}

/// kebab-case to PascalCase: `runs-per-lease` becomes `RunsPerLease`.
fn pascal_case(name: &str) -> String {
    name.split('-')
        .map(|word| {
            let mut chars = word.chars();
            match chars.next() {
                Some(first) => first.to_ascii_uppercase().to_string() + chars.as_str(),
                None => String::new(),
            }
        })
        .collect()
}

/// Runs `pwsh -NoProfile -File <scripts dir>\<relative> <leading...> <ctx.raw
/// in PowerShell form>` with inherited stdio and returns the child's exit code
/// (255 when it has none). 1, with an error, when the script or `pwsh` is missing.
pub fn run_script(ctx: &Ctx, relative: &str, leading: &[&str]) -> u8 {
    let Some(script) = script_path(relative) else {
        ctx.ui.error(
            &format!("the PowerShell script {relative} is not installed"),
            Some("run: pwsh <checkout>\\tools\\lab\\cli\\setup.ps1"),
        );
        return 1;
    };
    match Process::new("pwsh")
        .args(command_line(&script, leading, &ctx.raw))
        .status()
    {
        Ok(status) => exit_byte(status.code()),
        Err(err) => {
            ctx.ui.error(
                &format!("could not start pwsh: {err}"),
                Some("install PowerShell 7 or put pwsh on PATH"),
            );
            1
        }
    }
}

/// The process exit code as one byte: passed through when it fits, else 255
/// (no code at all, a signal, or a value outside 0..=255).
pub fn exit_byte(code: Option<i32>) -> u8 {
    code.and_then(|code| u8::try_from(code).ok()).unwrap_or(255)
}

    /// The line was appended.
    Added,
    /// A line that runs `lab completions powershell` was already there.

/// Adds [`PROFILE_LINE`] to the profile at `path` unless a line that is not a
/// comment already runs `lab completions powershell`. Creates the folder and
/// the file when they are missing, keeps what is there, and uses the file's own
/// line ending.
pub fn install_into(path: &Path) -> io::Result<Installed> {
    let existing = match std::fs::read_to_string(path) {
        Ok(text) => text,
        Err(err) if err.kind() == io::ErrorKind::NotFound => String::new(),
        Err(err) => return Err(err),
    };
    let present = existing.lines().any(|line| {
        let line = line.trim_start();
        !line.starts_with('#') && line.contains("lab completions powershell")
    });
    if present {
        return Ok(Installed::AlreadyThere);
    }
    if let Some(parent) = path.parent() {
        std::fs::create_dir_all(parent)?;
    }
    let newline = if existing.contains("\r\n") || (existing.is_empty() && cfg!(windows)) {
        "\r\n"
    } else {
        "\n"
    };
    let mut text = String::new();
    if !existing.is_empty() && !existing.ends_with('\n') {
        text.push_str(newline);
    }
    text.push_str(&format!("{PROFILE_COMMENT}{newline}{PROFILE_LINE}{newline}"));
    let mut file = std::fs::OpenOptions::new()
        .create(true)
        .append(true)
        .open(path)?;
    file.write_all(text.as_bytes())?;
    Ok(Installed::Added)
}

/// `$PROFILE.CurrentUserAllHosts`, asked of `pwsh`.
pub fn pwsh_profile_path() -> Result<PathBuf, String> {
    let output = Process::new("pwsh")
        .args([
            "-NoProfile",
            "-NonInteractive",
            "-Command",
            "$PROFILE.CurrentUserAllHosts",
        ])
        .stdin(Stdio::null())
        .stderr(Stdio::null())
        .output()
        .map_err(|err| format!("could not start pwsh: {err}"))?;
    let text = String::from_utf8_lossy(&output.stdout).trim().to_string();
    if !output.status.success() || text.is_empty() {
        return Err("pwsh did not report its profile path".to_string());
    }
    Ok(PathBuf::from(text))
}

/// Runs `lab completions`: 0, or 1 when the script or the profile cannot be
/// written, or 2 for `--install` with a shell other than powershell.
pub fn run(args: &Args, ctx: &Ctx) -> u8 {

        return install(args, ctx);

        .write_all(script(args.shell).as_bytes())
        .and_then(|()| out.flush())

        Err(err) if err.kind() == io::ErrorKind::BrokenPipe => 0,
        Err(err) => {
            ctx.ui
                .error(&format!("cannot write the script: {err}"), None);
            1
        }
    }
}

## Step C1-1 (F-LX10): the lease wait takes a stop check and a clock

Today `wait_lease_free_with` sleeps `interval` (10 s) in one piece and knows nothing of Ctrl+C. After a run times out, `lane_body` waits up to 10 minutes for the lease. After the first Ctrl+C no new run starts, so that wait is pointless, yet it holds the batch (the summary is written only after the lanes end). The fix: one core, `wait_lease_free_until`, that checks a stop closure at the top of every poll and before every sleep slice (200 ms), and takes its clock and sleeper as a trait so a test can run it on a fake clock. `wait_lease_free_with` stays, as a thin wrapper that never stops, so `tests/mcp.rs` keeps passing unchanged.

```rust
pub fn wait_lease_free_with(
    poll: &mut dyn FnMut() -> Option<Value>,
    instance: &str,
    max: Duration,
    interval: Duration,
) -> bool {
    wait_lease_free_until(
        poll,
        instance,
        max,
        interval,
        &|| false,
        &mut RealClock::default(),
    )
}

/// The longest single sleep inside a lease wait: a stop request is seen within it.
pub const STOP_SLICE: Duration = Duration::from_millis(200);

/// The clock and the sleeper of a lease wait. The real one is [`RealClock`];
/// tests pass a fake one that moves time without sleeping.
pub trait LeaseClock {
    /// Time since the wait began.
    fn elapsed(&mut self) -> Duration;
    /// Sleeps for `d`.
    fn sleep(&mut self, d: Duration);
}

/// [`Instant`] and `thread::sleep`; the wait begins when it is made.
pub struct RealClock(Instant);

impl Default for RealClock {
    fn default() -> Self {
        RealClock(Instant::now())
    }
}

impl LeaseClock for RealClock {
    fn elapsed(&mut self) -> Duration {
        self.0.elapsed()
    }

    fn sleep(&mut self, d: Duration) {
        std::thread::sleep(d);
    }
}

/// Whether `instance`'s row of a `/status` body says its lease is held. An
/// instance missing from the body counts as free, as the PowerShell does.
fn lease_held(status: &Value, instance: &str) -> bool {
    status
        .get("instances")
        .and_then(Value::as_array)
        .and_then(|rows| {
            rows.iter()
                .find(|r| r.get("instance").and_then(Value::as_str) == Some(instance))
        })
        .and_then(|row| row.get("lease"))
        .and_then(|lease| lease.get("held"))
        .and_then(Value::as_bool)
        .unwrap_or(false)
}

/// [`wait_lease_free_with`] with a stop check and a clock. `stop` is asked at
/// the top of every round and before every sleep slice (each at most
/// [`STOP_SLICE`] long), so a `true` ends the wait within one slice and the
/// answer is `false` (the lease did not clear). A stop that is already `true`
/// polls nothing. The wait never sleeps past `max`; the poll after the last
/// sleep is the last look.
pub fn wait_lease_free_until(
    poll: &mut dyn FnMut() -> Option<Value>,
    instance: &str,
    max: Duration,
    interval: Duration,
    stop: &dyn Fn() -> bool,
    clock: &mut dyn LeaseClock,
) -> bool {
    loop {
        if stop() {
            return false;
        }
        if let Some(status) = poll() {
            if !lease_held(&status, instance) {
                return true;
            }
        }
        let left = max.saturating_sub(clock.elapsed());
        if left.is_zero() {
            return false;
        }
        let mut rest = interval.min(left);
        while !rest.is_zero() {
            if stop() {
                return false;
            }
            let slice = rest.min(STOP_SLICE);
            clock.sleep(slice);
            rest -= slice;
        }
    }
}
```

## Step C1-3 (F-LX10): a wait that ended on Ctrl+C is not a stuck lease

/// A clock that moves time without sleeping and calls `on_sleep(n)` after the
/// n-th sleep (1-based).
struct FakeClock<'a> {
    now: Duration,
    sleeps: Vec<Duration>,
    on_sleep: Box<dyn FnMut(usize) + 'a>,
}

impl LeaseClock for FakeClock<'_> {
    fn elapsed(&mut self) -> Duration {
        self.now
    }

    fn sleep(&mut self, d: Duration) {
        self.now += d;
        self.sleeps.push(d);
        (self.on_sleep)(self.sleeps.len());
    }
}

fn clock<'a>(on_sleep: impl FnMut(usize) + 'a) -> FakeClock<'a> {
    FakeClock {
        now: Duration::ZERO,
        sleeps: Vec::new(),
        on_sleep: Box::new(on_sleep),
    }
}

fn status(instance: &str, held: bool) -> Option<Value> {
    Some(json!({"instances": [{"instance": instance, "lease": {"held": held}}]}))

## Step C1-6 (F-LX11): `clip_text` counts UTF-16 units

PowerShell's `Limit-UatText` is `if ($Text.Length -le $Max) { return $Text }; return $Text.Substring(0, $Max - 3) + '...'`. `String.Length` counts UTF-16 code units, so an emoji (U+1F600) counts 2 and the cut falls between units. `clip_text` counts Unicode chars (an emoji is 1), so a line with emoji ends up longer than its budget (the 300 character golden value, the `--json` line budgets, the watch panel's 72). Rust cannot cut inside a surrogate pair (a `String` cannot hold half of one), so the rule is: count units, and cut before a character that would not fit whole.

/// `Limit-UatText`: `text` cut to `max` UTF-16 code units (PowerShell's
/// `String.Length`), ending in `...` when cut. A character outside the Basic
/// Multilingual Plane (an emoji) counts 2. The cut never splits such a pair:
/// a character that does not fit whole is left out, so the result is at most
/// `max` units (PowerShell would leave half a pair there).
pub fn clip_text(text: &str, max: usize) -> String {
    if text.encode_utf16().count() <= max {
        return text.to_string();
    }
    let keep = max.saturating_sub(3);
    let mut used = 0;
    let mut out = String::new();
    for c in text.chars() {
        let units = c.len_utf16();
        if used + units > keep {
            break;
        }
        used += units;
        out.push(c);
    }
    out.push_str("...");
    out
}

/// What a scripted `tools/call` answers.
#[derive(Debug, Clone)]
pub enum ToolReply {
    /// The tool's result: sent as the text of the first content item (the
    /// client parses it back into JSON).
    Json(Value),
    /// The tool ran and said no (`isError`), with this text.
    Error(String),
}

/// The MCP half of a stub: `GET /status` answers `status`; `POST /mcp` answers
/// `initialize` (with a session id), `notifications/initialized` (202) and
/// `tools/call` by tool name from `tools`, one scripted reply per call, in
/// order; `DELETE /mcp` answers 200. A call to a tool with nothing left in its
/// script gets an `isError` reply naming the tool.
pub struct McpScript {
    status: String,
    tools: Mutex<HashMap<String, VecDeque<ToolReply>>>,
}

impl McpScript {
    /// `(status, extra header lines, body)` for a request this script answers,
    /// else `None` (the stub's routes then decide).
    fn answer(&self, method: &str, path: &str, body: &str) -> Option<(u16, &'static str, String)> {
        if method.eq_ignore_ascii_case("GET") && path == "/status" {
            return Some((200, "", self.status.clone()));
        }
        if path != "/mcp" {
            return None;
        }
        if method.eq_ignore_ascii_case("DELETE") {
            return Some((200, "", String::new()));
        }
        if !method.eq_ignore_ascii_case("POST") {
            return None;
        }
        let request: Value = serde_json::from_str(body).unwrap_or(Value::Null);
        let id = request.get("id").cloned().unwrap_or(Value::Null);
        match request.get("method").and_then(Value::as_str) {
            Some("initialize") => Some((
                200,
                "Mcp-Session-Id: stub-session\r\n",
                json!({"jsonrpc": "2.0", "id": id, "result": {
                    "protocolVersion": "2025-06-18",
                    "capabilities": {},
                    "serverInfo": {"name": "stub", "version": "0"},
                }})
                .to_string(),
            )),
            Some("notifications/initialized") => Some((202, "", String::new())),
            Some("tools/call") => {
                let name = request
                    .pointer("/params/name")
                    .and_then(Value::as_str)
                    .unwrap_or("")
                    .to_string();
                let next = self
                    .tools
                    .lock()
                    .expect("script lock")
                    .get_mut(&name)
                    .and_then(VecDeque::pop_front);
                let result = match next {
                    Some(ToolReply::Json(v)) => {
                        json!({"content": [{"type": "text", "text": v.to_string()}]})
                    }
                    Some(ToolReply::Error(m)) => {
                        json!({"content": [{"type": "text", "text": m}], "isError": true})
                    }
                    None => json!({
                        "content": [{"type": "text", "text": format!("no scripted reply for {name}")}],
                        "isError": true,
                    }),
                };
                Some((
                    200,
                    "",
                    json!({"jsonrpc": "2.0", "id": id, "result": result}).to_string(),
                ))
            }
            _ => None,
        }
    }
}
