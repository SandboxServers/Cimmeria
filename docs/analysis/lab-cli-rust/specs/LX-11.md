## Decisions (already made; do not revisit)

- **The child is spelled the PowerShell way.** `record` runs `<this exe> uat <section> -Rows ... -Leases N -RunsPerLease M -ServerVersion SHA [-SpecsDir D] -Fingerprint -MaxConsecutiveFailures 1 -ProgressToStderr -Json`, the exact list `Get-GoldenBatchArgs` builds. `argv::rewrite` turns each into `--rows`, `--runs-per-lease` ... when `uat` is native, and the LX-01 fallback forwards the raw tokens to `uat.ps1` while `uat` is still PowerShell. So one list works in both states. (The packet text writes `uat <section> --fingerprint ...`; that spelling would break the PowerShell `uat`.)
- **The child's stderr is inherited** (`Stdio::inherit()`), which streams it live; only stdout is captured.
- **Messages use `--flag` names** (`--runs must be 5 to 20 ...`, `a golden exists; re-blessing needs --rebless and five fresh runs`), not `-Runs`. Everything else in plain mode is the PowerShell text.
- **JSON is built from `#[derive(Serialize)]` structs** (field order = key order; absent keys are skipped). serde_json does not escape `<`, `>`, `&`, `'` the way PowerShell's `ConvertTo-Json` does; consumers parse the JSON, so that is accepted.
- **Rust has no `Get-LabToken` throw.** `Endpoint::token == None` or empty gives the "not answering" message; the "cannot reach the lab daemon" message of the PowerShell has no Rust equivalent and is dropped.
- **Keep `tools/lab/cli/golden-lib.ps1`.** `uat.ps1` line 135 dot-sources it and line 171 calls `Get-GoldenPath`. LX-17 deletes it. Delete `golden.ps1` and `test-golden.ps1` (every check is ported below).
- **`tools/README.md` row for `golden.ps1` is inside `<!-- gen:tools-catalog -->`: do not edit it** (the regen workflow owns it).

```rust
//! Argument rules, the golden file's path and the `--rows` cleanup.
//! Ports `Test-GoldenArguments` and `Get-GoldenPath` (golden-lib.ps1) and

use std::path::{Path, PathBuf};

/// The owner's rule is "golden only when five runs agree" (README D-GD6).
pub const MIN_RUNS: u32 = 5;
pub const MAX_RUNS: u32 = 20;
pub const MAX_LEASES: u32 = 5;

/// The argument checks that need no daemon: an error message, or `None`.
pub fn validate_arguments(
    action: &str,
    section: &str,
    runs: u32,
    leases: u32,
    server_version: &str,
) -> Option<String> {
    if action != "record" && action != "show" {
        return Some("usage: lab golden record|show <section> ...".to_string());
    }
    if section.is_empty() {
        return Some(format!("usage: lab golden {action} <section>"));
    }
    // The section names a file under golden/, so it may not reach outside it.
    if !is_section_id(section) || section.contains("..") {
        return Some(format!(
            "section '{section}' is not a section id (letters, digits, '.', '_' and '-')"
        ));
    }
    if !(MIN_RUNS..=MAX_RUNS).contains(&runs) {
        return Some(format!(
            "--runs must be {MIN_RUNS} to {MAX_RUNS} (a golden needs five agreeing runs)"
        ));
    }
    if !(1..=MAX_LEASES).contains(&leases) {
        return Some(format!("--leases must be 1 to {MAX_LEASES}"));
    }
    if !runs.is_multiple_of(leases) {
        return Some(format!(
            "--runs {runs} does not split evenly over --leases {leases}"
        ));
    }
    if action == "record" && !is_build_sha(server_version) {
        return Some(
            "--server-version must be the server build's commit (7 to 40 lowercase hex characters)"
                .to_string(),
        );
    }
    None
}

/// `^[A-Za-z0-9][A-Za-z0-9._-]*$`, ASCII only and case-sensitive.
fn is_section_id(section: &str) -> bool {
    let mut chars = section.chars();
    let Some(first) = chars.next() else {
        return false;
    };
    first.is_ascii_alphanumeric()
        && chars.all(|c| c.is_ascii_alphanumeric() || matches!(c, '.' | '_' | '-'))
}

/// `^[0-9a-f]{7,40}$`.
fn is_build_sha(value: &str) -> bool {
    (7..=40).contains(&value.len())
        && value
            .bytes()
            .all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b))
}

/// The committed golden of a section under a spec folder, named as the
/// daemon names it (`evidence.rs path_safe`): every character outside
/// `[A-Za-z0-9_-]` becomes `_`, so `v1.2` is `golden/v1_2.json`.
pub fn golden_path(specs_dir: &Path, section: &str) -> PathBuf {
    let safe: String = section
        .chars()
        .map(|c| {
            if c.is_ascii_alphanumeric() || c == '_' || c == '-' {
                c
            } else {
                '_'
            }
        })
        .collect();
    specs_dir.join("golden").join(format!("{safe}.json"))
}

/// `--rows` as clean ids: split on commas, trimmed, blanks and repeats
/// (compared ignoring case, as PowerShell's `-notcontains` does) dropped.
pub fn row_ids(rows: &[String]) -> Vec<String> {
    let mut ids: Vec<String> = Vec::new();
    for raw in rows {
        for part in raw.split(',') {
            let id = part.trim();
            if !id.is_empty() && !ids.iter().any(|seen| seen.eq_ignore_ascii_case(id)) {
                ids.push(id.to_string());
            }
        }
    }
    ids
}
```

```rust
//! The `lab_golden_record` reply as an exit code, compact JSON and the line
//! `lab golden record` prints. Ports `ConvertTo-GoldenExit`,
//! `ConvertTo-GoldenCompactJson` and `Format-GoldenRecordLine`.

use serde::Serialize;
use serde_json::Value;

use crate::ui::{Mark, Tone, Ui};

/// A JSON value as text: `null` and a missing key are empty, a string is
/// itself, anything else its JSON text (what PowerShell's `"$v"` gives).
pub fn text_of(value: Option<&Value>) -> String {
    match value {
        None | Some(Value::Null) => String::new(),
        Some(Value::String(s)) => s.clone(),
        Some(other) => other.to_string(),
    }
}

/// The first line of an error text.
pub fn first_line(text: &str) -> &str {
    text.lines().next().unwrap_or("")
}

/// `lab_golden_record`'s reply as the command's exit code (README Review
/// outcomes, GD-07): `ok` 0; `exists` 2; `disagree` 6; `not_passing` 1;
/// `cannot_record` and every other code, or none, 7.
pub fn reply_exit(reply: &Value) -> u8 {
    if reply.get("ok").and_then(Value::as_bool) == Some(true) {
        return 0;
    }
    match text_of(reply.get("code")).as_str() {
        "exists" => 2,
        "disagree" => 6,
        "not_passing" => 1,
        _ => 7,
    }
}

/// A reply field that is neither missing, `null` nor an empty string.
fn present<'a>(reply: &'a Value, key: &str) -> Option<&'a Value> {
    reply
        .get(key)
        .filter(|v| !v.is_null() && v.as_str() != Some(""))
}

/// Key order is the PowerShell's: `ok, exit`, then on success `path, rows,
/// events, bytes, runs`, else `code, error, batch`. Absent keys are skipped.
#[derive(Serialize)]
struct Compact<'a> {
    ok: bool,
    exit: u8,
    #[serde(skip_serializing_if = "Option::is_none")]
    path: Option<&'a Value>,
    #[serde(skip_serializing_if = "Option::is_none")]
    rows: Option<&'a Value>,
    #[serde(skip_serializing_if = "Option::is_none")]
    events: Option<&'a Value>,
    #[serde(skip_serializing_if = "Option::is_none")]
    bytes: Option<&'a Value>,
    #[serde(skip_serializing_if = "Option::is_none")]
    runs: Option<&'a Value>,
    #[serde(skip_serializing_if = "Option::is_none")]
    code: Option<&'a Value>,
    #[serde(skip_serializing_if = "Option::is_none")]
    error: Option<&'a Value>,
    #[serde(skip_serializing_if = "Option::is_none")]
    batch: Option<&'a str>,
}

/// The reply as one compact JSON object with no nulls. A refusal after the
/// runs names the batch folder they are in.
pub fn compact_json(reply: &Value, exit: u8, batch: Option<&str>) -> String {
    let success = exit == 0;
    let pick = |key: &str, on_success: bool| {
        if on_success == success {
            present(reply, key)
        } else {
            None
        }
    };
    let compact = Compact {
        ok: success,
        exit,
        path: pick("path", true),
        rows: pick("rows", true),
        events: pick("events", true),
        bytes: pick("bytes", true),
        runs: pick("runs", true),
        code: pick("code", false),
        error: pick("error", false),
        batch: if success {
            None
        } else {
            batch.filter(|b| !b.is_empty())
        },
    };
    serde_json::to_string(&compact).unwrap_or_default()
}

/// The one line `lab golden record` prints on success (plain mode).
pub fn record_line(section: &str, reply: &Value) -> String {
    let f = |key: &str| text_of(reply.get(key));
    format!(
        "golden {section}: {} rows, {} events, {} bytes from {} runs; {}",
        f("rows"),
        f("events"),
        f("bytes"),
        f("runs"),
        f("path")
    )
}

/// Fancy mode: a green check and the counts, then the path in cyan.
pub fn record_line_fancy(ui: &Ui, section: &str, reply: &Value) -> String {
    let f = |key: &str| text_of(reply.get(key));
    let head = format!(
        "{} golden {section}: {} rows, {} events from {} runs",
        ui.paint(Tone::Ok, ui.mark(Mark::Pass)),
        f("rows"),
        f("events"),
        f("runs")
    );
    format!("{head}\n  {}", ui.paint(Tone::Cmd, &f("path")))
}
```

```rust
//! The child `lab uat` batch: its arguments, how it is started, its result
//! and the run folders of its `batch.jsonl`. Ports `Get-GoldenBatchArgs`,
//! `Get-GoldenRunDirs`, `ConvertFrom-GoldenBatchOutput` (golden-lib.ps1).

use std::io;
use std::path::Path;
use std::process::{Command, Stdio};

use serde_json::Value;

use super::reply::text_of;

/// What the child printed on stdout and how it exited (`-1` when it had no
/// exit code). Its stderr is not captured: it reaches our stderr live.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct ChildOutput {
    pub stdout: String,
    pub code: i32,
}

/// The child `lab uat` arguments for a golden batch: fingerprint runs, `runs`
/// split evenly over `leases`, a lane brake of one failed run, progress on
/// stderr and one JSON line on stdout. They are spelled as the PowerShell
/// `uat` takes them (`argv::rewrite` maps them for the native one). Each value
/// is its own element, so a path with spaces stays one argument.
pub fn batch_args(
    section: &str,
    rows: &[String],
    runs: u32,
    leases: u32,
    server_version: &str,
    specs_dir: Option<&str>,
) -> Result<Vec<String>, String> {
    if leases < 1 || !runs.is_multiple_of(leases) {
        return Err(format!(
            "--runs {runs} does not split evenly over --leases {leases}"
        ));
    }
    let mut args = vec![section.to_string()];
    let rows: Vec<&str> = rows
        .iter()
        .map(String::as_str)
        .filter(|r| !r.is_empty())
        .collect();
    if !rows.is_empty() {
        args.push("-Rows".to_string());
        args.push(rows.join(","));
    }
    args.push("-Leases".to_string());
    args.push(leases.to_string());
    args.push("-RunsPerLease".to_string());
    args.push((runs / leases).to_string());
    args.push("-ServerVersion".to_string());
    args.push(server_version.to_string());
    if let Some(dir) = specs_dir.filter(|d| !d.is_empty()) {
        args.push("-SpecsDir".to_string());
        args.push(dir.to_string());
    }
    for fixed in [
        "-Fingerprint",
        "-MaxConsecutiveFailures",
        "1",
        "-ProgressToStderr",
        "-Json",
    ] {
        args.push(fixed.to_string());
    }
    Ok(args)
}

/// Runs `<this exe> uat <args>`: stdout captured, stderr inherited (so the
/// per-run progress lines show live), stdin closed.
pub fn spawn_uat(args: &[String]) -> io::Result<ChildOutput> {
    let exe = std::env::current_exe()?;
    let output = Command::new(exe)
        .arg("uat")
        .args(args)
        .stdin(Stdio::null())
        .stderr(Stdio::inherit())
        .output()?;
    Ok(ChildOutput {
        stdout: String::from_utf8_lossy(&output.stdout).into_owned(),
        code: output.status.code().unwrap_or(-1),
    })
}

/// The records of a batch's `batch.jsonl` (none when the file is missing).
/// One compact JSON object per non-blank line. LX-13 adds a shared reader in
/// `batch/mod.rs`; switch to it then.
pub fn read_records(path: &Path) -> Result<Vec<Value>, String> {
    if !path.exists() {
        return Ok(Vec::new());
    }
    let text = std::fs::read_to_string(path).map_err(|e| e.to_string())?;
    let mut records = Vec::new();
    for (index, line) in text.lines().enumerate() {
        let line = line.trim();
        if line.is_empty() {
            continue;
        }
        let record: Value =
            serde_json::from_str(line).map_err(|e| format!("line {}: {e}", index + 1))?;
        records.push(record);
    }
    Ok(records)
}

/// The `RunDir` of every record, in order; `None` when there are no records,
/// or any record did not pass (`Ok` is not `true`) or has no `RunDir` (every
/// run must pass). `null` records are skipped.
pub fn run_dirs(records: &[Value]) -> Option<Vec<String>> {
    let mut dirs = Vec::new();
    for record in records {
        if record.is_null() {
            continue;
        }
        let dir = record.get("RunDir").and_then(Value::as_str).unwrap_or("");
        let passed = record.get("Ok").and_then(Value::as_bool) == Some(true);
        if !passed || dir.is_empty() {
            return None;
        }
        dirs.push(dir.to_string());
    }
    if dirs.is_empty() {
        None
    } else {
        Some(dirs)
    }
}

/// The child's result: the exit code `golden` ends with, the message, and the
/// batch folder when there is one.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct BatchOutcome {
    pub exit: u8,
    pub error: Option<String>,
    pub batch: Option<String>,
}

/// The child `lab uat --json` result. Its exit 0 gives the batch folder; 2 and
/// 3 pass through with its error; 1 and 4 are exit 1 with the first failure
/// group, because every run must pass (D-GD6). `lines` are the child's
/// stdout; its JSON object is the last line that starts with `{` and parses.
pub fn parse_batch_output(lines: &[String], exit_code: i32) -> BatchOutcome {
    let mut obj: Option<Value> = None;
    for line in lines {
        let trimmed = line.trim();
        if trimmed.starts_with('{') {
            if let Ok(parsed) = serde_json::from_str::<Value>(trimmed) {
                obj = Some(parsed);
            }
        }
    }
    let Some(obj) = obj else {
        return BatchOutcome {
            exit: 3,
            error: Some(format!("lab uat exited {exit_code} without a result")),
            batch: None,
        };
    };
    let batch = text_of(obj.get("batch"));
    let batch_opt = if batch.is_empty() {
        None
    } else {
        Some(batch.clone())
    };
    match exit_code {
        0 => {
            if batch.is_empty() {
                BatchOutcome {
                    exit: 3,
                    error: Some("lab uat passed but named no batch folder".to_string()),
                    batch: None,
                }
            } else {
                BatchOutcome { exit: 0, error: None, batch: batch_opt }
            }
        }
        2 | 3 => BatchOutcome {
            exit: u8::try_from(exit_code).unwrap_or(3),
            error: Some(text_of(obj.get("error"))),
            batch: None,
        },
        1 | 4 => {
            let first = obj
                .get("failures")
                .and_then(Value::as_array)
                .and_then(|failures| failures.first())
                .filter(|f| !f.is_null());
            let why = match first {
                Some(failure) => {
                    let runs = failure
                        .get("runs")
                        .and_then(Value::as_array)
                        .map(|runs| {
                            runs.iter()
                                .map(|r| text_of(Some(r)))
                                .collect::<Vec<_>>()
                                .join(", ")
                        })
                        .unwrap_or_default();
                    let mut why = text_of(failure.get("why"));
                    if !runs.is_empty() {
                        why.push_str(&format!(" (runs: {runs})"));
                    }
                    why
                }
                None => text_of(obj.get("error")),
            };
            let passed = format!("{}/{}", text_of(obj.get("passed")), text_of(obj.get("runs")));
            let mut message = format!(
                "every run must pass to record a golden; {passed} passed, first failure: {why}"
            );
            if !batch.is_empty() {
                message.push_str("; ");
                message.push_str(&batch);
            }
            BatchOutcome { exit: 1, error: Some(message), batch: batch_opt }
        }
        other => BatchOutcome {
            exit: 3,
            error: Some(format!("lab uat exited {other}")),
            batch: batch_opt,
        },
    }
}
```

```rust
//! `Get-GoldenSummary` and `Format-GoldenSummary` (golden-lib.ps1).

use serde::Serialize;
use serde_json::Value;

use super::reply::text_of;
use crate::ui::{Tone, Ui};

/// Section, rows, total ordered events, build, recorded time and evidence run
/// ids. Field order is the JSON key order; absent options are left out.
#[derive(Debug, Clone, PartialEq, Eq, Serialize)]
pub struct Summary {
    pub section: String,
    pub rows: Vec<String>,
    pub events: usize,
    pub server: String,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub sgw_sha256: Option<String>,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub recorded: Option<String>,
    pub runs: Vec<String>,
}

fn truthy(value: &Value) -> bool {
    !value.is_null() && value.as_str() != Some("")
}

fn non_empty(value: Option<&Value>) -> Option<String> {
    let text = text_of(value);
    if text.is_empty() {
        None
    } else {
        Some(text)
    }
}

/// The summary of a parsed golden file.
pub fn summarize(golden: &Value) -> Summary {
    let events = golden
        .pointer("/fingerprint/rows")
        .and_then(Value::as_array)
        .map(|rows| {
            rows.iter()
                .filter(|row| !row.is_null())
                .map(|row| {
                    row.get("events")
                        .and_then(Value::as_array)
                        .map(|events| events.iter().filter(|e| !e.is_null()).count())
                        .unwrap_or(0)
                })
                .sum()
        })
        .unwrap_or(0);
    let rows = golden
        .get("rows")
        .and_then(Value::as_array)
        .map(|rows| {
            rows.iter()
                .filter(|r| truthy(r))
                .map(|r| text_of(Some(r)))
                .collect()
        })
        .unwrap_or_default();
    let runs = golden
        .get("evidence")
        .and_then(Value::as_array)
        .map(|evidence| {
            evidence
                .iter()
                .filter(|e| !e.is_null())
                .map(|e| text_of(e.get("run_id")))
                .collect()
        })
        .unwrap_or_default();
    Summary {
        section: text_of(golden.get("section")),
        rows,
        events,
        server: text_of(golden.pointer("/build/server")),
        sgw_sha256: non_empty(golden.pointer("/build/sgw_sha256")),
        recorded: non_empty(golden.get("recorded_utc")),
        runs,
    }
}

/// The `--json` object of `show`: `{"ok":true,"exit":0,` then the summary keys.
pub fn show_json(summary: &Summary) -> String {
    #[derive(Serialize)]
    struct Shown<'a> {
        ok: bool,
        exit: u8,
        #[serde(flatten)]
        summary: &'a Summary,
    }
    serde_json::to_string(&Shown { ok: true, exit: 0, summary }).unwrap_or_default()
}

/// Plain mode: the lines the PowerShell printed, joined by `\n`.
pub fn summary_plain(s: &Summary) -> String {
    let mut lines = vec![
        format!("golden {}: {} rows, {} events", s.section, s.rows.len(), s.events),
        format!("rows:   {}", s.rows.join(", ")),
        format!("server: {}", s.server),
    ];
    if let Some(sgw) = &s.sgw_sha256 {
        lines.push(format!("sgw:    {sgw}"));
    }
    if let Some(when) = &s.recorded {
        lines.push(format!("when:   {when}"));
    }
    lines.push(format!("runs:   {}", s.runs.join(", ")));
    lines.join("\n")
}

/// Fancy mode: the section as a bold heading, then a dim-keyed block.
pub fn summary_fancy(ui: &Ui, s: &Summary) -> String {
    let mut lines = vec![format!(
        "{}  {}",
        ui.paint(Tone::Bold, &format!("golden {}", s.section)),
        ui.paint(Tone::Dim, &format!("{} rows · {} events", s.rows.len(), s.events))
    )];
    let mut line = |key: &str, value: &str| {
        lines.push(format!("  {}  {value}", ui.paint(Tone::Dim, &format!("{key:<6}"))));
    };
    line("rows", &s.rows.join(", "));
    line("server", &s.server);
    if let Some(sgw) = &s.sgw_sha256 {
        line("sgw", sgw);
    }
    if let Some(when) = &s.recorded {
        line("when", when);
    }
    line("runs", &s.runs.join(", "));
    lines.join("\n")
}
```

```rust
//! `lab golden` itself. Everything outside the process (the child `uat`, the
//! daemon endpoint, the default spec folder, stdout and stderr) is passed in,
//! so tests stub it. Ports `golden.ps1`.

use std::io::{self, Write};
use std::path::{Path, PathBuf};
use std::time::Duration;

use serde_json::{json, Value};

/// `Invoke-McpTool ... 300`: the tool reads the run bundles, so it can take a while.
const TOOL_TIMEOUT: Duration = Duration::from_secs(300);

/// What `execute` needs from outside.
pub struct Deps<'a> {
    /// Runs `<this exe> uat <args>` and returns its stdout and exit code.
    pub run_uat: &'a dyn Fn(&[String]) -> io::Result<ChildOutput>,
    /// The daemon `lab_golden_record` is sent to.

/// `{"ok":false,"exit":N,"error":...[,"batch":...]}`.
#[derive(Serialize)]
struct StopJson<'a> {
    ok: bool,
    exit: u8,
    error: &'a str,
    #[serde(skip_serializing_if = "Option::is_none")]
    batch: Option<&'a str>,
}

struct Io<'a> {
    ui: &'a Ui,
    json: bool,
    out: &'a mut dyn Write,
    err: &'a mut dyn Write,
}

impl Io<'_> {
    /// A failure: plain text on stderr (fancy: after a red `error:`), or under
    /// `--json` one object on stdout naming the batch folder once the runs
    /// exist. Lease ids never leave. Returns the exit code.
    fn stop(&mut self, code: u8, message: &str, batch: Option<&str>) -> u8 {
        let message = hide_lease_ids(message);
        if self.json {
            let body = StopJson {
                ok: false,
                exit: code,
                error: &message,
                batch: batch.filter(|b| !b.is_empty()),
            };
            let text = serde_json::to_string(&body).unwrap_or_default();
            let _ = writeln!(self.out, "{text}");
        } else {
            self.fail_line(&message);
        }
        code
    }

    /// One failure line on stderr; plain mode prints the bare message.
    fn fail_line(&mut self, message: &str) {
        let text = if self.ui.fancy {
            format!("{} {message}", self.ui.paint(Tone::Fail, "error:"))
        } else {
            message.to_string()
        };
        let _ = writeln!(self.err, "{text}");
    }
}

/// Runs `lab golden` and returns its exit code: 0 recorded (or shown); 1 a run
/// or row did not pass; 2 usage, or a golden exists without `--rebless`
/// (show: no golden); 3 the `lab uat` pre-flight or plan failed (nothing
/// driven); 6 the runs disagree; 7 the runs cannot be recorded as given, the
/// daemon failed after the runs, or (show) the golden cannot be read.
pub fn execute(
    args: &Args,
    ui: &Ui,
    deps: &Deps,
    out: &mut dyn Write,
    err: &mut dyn Write,
) -> u8 {
    let action = args.action.as_deref().unwrap_or("");
    let section = args.section.as_deref().unwrap_or("");
    let server_version = args.server_version.as_deref().unwrap_or("");
    let mut io = Io { ui, json: args.json, out, err };
    if let Some(bad) =
        check::validate_arguments(action, section, args.runs, args.leases, server_version)
    {
        return io.stop(2, &bad, None);
    }
    let specs: Option<PathBuf> = args
        .specs_dir
        .as_deref()
        .filter(|d| !d.is_empty())
        .map(PathBuf::from)
        .or_else(|| deps.default_specs.clone());
    if action == "show" {
        return show(&mut io, section, specs.as_deref());
    }
    record(&mut io, args, deps, section, server_version, specs.as_deref())
}

/// `lab golden show`: no daemon.
fn show(io: &mut Io, section: &str, specs: Option<&Path>) -> u8 {
    let Some(specs) = specs else {
        return io.stop(2, "no spec folder here; pass --specs-dir", None);
    };
    let path = check::golden_path(specs, section);
    if !path.exists() {
        let message = format!(
            "no golden for {section} ({}); record one with lab golden record",
            path.display()
        );
        return io.stop(2, &message, None);
    }
    let parsed: Result<Value, String> = std::fs::read_to_string(&path)
        .map_err(|e| e.to_string())
        .and_then(|text| {
            serde_json::from_str(text.trim_start_matches('\u{feff}')).map_err(|e| e.to_string())
        });
    let golden = match parsed {
        Ok(golden) => golden,
        Err(e) => {
            let message = format!("cannot read {}: {}", path.display(), first_line(&e));
            return io.stop(7, &message, None);
        }
    };
    let summary = summary::summarize(&golden);
    let text = if io.json {
        summary::show_json(&summary)
    } else if io.ui.fancy {
        summary::summary_fancy(io.ui, &summary)
    } else {
        summary::summary_plain(&summary)
    };
    let _ = writeln!(io.out, "{text}");
    0
}

/// `lab golden record`: the batch, then `lab_golden_record`.
fn record(
    io: &mut Io,
    args: &Args,
    deps: &Deps,
    section: &str,
    server_version: &str,
    specs: Option<&Path>,
) -> u8 {
    // Away from a checkout the daemon picks its own spec folder, and its
    // `exists` refusal says the same thing after the runs.
    if let Some(specs) = specs {
        if !args.rebless && check::golden_path(specs, section).exists() {
            return io.stop(
                2,
                "a golden exists; re-blessing needs --rebless and five fresh runs",
                None,
            );
        }
    }
    let requested = check::row_ids(&args.rows);
    let specs_text = specs.map(|p| p.to_string_lossy().into_owned());
    let child_args = match batch::batch_args(
        section,
        &requested,
        args.runs,
        args.leases,
        server_version,
        specs_text.as_deref(),
    ) {
        Ok(child_args) => child_args,
        Err(message) => return io.stop(2, &message, None),
    };
    if !io.json {
        let line = format!(
            "lab golden record {section}: {} fingerprint run(s) on {} lane(s), then lab_golden_record (a run takes a few minutes)",
            args.runs, args.leases
        );
        let text = if io.ui.fancy {
            format!("{} {line}", io.ui.paint(Tone::Cmd, io.ui.mark(Mark::Active)))
        } else {
            line
        };
        let _ = writeln!(io.out, "{text}");
    }
    let child = match (deps.run_uat)(&child_args) {
        Ok(child) => child,
        Err(e) => return io.stop(3, &format!("lab uat could not start: {e}"), None),
    };
    let lines: Vec<String> = child.stdout.lines().map(str::to_owned).collect();
    let outcome = batch::parse_batch_output(&lines, child.code);
    if outcome.exit != 0 {
        return io.stop(
            outcome.exit,
            outcome.error.as_deref().unwrap_or(""),
            outcome.batch.as_deref(),
        );
    }
    let batch_dir = outcome.batch.unwrap_or_default();

    // Run folders come from batch.jsonl: --json names only the batch (F16).
    let records = match batch::read_records(&Path::new(&batch_dir).join("batch.jsonl")) {
        Ok(records) => records,
        Err(e) => {
            let message = format!("cannot read the batch's runs: {}; {batch_dir}", first_line(&e));
            return io.stop(7, &message, Some(&batch_dir));
        }
    };
    let Some(dirs) = batch::run_dirs(&records) else {
        let message = format!("the batch passed but a run has no run folder; {batch_dir}");
        return io.stop(7, &message, Some(&batch_dir));
    };
    let mut tool_args = json!({"section": section, "run_dirs": dirs, "rebless": args.rebless});
    if !requested.is_empty() {
        tool_args["rows"] = json!(requested);
    }
    if let Some(specs) = &specs_text {
        tool_args["specs_dir"] = json!(specs);
    }
    if deps.endpoint.token.as_deref().is_none_or(str::is_empty) {
        let message = format!(
            "the lab daemon is not answering (lab status); the runs are in {batch_dir}"
        );
        return io.stop(7, &message, Some(&batch_dir));
    }
    let reply = match call_record(&deps.endpoint, tool_args) {
        Ok(reply) => reply,
        Err(e) => {
            let message = format!(
                "lab_golden_record failed: {}; the runs are in {batch_dir}",
                first_line(&e.to_string())
            );
            return io.stop(7, &message, Some(&batch_dir));
        }
    };
    let code = reply::reply_exit(&reply);
    if io.json {
        let text = hide_lease_ids(&reply::compact_json(&reply, code, Some(&batch_dir)));
        let _ = writeln!(io.out, "{text}");
    } else if code == 0 {
        let text = if io.ui.fancy {
            reply::record_line_fancy(io.ui, section, &reply)
        } else {
            reply::record_line(section, &reply)
        };
        let _ = writeln!(io.out, "{text}");
    } else {
        let message = format!(
            "{}: {}; the runs are in {batch_dir}",
            text_of(reply.get("code")),
            text_of(reply.get("error"))
        );
        io.fail_line(&hide_lease_ids(&message));
    }
    code
}

/// One `lab_golden_record` call on its own MCP session.
fn call_record(endpoint: &Endpoint, tool_args: Value) -> Result<Value, McpError> {
    let mut client = McpClient::connect(endpoint)?;
    let reply = client.call_tool("lab_golden_record", tool_args, TOOL_TIMEOUT);
    client.close();
    reply
}
```

```rust
/// A batch folder holding `batch.jsonl` with `n` passing run records.
fn batch_dir(n: usize) -> TempDir {
    let dir = TempDir::new().unwrap();
    let lines: String = (1..=n)
        .map(|i| {
            format!(
                "{}\n",
                json!({"Lane": 1, "Instance": "default", "Index": i, "Ok": true,
                       "Rows": [], "RunDir": format!("C:\\runs\\run-{i}"), "Error": null})
            )
        })
        .collect();
    std::fs::write(dir.path().join("batch.jsonl"), lines).unwrap();
    dir
}

fn uat_ok(batch: &TempDir) -> ChildOutput {
    let line = json!({"ok": true, "runs": 5, "passed": 5, "failed": 0,
                      "batch": path_text(batch)});
    ChildOutput { stdout: format!("{line}\n"), code: 0 }
}

/// The body every stub POST /mcp answers: a tools/call result whose text is `reply`.
fn mcp_body(reply: &Value) -> String {
    json!({"jsonrpc": "2.0", "id": 2,
           "result": {"content": [{"type": "text", "text": reply.to_string()}]}})
    .to_string()
}

fn ok_reply() -> Value {
    json!({"ok": true, "path": "C:\\specs\\golden\\first-session.json",
           "rows": 7, "events": 312, "bytes": 21456, "runs": 5})
}

fn run(a: &Args, deps: &Deps) -> (u8, String, String) {
    let (mut out, mut err) = (Vec::new(), Vec::new());
    let code = execute(a, &Ui::plain(), deps, &mut out, &mut err);
    (code, String::from_utf8(out).unwrap(), String::from_utf8(err).unwrap())
}

fn deps<'a>(
    run_uat: &'a dyn Fn(&[String]) -> io::Result<ChildOutput>,
    endpoint: Endpoint,
    specs: &TempDir,
) -> Deps<'a> {
    Deps { run_uat, endpoint, default_specs: Some(specs.path().to_path_buf()) }
}

/// The `tools/call` request the stub received, as JSON.
fn tool_call(stub: &StubServer) -> Value {
    let requests = stub.requests();
    let call = requests
        .iter()
        .find(|r| r.method == "POST" && r.body.contains("tools/call"))
        .expect("a tools/call request");
    assert_eq!(call.header("authorization"), Some(format!("Bearer {STUB_TOKEN}").as_str()));
    serde_json::from_str(&call.body).unwrap()
}
```
