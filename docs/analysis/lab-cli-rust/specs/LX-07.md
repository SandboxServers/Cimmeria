## Behaviour (what the PowerShell command did; the Rust port matches it)

Plain mode (`Ui.fancy == false`, i.e. any redirected run) prints exactly these
texts. Stdout lines end in `\n`.

| Invocation | Stdout | Stderr | Exit |
|---|---|---|---|
| `lab env`, labd.env missing | | `no labd.env at <path>` | 1 |
| `lab env`, labd.env present | one line per file line, `env_file::mask_line` text (a blank line prints an empty line) | | 0 |
| `lab env <verb>` with a verb other than get/set/unset (any case) | | `unknown verb '<verb as typed>'` then the usage line | 2 |
| a verb, but KEY missing or not `^[A-Z][A-Z0-9_]*$` | | `KEY must match ^[A-Z][A-Z0-9_]*$` then the usage line | 2 |
| `get KEY`, key set (any case in the file) | `env_file::mask(KEY, value)` | | 0 |
| `get KEY`, key not set or no file | | `<KEY> is not set` | 1 |
| `set KEY` with no VALUE (neither the third positional nor `--value`) | | `set needs KEY and VALUE` then the usage line | 2 |
| `set KEY VALUE`, VALUE has `\r` or `\n` | | `VALUE must be one line` | 2 |
| `set KEY VALUE`, VALUE differs from its trimmed form | | `VALUE must not start or end with spaces (labd.env lines are trimmed when read)` | 2 |
| `set KEY VALUE` ok | `backup: <backup path>` (only when a labd.env existed) then `restart the daemon for this to apply: lab restart` | | 0 |
| `unset KEY`, key not set | `<KEY> is not set; labd.env unchanged` (nothing written, no backup) | | 0 |
| `unset KEY` ok | the same two lines as `set` | | 0 |

The usage line is exactly:
`usage: lab env [get KEY | set KEY VALUE | unset KEY]  (a VALUE starting with '-': -Value:<value>)`
(two spaces before the parenthesis). A usage error prints the message, a newline,
then the usage line, then a newline, to stderr.

Rust-only additions (not in the PowerShell): a labd.env that exists but cannot be
read prints `cannot read <path>: <io error>` and exits 1; a failed write prints
`cannot write <path>: <io error>` and exits 1; a stray fourth positional or a
positional VALUE together with `--value` is a clap usage error, exit 2.

Fancy mode (`Ui.fancy`) uses the D-LX5 style: keys cyan, masked values dim,
comments and unparsed placeholders dim; success lines start with a green `✓`;
a warning starts with a yellow `!`; errors are `error: <message>` in bold red
with an indented dim `hint: <hint>`; backup and hint lines are indented two spaces
and dim; the restart hint is `hint: run: lab restart`.

```rust
/// What a `lab env` call produced: the text for each stream and the exit code.
#[derive(Debug)]
pub(crate) struct Outcome {
    pub code: u8,
    pub out: String,
    pub err: String,
}

impl Outcome {
    fn ok(out: String) -> Self {
        Self {
            code: 0,
            out,
            err: String::new(),
        }
    }

    /// An error. Plain mode prints the message only (what the PowerShell
    /// printed); fancy mode adds the `error:` label and the dim hint.
    fn fail(ui: &Ui, code: u8, message: &str, hint: Option<&str>) -> Self {
        let mut err = if ui.fancy {
            format!("{} {message}\n", ui.paint(Tone::Fail, "error:"))
        } else {
            format!("{message}\n")
        };
        if let (true, Some(hint)) = (ui.fancy, hint) {
            err.push_str(&format!(
                "  {}\n",
                ui.paint(Tone::Dim, &format!("hint: {hint}"))
            ));
        }
        Self {
            code,
            out: String::new(),
            err,
        }
    }
}

/// A usage error (exit 2): the message, then the usage line.
fn usage(ui: &Ui, message: &str) -> Outcome {
    if ui.fancy {
        Outcome::fail(ui, 2, message, Some(USAGE))
    } else {
        Outcome {
            code: 2,
            out: String::new(),
            err: format!("{message}\n{USAGE}\n"),
        }
    }
}

/// Runs `lab env` against the labd.env at `path`.
pub(crate) fn exec(args: &Args, ui: &Ui, path: &Path) -> Outcome {
    let Some(verb_text) = args.verb.as_deref() else {
        return list(ui, path);
    };
    let verb = verb_text.to_ascii_lowercase();
    if !matches!(verb.as_str(), "get" | "set" | "unset") {
        return usage(ui, &format!("unknown verb '{verb_text}'"));
    }
    let key = args.key.as_deref().unwrap_or("");
    if !env_file::is_valid_key(key) {
        return usage(ui, KEY_RULE);
    }
    match verb.as_str() {
        "get" => get_value(ui, path, key),
        "set" => set_value(args, ui, path, key),
        _ => unset_value(ui, path, key),
    }
}

/// The lines of labd.env; a missing file is no lines, any other read error
/// is an outcome to return.
fn load(ui: &Ui, path: &Path) -> Result<Vec<EnvLine>, Outcome> {
    match env_file::read(path) {
        Ok(lines) => Ok(lines),
        Err(e) if e.kind() == ErrorKind::NotFound => Ok(Vec::new()),
        Err(e) => Err(Outcome::fail(
            ui,
            1,
            &format!("cannot read {}: {e}", path.display()),
            None,
        )),
    }
}

/// One line for the listing. In plain mode this equals `env_file::mask_line`.
fn list_line(ui: &Ui, line: &EnvLine) -> String {
    match line {
        EnvLine::Pair { key, value } => format!(
            "{}={}",
            ui.paint(Tone::Cmd, key),
            ui.paint(Tone::Dim, &env_file::mask(key, value))
        ),
        EnvLine::Comment(_) | EnvLine::Unparsed(_) => {
            ui.paint(Tone::Dim, &env_file::mask_line(line))
        }
        EnvLine::Blank => String::new(),
    }
}

/// `lab env`: every line, masked.
fn list(ui: &Ui, path: &Path) -> Outcome {
    let lines = match env_file::read(path) {
        Ok(lines) => lines,
        Err(e) if e.kind() == ErrorKind::NotFound => {
            return Outcome::fail(
                ui,
                1,
                &format!("no labd.env at {}", path.display()),
                Some("lab env set KEY VALUE creates it"),
            );
        }
        Err(e) => {
            return Outcome::fail(
                ui,
                1,
                &format!("cannot read {}: {e}", path.display()),
                None,
            );
        }
    };
    let mut out = String::new();
    for line in &lines {
        out.push_str(&list_line(ui, line));
        out.push('\n');
    }
    Outcome::ok(out)
}

/// `lab env get KEY`.
fn get_value(ui: &Ui, path: &Path, key: &str) -> Outcome {
    let lines = match load(ui, path) {
        Ok(lines) => lines,
        Err(outcome) => return outcome,
    };
    match env_file::get(&lines, key) {
        Some(value) => Outcome::ok(format!(
            "{}\n",
            ui.paint(Tone::Dim, &env_file::mask(key, value))
        )),
        None => Outcome::fail(ui, 1, &format!("{key} is not set"), None),
    }
}

/// `lab env set KEY VALUE`.
fn set_value(args: &Args, ui: &Ui, path: &Path, key: &str) -> Outcome {
    let Some(value) = args.value.as_deref().or(args.value_flag.as_deref()) else {
        return usage(ui, "set needs KEY and VALUE");
    };
    if value.contains('\r') || value.contains('\n') {
        return Outcome::fail(ui, 2, "VALUE must be one line", None);
    }
    if value != value.trim() {
        return Outcome::fail(
            ui,
            2,
            "VALUE must not start or end with spaces (labd.env lines are trimmed when read)",
            None,
        );
    }
    let mut lines = match load(ui, path) {
        Ok(lines) => lines,
        Err(outcome) => return outcome,
    };
    env_file::set(&mut lines, key, value);
    save(ui, path, &lines, &format!("set {key}"))
}

/// `lab env unset KEY`.
fn unset_value(ui: &Ui, path: &Path, key: &str) -> Outcome {
    let mut lines = match load(ui, path) {
        Ok(lines) => lines,
        Err(outcome) => return outcome,
    };
    if !env_file::unset(&mut lines, key) {
        let text = format!("{key} is not set; labd.env unchanged");
        let line = if ui.fancy {
            ui.paint(Tone::Warn, &format!("{} {text}", ui.mark(Mark::Warn)))
        } else {
            text
        };
        return Outcome::ok(format!("{line}\n"));
    }
    save(ui, path, &lines, &format!("unset {key}"))
}

/// Writes `lines` with a backup and builds the success text: the backup path
/// (when a file existed) and the restart hint.
fn save(ui: &Ui, path: &Path, lines: &[EnvLine], done: &str) -> Outcome {
    let backup = match env_file::write_with_backup(path, lines) {
        Ok(backup) => backup,
        Err(e) => {
            return Outcome::fail(
                ui,
                1,
                &format!("cannot write {}: {e}", path.display()),
                None,
            );
        }
    };
    let pad = if ui.fancy { "  " } else { "" };
    let mut out = String::new();
    if ui.fancy {
        out.push_str(&ui.paint(Tone::Ok, &format!("{} {done}", ui.mark(Mark::Pass))));
        out.push('\n');
    }
    if let Some(backup) = backup {
        out.push_str(pad);
        out.push_str(&ui.paint(Tone::Dim, &format!("backup: {}", backup.display())));
        out.push('\n');
    }
    out.push_str(pad);
    if ui.fancy {
        out.push_str(&ui.paint(Tone::Dim, "hint: run: lab restart"));
    } else {
        out.push_str(RESTART_PLAIN);
    }
    out.push('\n');
    Outcome::ok(out)
}

/// Runs `lab env` and returns its exit code.
pub fn run(args: &Args, ctx: &Ctx) -> u8 {
    let outcome = exec(args, &ctx.ui, &paths::labd_env_path());
    anstream::print!("{}", outcome.out);
    anstream::eprint!("{}", outcome.err);
    outcome.code
}
```

```rust
//! `lab env` as the binary, on a temp lab home. Ports the command runs of
//! tools/lab/cli/test-logs-env.ps1 (the PowerShell check is named above each
//! test). The pure masking, key and line-rewrite checks of that file are
//! ported by LX-04 in tests/daemon_env.rs.

use std::path::PathBuf;
use std::process::Command;

const START: &str =
    "# head\r\nFOO=1\r\nCIMMERIA_LAB_DAEMON_TOKEN=test-token-not-real\r\n# tail\r\n";
const USAGE: &str =
    "usage: lab env [get KEY | set KEY VALUE | unset KEY]  (a VALUE starting with '-': -Value:<value>)";
const RESTART: &str = "restart the daemon for this to apply: lab restart";

struct Lab {
    home: tempfile::TempDir,
}

struct Run {
    code: i32,
    out: String,
    err: String,
}

impl Lab {
    fn new() -> Self {
        Self {
            home: tempfile::TempDir::new().expect("temp dir"),
        }
    }

    fn file(&self) -> PathBuf {
        self.home.path().join("labd.env")
    }

    fn seed(&self, text: &str) {
        std::fs::write(self.file(), text).expect("seed labd.env");
    }

    fn read(&self) -> String {
        std::fs::read_to_string(self.file()).expect("read labd.env")
    }

    /// File names in the lab home that start with `labd.env.bak-`, sorted.
    fn backups(&self) -> Vec<String> {
        let mut names: Vec<String> = std::fs::read_dir(self.home.path())
            .expect("read lab home")
            .map(|entry| entry.expect("dir entry").file_name().to_string_lossy().into_owned())
            .filter(|name| name.starts_with("labd.env.bak-"))
            .collect();
        names.sort();
        names
    }

    /// Runs `lab env <args>`.
    fn run(&self, args: &[&str]) -> Run {
        let output = Command::new(env!("CARGO_BIN_EXE_lab"))
            .arg("env")
            .args(args)
            .env("CIMMERIA_LAB_HOME", self.home.path())
            .output()
            .expect("the lab binary should start");
        Run {
            code: output.status.code().expect("an exit code"),
            out: String::from_utf8_lossy(&output.stdout).replace("\r\n", "\n"),
            err: String::from_utf8_lossy(&output.stderr).replace("\r\n", "\n"),
        }
    }
}
```
