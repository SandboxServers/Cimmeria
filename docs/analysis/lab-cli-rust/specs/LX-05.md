## Behaviour table (the error mapping; implement exactly this)

| Situation | Result |
|---|---|
| `send()` fails: connection refused, timeout, DNS, any transport error | `McpError::Unreachable(<reqwest message>)` |
| HTTP status not 2xx (401, 403, 404, 422, 500 ...) | `McpError::Http(<status>, <first 200 chars of body>)` |
| Reading the reply body fails (not a timeout) | `McpError::Protocol("reading the reply: <e>")` |
| Reading the reply body times out | `McpError::Unreachable(<e>)` |
| Body is empty or has no `data:` line (a `tools/call` needs a reply) | `McpError::Protocol("<tool> returned nothing")` |
| Body is not valid JSON (plain or the SSE data line) | `McpError::Protocol("reply is not JSON: <e>")` |
| JSON-RPC `error` object in the reply | `McpError::Tool(<error.message>)` (text `no message` when missing) |
| `result` missing | `McpError::Protocol("<tool> returned no result")` |
| `result.content[0].text` missing | `McpError::Protocol("<tool> returned no content")` |
| `result.isError == true` | `McpError::Tool(<content[0].text verbatim>)` |
| Text parses as JSON | that value |
| Text is not JSON | `serde_json::Value::String(<text>)` |

JSON-RPC ids: `initialize` is id 1, each `tools/call` takes 2, 3, 4, ... in order;
`notifications/initialized` has no id. Headers on every request: `Accept:
application/json, text/event-stream`, `Content-Type: application/json`,
`Authorization: Bearer <token>` only when a token is given, `Mcp-Session-Id` once
the server has issued one (any response carrying the header replaces it, as
`Send-McpRequest` does). Timeouts: 30 s for `initialize` and the notice (as the
PowerShell), the caller's `timeout` for `tools/call`, 10 s for the DELETE, 5 s for
`GET /status`. The PowerShell has no retry on a lost session; add none.

```rust
//! The client for the daemon's MCP endpoint.
//!
//! A port of `Read-McpBody`, `New-McpSession`, `Send-McpRequest`,
//! `Invoke-McpTool`, `Close-McpSession`, `Wait-UatLeaseFree` and
//! `Hide-LeaseIds` from `tools/lab/cli/uat-lib.ps1`. The protocol is the
//! streamable-HTTP one: `initialize`, the `Mcp-Session-Id` it returns,
//! `notifications/initialized`, then one POST per `tools/call`, whose reply
//! is plain JSON or an SSE stream (the last `data:` line is the answer).
//!
//! This module takes a base URL and an optional token, not an `Endpoint`, so it
//! stands alone; `connect` and `wait_lease_free` over an `Endpoint` are thin
//! wrappers added once `daemon/status.rs` has that type.

use std::fmt;
use std::sync::OnceLock;
use std::time::{Duration, Instant};

use regex::Regex;
use reqwest::blocking::Client;
use serde_json::{json, Value};

const PROTOCOL_VERSION: &str = "2025-06-18";
const ACCEPT: &str = "application/json, text/event-stream";
const SESSION_HEADER: &str = "Mcp-Session-Id";
const HANDSHAKE_TIMEOUT: Duration = Duration::from_secs(30);
const CLOSE_TIMEOUT: Duration = Duration::from_secs(10);
const STATUS_TIMEOUT: Duration = Duration::from_secs(5);

/// How long `wait_lease_free_at` sleeps between two looks at `/status`
/// (`Start-Sleep -Seconds 10` in `Wait-UatLeaseFree`).
pub const LEASE_POLL_INTERVAL: Duration = Duration::from_secs(10);

/// Why a call to the daemon's MCP endpoint did not give a result.
#[derive(Debug)]
pub enum McpError {
    /// Nothing answered: connection refused, a timeout, any transport error.
    Unreachable(String),
    /// The daemon answered with a non-2xx status (401 and 403 included): status, start of the body.
    Http(u16, String),
    /// The tool ran and said no (`isError`), or the reply was a JSON-RPC error.
    Tool(String),
    /// The reply was not the shape the protocol promises.
    Protocol(String),
}

impl fmt::Display for McpError {
    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
        match self {
            McpError::Unreachable(m) => write!(f, "cannot reach the lab daemon: {m}"),
            McpError::Http(code, body) => write!(f, "the lab daemon answered HTTP {code}: {body}"),
            McpError::Tool(m) => write!(f, "{m}"),
            McpError::Protocol(m) => write!(f, "unexpected reply from the lab daemon: {m}"),
        }
    }
}

impl std::error::Error for McpError {}

/// One open MCP session on the daemon.
pub struct McpClient {
    http: Client,
    url: String,
    token: Option<String>,
    session_id: Option<String>,
    next_id: u64,
}

impl McpClient {
    /// Opens a session on `base_url` (for example `http://127.0.0.1:8779`):
    /// POST `initialize` (protocol 2025-06-18, client `lab-cli`), keep the
    /// `Mcp-Session-Id` header, POST `notifications/initialized`.
    pub fn open(base_url: &str, token: Option<&str>) -> Result<Self, McpError> {
        // no_proxy: this is loopback, and a system proxy must never see the token.
        let http = Client::builder()
            .no_proxy()
            .build()
            .map_err(|e| McpError::Protocol(format!("http client: {e}")))?;
        let mut client = McpClient {
            http,
            url: format!("{}/mcp", base_url.trim_end_matches('/')),
            token: token.map(str::to_owned),
            session_id: None,
            next_id: 2,
        };
        let init = json!({
            "jsonrpc": "2.0",
            "id": 1,
            "method": "initialize",
            "params": {
                "protocolVersion": PROTOCOL_VERSION,
                "capabilities": {},
                "clientInfo": { "name": "lab-cli", "version": env!("CARGO_PKG_VERSION") },
            },
        });
        let text = client.post(&init, HANDSHAKE_TIMEOUT)?;
        if let Some(reply) = read_body(&text)? {
            if let Some(message) = rpc_error_message(&reply) {
                return Err(McpError::Tool(message));
            }
        }
        let notice = json!({ "jsonrpc": "2.0", "method": "notifications/initialized" });
        client.post(&notice, HANDSHAKE_TIMEOUT)?;
        Ok(client)
    }

    /// Calls one tool and returns its first text content parsed as JSON (a
    /// text that is not JSON comes back as a JSON string). `timeout` bounds
    /// this one request.
    pub fn call_tool(
        &mut self,
        name: &str,
        args: Value,
        timeout: Duration,
    ) -> Result<Value, McpError> {
        let id = self.next_id;
        self.next_id += 1;
        let body = json!({
            "jsonrpc": "2.0",
            "id": id,
            "method": "tools/call",
            "params": { "name": name, "arguments": args },
        });
        let text = self.post(&body, timeout)?;
        let reply = read_body(&text)?
            .ok_or_else(|| McpError::Protocol(format!("{name} returned nothing")))?;
        if let Some(message) = rpc_error_message(&reply) {
            return Err(McpError::Tool(message));
        }
        let result = reply
            .get("result")
            .ok_or_else(|| McpError::Protocol(format!("{name} returned no result")))?;
        let content_text = result
            .get("content")
            .and_then(Value::as_array)
            .and_then(|c| c.first())
            .and_then(|c| c.get("text"))
            .and_then(Value::as_str)
            .ok_or_else(|| McpError::Protocol(format!("{name} returned no content")))?;
        if result.get("isError").and_then(Value::as_bool) == Some(true) {
            return Err(McpError::Tool(content_text.to_owned()));
        }
        Ok(serde_json::from_str(content_text)
            .unwrap_or_else(|_| Value::String(content_text.to_owned())))
    }

    /// Ends the session with a DELETE. Best effort: every error is ignored
    /// (the daemon also drops idle sessions). A server that issued no session
    /// id gets no DELETE.
    pub fn close(self) {
        let Some(sid) = &self.session_id else { return };
        let mut req = self
            .http
            .delete(&self.url)
            .header("Accept", ACCEPT)
            .header(SESSION_HEADER, sid)
            .timeout(CLOSE_TIMEOUT);
        if let Some(t) = &self.token {
            req = req.bearer_auth(t);
        }
        let _ = req.send();
    }

    /// POSTs one JSON-RPC body; returns the reply text. Records a session id
    /// the reply carries. Maps transport and HTTP failures per the table.
    fn post(&mut self, body: &Value, timeout: Duration) -> Result<String, McpError> {
        let mut req = self
            .http
            .post(&self.url)
            .header("Accept", ACCEPT)
            .header("Content-Type", "application/json")
            .timeout(timeout)
            .body(body.to_string());
        if let Some(t) = &self.token {
            req = req.bearer_auth(t);
        }
        if let Some(sid) = &self.session_id {
            req = req.header(SESSION_HEADER, sid);
        }
        let resp = req
            .send()
            .map_err(|e| McpError::Unreachable(e.to_string()))?;
        let status = resp.status();
        if let Some(sid) = resp
            .headers()
            .get(SESSION_HEADER)
            .and_then(|v| v.to_str().ok())
        {
            self.session_id = Some(sid.to_owned());
        }
        let text = resp.text().map_err(|e| {
            if e.is_timeout() {
                McpError::Unreachable(e.to_string())
            } else {
                McpError::Protocol(format!("reading the reply: {e}"))
            }
        })?;
        if !status.is_success() {
            return Err(McpError::Http(status.as_u16(), text.chars().take(200).collect()));
        }
        Ok(text)
    }
}

/// The message of a JSON-RPC `error` object, `no message` when it has none.
fn rpc_error_message(reply: &Value) -> Option<String> {
    let err = reply.get("error").filter(|e| !e.is_null())?;
    Some(
        err.get("message")
            .and_then(Value::as_str)
            .unwrap_or("no message")
            .to_owned(),
    )
}

/// The JSON-RPC body of a reply: plain JSON, or the last SSE `data:` line.
/// `None` for an empty body or an SSE body with no `data:` line
/// (`Read-McpBody`; a notification's 202 has no body).
fn read_body(text: &str) -> Result<Option<Value>, McpError> {
    if text.trim().is_empty() {
        return Ok(None);
    }
    let json_text = if text.trim_start().starts_with('{') {
        text.trim().to_owned()
    } else {
        let Some(line) = text.split('\n').filter(|l| l.starts_with("data:")).last() else {
            return Ok(None);
        };
        line[5..].trim().to_owned()
    };
    serde_json::from_str(&json_text)
        .map(Some)
        .map_err(|e| McpError::Protocol(format!("reply is not JSON: {e}")))
}

/// `GET <base_url>/status` as raw JSON; `None` on any failure or when the
/// reply has no `daemon` object and `instances` array (as `Get-LabStatus`).
fn fetch_status_value(http: &Client, base_url: &str, token: Option<&str>) -> Option<Value> {
    let mut req = http
        .get(format!("{}/status", base_url.trim_end_matches('/')))
        .timeout(STATUS_TIMEOUT);
    if let Some(t) = token {
        req = req.bearer_auth(t);
    }
    let resp = req.send().ok()?;
    if !resp.status().is_success() {
        return None;
    }
    let value: Value = serde_json::from_str(&resp.text().ok()?).ok()?;
    let shaped = value.get("daemon").is_some_and(Value::is_object)
        && value.get("instances").is_some_and(Value::is_array);
    shaped.then_some(value)
}

/// Waits until `instance`'s lease is free, polling `poll` (a `/status`
/// reader) every `interval` until `max` has passed (`Wait-UatLeaseFree`).
/// True when it cleared in time. A poll that returns `None` (daemon down) is
/// "unknown", not "free": keep waiting. An instance missing from the reply
/// counts as free, as the PowerShell does.
pub fn wait_lease_free_with(
    poll: &mut dyn FnMut() -> Option<Value>,
    instance: &str,
    max: Duration,
    interval: Duration,
) -> bool {
    let deadline = Instant::now() + max;
    loop {
        if let Some(status) = poll() {
            let held = status
                .get("instances")
                .and_then(Value::as_array)
                .and_then(|rows| {
                    rows.iter()
                        .find(|r| r.get("instance").and_then(Value::as_str) == Some(instance))
                })
                .and_then(|row| row.get("lease"))
                .and_then(|lease| lease.get("held"))
                .and_then(Value::as_bool)
                .unwrap_or(false);
            if !held {
                return true;
            }
        }
        std::thread::sleep(interval);
        if Instant::now() >= deadline {
            return false;
        }
    }
}

/// `wait_lease_free_with` reading `GET /status` on `base_url` every `interval`.
pub fn wait_lease_free_at_every(
    base_url: &str,
    token: Option<&str>,
    instance: &str,
    max: Duration,
    interval: Duration,
) -> bool {
    let Ok(http) = Client::builder().no_proxy().build() else {
        return false;
    };
    wait_lease_free_with(
        &mut || fetch_status_value(&http, base_url, token),
        instance,
        max,
        interval,
    )
}

/// `wait_lease_free_at_every` at the PowerShell's 10 second interval.
pub fn wait_lease_free_at(
    base_url: &str,
    token: Option<&str>,
    instance: &str,
    max: Duration,
) -> bool {
    wait_lease_free_at_every(base_url, token, instance, max, LEASE_POLL_INTERVAL)
}

/// Lease ids (`lease-<hex>`) replaced in text that leaves the command
/// (`Hide-LeaseIds`). Matches ignoring case, as PowerShell's `-replace` does.
pub fn hide_lease_ids(text: &str) -> String {
    static LEASE_ID: OnceLock<Regex> = OnceLock::new();
    let re = LEASE_ID.get_or_init(|| Regex::new(r"(?i)lease-[0-9a-f]{8,}").expect("lease id regex"));
    re.replace_all(text, "lease-<redacted>").into_owned()
}
```

```rust
//! LX-05: the MCP client against a private stub of the daemon's endpoint.
//! Ports the `Hide-LeaseIds` check of `tools/lab/cli/test-uat.ps1`.

use std::io::{BufRead, BufReader, Read, Write};
use std::net::{SocketAddr, TcpListener, TcpStream};
use std::sync::atomic::{AtomicBool, Ordering};
use std::sync::{Arc, Mutex};
use std::time::Duration;

use cimmeria_lab_cli::daemon::mcp::{
    hide_lease_ids, wait_lease_free_at_every, wait_lease_free_with, McpClient, McpError,
};
use serde_json::{json, Value};

const TOKEN: &str = "tok-123";
const T: Duration = Duration::from_secs(5);

// ---- the stub: one request per connection, replies recorded, routes matched in order ----

#[derive(Clone)]
struct Route {
    method: &'static str,
    path: &'static str,
    /// The request body must contain this text.
    body_has: Option<&'static str>,
    status: u16,
    headers: Vec<(&'static str, &'static str)>,
    body: String,
    delay_ms: u64,
}

fn route(method: &'static str, path: &'static str, body_has: Option<&'static str>, status: u16, body: &str) -> Route {
    Route { method, path, body_has, status, headers: vec![], body: body.to_owned(), delay_ms: 0 }
}

#[derive(Clone, Debug)]
struct Recorded {
    method: String,
    path: String,
    headers: Vec<(String, String)>, // names lower-cased
    body: String,
}

impl Recorded {
    fn header(&self, name: &str) -> Option<&str> {
        self.headers.iter().find(|(k, _)| k == name).map(|(_, v)| v.as_str())
    }
}

struct Stub {
    addr: SocketAddr,
    seen: Arc<Mutex<Vec<Recorded>>>,
    stop: Arc<AtomicBool>,
}

impl Stub {
    fn start(routes: Vec<Route>) -> Stub {
        let listener = TcpListener::bind("127.0.0.1:0").expect("bind the stub");
        let addr = listener.local_addr().expect("stub address");
        let routes = Arc::new(routes);
        let seen = Arc::new(Mutex::new(Vec::new()));
        let stop = Arc::new(AtomicBool::new(false));
        let (r, s, st) = (routes.clone(), seen.clone(), stop.clone());
        std::thread::spawn(move || {
            for conn in listener.incoming() {
                if st.load(Ordering::SeqCst) {
                    break;
                }
                let Ok(stream) = conn else { continue };
                let (r, s) = (r.clone(), s.clone());
                std::thread::spawn(move || serve(stream, &r, &s));
            }
        });
        Stub { addr, seen, stop }
    }
    fn base_url(&self) -> String {
        format!("http://{}", self.addr)
    }
    fn requests(&self) -> Vec<Recorded> {
        self.seen.lock().unwrap().clone()
    }
    /// The recorded POST whose body contains `needle`.
    fn post_with(&self, needle: &str) -> Recorded {
        self.requests()
            .into_iter()
            .find(|r| r.method == "POST" && r.body.contains(needle))
            .unwrap_or_else(|| panic!("no POST containing {needle}"))
    }
}

impl Drop for Stub {
    fn drop(&mut self) {
        self.stop.store(true, Ordering::SeqCst);
        let _ = TcpStream::connect(self.addr); // wakes the accept loop
    }
}

fn serve(stream: TcpStream, routes: &[Route], seen: &Mutex<Vec<Recorded>>) {
    let _ = stream.set_read_timeout(Some(Duration::from_secs(5)));
    let mut reader = BufReader::new(stream.try_clone().expect("clone stream"));
    let mut first = String::new();
    if reader.read_line(&mut first).unwrap_or(0) == 0 {
        return;
    }
    let mut parts = first.split_whitespace();
    let method = parts.next().unwrap_or("").to_owned();
    let path = parts.next().unwrap_or("").to_owned();
    let mut headers = Vec::new();
    let mut length = 0usize;
    loop {
        let mut line = String::new();
        if reader.read_line(&mut line).unwrap_or(0) == 0 {
            break;
        }
        let line = line.trim_end().to_owned();
        if line.is_empty() {
            break;
        }
        if let Some((k, v)) = line.split_once(':') {
            let (k, v) = (k.trim().to_ascii_lowercase(), v.trim().to_owned());
            if k == "content-length" {
                length = v.parse().unwrap_or(0);
            }
            headers.push((k, v));
        }
    }
    let mut body = vec![0u8; length];
    let _ = reader.read_exact(&mut body);
    let body = String::from_utf8_lossy(&body).into_owned();
    seen.lock().unwrap().push(Recorded { method: method.clone(), path: path.clone(), headers, body: body.clone() });
    let hit = routes.iter().find(|r| {
        r.method == method && r.path == path && r.body_has.is_none_or(|n| body.contains(n))
    });
    let (status, extra, out, delay) = match hit {
        Some(r) => (r.status, r.headers.clone(), r.body.clone(), r.delay_ms),
        None => (500, vec![], "no route".to_owned(), 0),
    };
    std::thread::sleep(Duration::from_millis(delay));
    let mut head = format!("HTTP/1.1 {status} X\r\nContent-Length: {}\r\nConnection: close\r\n", out.len());
    for (k, v) in extra {
        head.push_str(&format!("{k}: {v}\r\n"));
    }
    let mut stream = stream;
    let _ = stream.write_all(format!("{head}\r\n{out}").as_bytes());
    let _ = stream.flush();
    let _ = stream.shutdown(std::net::Shutdown::Both);
}

const INIT_REPLY: &str = r#"{"jsonrpc":"2.0","id":1,"result":{"protocolVersion":"2025-06-18","capabilities":{},"serverInfo":{"name":"stub","version":"1"}}}"#;

fn rpc_text(text: &str) -> String {
    json!({"jsonrpc":"2.0","id":2,"result":{"content":[{"type":"text","text":text}]}}).to_string()
}

fn handshake() -> Vec<Route> {
    let mut init = route("POST", "/mcp", Some(r#""method":"initialize""#), 200, INIT_REPLY);
    init.headers = vec![("Mcp-Session-Id", "sess-1")];
    vec![init, route("POST", "/mcp", Some(r#""method":"notifications/initialized""#), 202, "")]
}

fn call_route(status: u16, body: &str) -> Route {
    route("POST", "/mcp", Some(r#""method":"tools/call""#), status, body)
}

/// A stub with the handshake plus one `tools/call` reply, and a client already open on it.
fn open_with(call: Route) -> (Stub, McpClient) {
    let mut routes = handshake();
    routes.push(call);
    routes.push(route("DELETE", "/mcp", None, 200, ""));
    let stub = Stub::start(routes);
    let client = McpClient::open(&stub.base_url(), Some(TOKEN)).expect("open");
    (stub, client)
}

#[test]
fn wait_returns_true_once_the_lease_clears() {
    let mut polls = 0;
    let mut poll = || {
        polls += 1;
        match polls {
            1 => None, // daemon not answering yet: keep waiting
            2 | 3 => status_with("i1", json!({"held": true})),
            _ => status_with("i1", json!({"held": false})),
        }
    };
    let ok = wait_lease_free_with(&mut poll, "i1", Duration::from_secs(5), Duration::from_millis(10));
    assert!(ok);
    assert_eq!(polls, 4, "polled until the lease cleared, no more");
}

#[test]
fn wait_returns_false_when_the_lease_stays_held() {
    let mut poll = || status_with("i1", json!({"held": true}));
    let t = std::time::Instant::now();
    assert!(!wait_lease_free_with(&mut poll, "i1", Duration::from_millis(100), Duration::from_millis(10)));
    assert!(t.elapsed() >= Duration::from_millis(100), "waited the full time");
}

#[test]
fn wait_returns_false_when_the_daemon_never_answers() {
    let mut poll = || None;
    assert!(!wait_lease_free_with(&mut poll, "i1", Duration::from_millis(60), Duration::from_millis(10)));
}

#[test]
fn wait_ignores_other_instances_and_treats_a_missing_one_as_free() {

    let mut no_lease = || status_with("i1", Value::Null);
    assert!(wait_lease_free_with(&mut no_lease, "i1", Duration::from_secs(1), Duration::from_millis(10)));
}
```
