```rust
//! Cutting text to a width.

/// `Limit-UatText`: `text` cut to `max` characters, ending in `...` when cut.
pub fn clip_text(text: &str, max: usize) -> String {
    if text.chars().count() <= max {
        return text.to_string();
    }
    let mut out: String = text.chars().take(max.saturating_sub(3)).collect();
    out.push_str("...");
    out
}

#[cfg(test)]

    use super::clip_text;

    #[test]
    fn short_text_is_unchanged() {
        assert_eq!(clip_text("abc", 3), "abc");

        assert_eq!(clip_text("abcdefgh", 6), "abc...");
```

```rust
//! The failures the lab already knows, tagged by signature. One table for
//! `lab watch` and `lab uat` (LX-15 to LX-17).

use std::sync::OnceLock;

use regex::Regex;

/// One known failure: the first row and reason that match both patterns get
/// its tag.
#[derive(Debug, Clone, Copy)]
pub struct KnownIssue {
    /// The issue tag, e.g. `#1341`.
    pub issue: &'static str,
    /// Regex for the failing row id.
    pub rows: &'static str,
    /// Regex for the failing reason.
    pub reason: &'static str,
    /// What the failure is.
    pub note: &'static str,
}

/// The table, in match order.
pub const KNOWN_ISSUES: &[KnownIssue] = &[
    KnownIssue {
        issue: "#1341",
        rows: "^FS-[PS]2$",
        reason: "movie-over",
        note: "client dropped the first-login entity batch (iterator residue)",
    },
    KnownIssue {
        issue: "#1343",
        rows: ".",
        reason: "Login_PasswordEdit holds 0 characters",
        note: "login password box read back empty on a slow-booting client",
    },
];

fn compiled() -> &'static [(Regex, Regex)] {
    static TABLE: OnceLock<Vec<(Regex, Regex)>> = OnceLock::new();
    TABLE.get_or_init(|| {
        KNOWN_ISSUES
            .iter()
            .map(|k| {
                (
                    Regex::new(&format!("(?i){}", k.rows)).expect("known issue row pattern"),
                    Regex::new(&format!("(?i){}", k.reason)).expect("known issue reason pattern"),
                )
            })
            .collect()
    })
}

/// The issue tag for a failing row and reason: the first entry whose row and
/// reason patterns both match, else `None`.
pub fn tag(row: &str, reason: &str) -> Option<&'static str> {
    KNOWN_ISSUES
        .iter()
        .zip(compiled())
        .find(|(_, (rows, why))| rows.is_match(row) && why.is_match(reason))
        .map(|(known, _)| known.issue)
}

#[cfg(test)]
mod tests {
    use super::tag;

    #[test]
    fn movie_over_on_fs_p2_and_fs_s2_is_1341() {
        assert_eq!(tag("FS-P2", "clause movie-over failed: client_entity_find/count gte 1"), Some("#1341"));
        assert_eq!(tag("FS-S2", "movie-over"), Some("#1341"));
    }

    #[test]
    fn the_row_pattern_is_anchored() {
        assert_eq!(tag("FS-P3", "movie-over"), None);
        assert_eq!(tag("XFS-P2", "movie-over"), None);
    }

    #[test]
    fn an_empty_password_box_on_any_row_is_1343() {
        assert_eq!(
            tag("FS-01", "setup failed: lab_login: Login_PasswordEdit holds 0 characters after typing, expected 4"),
            Some("#1343")
        );
    }

    #[test]
    fn matching_ignores_case_and_an_empty_row_matches_nothing() {
        assert_eq!(tag("fs-p2", "MOVIE-OVER"), Some("#1341"));
        assert_eq!(tag("", "Login_PasswordEdit holds 0 characters"), None);
        assert_eq!(tag("FS-01", "something else"), None);
    }
}
```

```rust
//! run), `batch.jsonl` (one line per run, appended as each run ends), then
//! `batch.json` and `summary.md` (written at the end). Shared by `lab watch`
//! and `lab uat` (LX-15 to LX-17).

pub mod progress;
pub mod read;
pub mod write;

pub use progress::format_progress_line;
pub use read::{find_latest_batch, read_plan, read_records};
pub use write::{append_record, write_plan};

use serde_json::{Map, Value};

/// File name of the plan.
pub const PLAN_JSON: &str = "plan.json";
/// File name of the run records.
pub const BATCH_JSONL: &str = "batch.jsonl";
/// File name of the finished summary; its presence means the batch is done.

/// `plan.json`. Keys, in this order: `schema`, `section`, `rows`, `lanes`,
/// `runs_per_lease`, `started` (UTC, round-trip format `...Z`), `pid` (the
/// process that runs the batch).
#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
#[serde(default)]
pub struct Plan {
    pub schema: u32,
    pub section: String,
    pub rows: Vec<String>,
    pub lanes: Vec<PlanLane>,
    pub runs_per_lease: u32,
    pub started: String,
    pub pid: Option<u32>,
}

/// One entry of `Plan::lanes`: keys `lane` (1-based) and `instance`.

#[serde(default)]
pub struct PlanLane {

    pub instance: String,

/// One line of `batch.jsonl`: a finished run, or (`Index` 0) a lane that died
/// outside a run. Keys in PascalCase, in the order below; `Chaos` and `Golden`
/// are written only when set. A missing key reads as its default, as
/// PowerShell's `Get-UatField` reads a missing property.
#[derive(Debug, Clone, Default, PartialEq, Serialize, Deserialize)]
#[serde(default, rename_all = "PascalCase")]
pub struct RunRecord {
    pub lane: u32,
    pub instance: String,
    /// 1-based run number in the lane; 0 for a lane death.
    pub index: u32,
    pub ok: bool,
    #[serde(deserialize_with = "lenient_rows")]
    pub rows: Vec<RowResult>,
    pub verdict: Option<Verdict>,
    pub run_dir: Option<String>,
    pub error: Option<String>,
    pub seconds: u64,
    pub braked: bool,
    pub started: Option<String>,
    pub ended: Option<String>,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub chaos: Option<String>,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub golden: Option<String>,
}

/// A run's verdict: keys `Ok`, `Row`, `Result`, `Reason`, `Issue` (the first
/// failing row and why; `Issue` the known-issue tag the writer found, if any).
#[derive(Debug, Clone, Default, PartialEq, Eq, Serialize, Deserialize)]
#[serde(default, rename_all = "PascalCase")]
pub struct Verdict {
    pub ok: bool,
    pub row: Option<String>,
    pub result: Option<String>,
    pub reason: Option<String>,
    pub issue: Option<String>,
}

/// One row of `lab_uat_run`'s reply, as stored in `Rows`. Keys the daemon adds
/// that are not named here are kept in `extra` and written back unchanged.
#[derive(Debug, Clone, Default, PartialEq, Serialize, Deserialize)]
#[serde(default)]
pub struct RowResult {
    pub row: String,
    pub result: String,
    #[serde(skip_serializing_if = "Option::is_none")]
    pub section: Option<String>,
    #[serde(deserialize_with = "null_as_default", skip_serializing_if = "Vec::is_empty")]
    pub reasons: Vec<String>,
    /// The heal hint (`{pose, step}`) of a failed row; LX-17 reads it.
    #[serde(skip_serializing_if = "Option::is_none")]
    pub heal: Option<Value>,
    #[serde(flatten)]
    pub extra: Map<String, Value>,
}

/// A JSON `null` reads as the type's default.
fn null_as_default<'de, D, T>(deserializer: D) -> Result<T, D::Error>
where
    D: Deserializer<'de>,
    T: Default + Deserialize<'de>,
{
    Ok(Option::<T>::deserialize(deserializer)?.unwrap_or_default())
}

/// `Rows`: `null` is no rows, and a `null` element (PowerShell wraps a missing
/// reply as `[null]`) is dropped.
fn lenient_rows<'de, D>(deserializer: D) -> Result<Vec<RowResult>, D::Error>
where
    D: Deserializer<'de>,
{
    let items: Option<Vec<Option<RowResult>>> = Option::deserialize(deserializer)?;
    Ok(items.unwrap_or_default().into_iter().flatten().collect())
}
```

```rust
//! Reading a batch folder. A lane may be appending while a watcher reads, so a
//! line that does not parse (a half-written last line) is skipped, never an
//! error.

use std::fs;
use std::io;
use std::path::{Path, PathBuf};

use super::{Plan, RunRecord};

fn read_text(path: &Path) -> io::Result<Option<String>> {
    match fs::read(path) {
        Ok(bytes) => Ok(Some(String::from_utf8_lossy(&bytes).into_owned())),
        Err(e) if e.kind() == io::ErrorKind::NotFound => Ok(None),
        Err(e) => Err(e),
    }
}

/// A batch's `plan.json`, or `None` when it is missing, unreadable or not a
/// plan (a batch from before plan.json existed has none).
pub fn read_plan(path: &Path) -> Option<Plan> {
    let text = read_text(path).ok()??;
    serde_json::from_str(text.trim_start_matches('\u{feff}')).ok()
}

/// The records of `batch.jsonl`. A missing file is no records; a line that is
/// blank or does not parse as a record is skipped; any other I/O error (a
/// locked or denied file) is returned.
pub fn read_records(path: &Path) -> io::Result<Vec<RunRecord>> {
    let Some(text) = read_text(path)? else {
        return Ok(Vec::new());
    };
    Ok(text
        .lines()
        .map(|line| line.trim().trim_start_matches('\u{feff}'))
        .filter(|line| !line.is_empty())
        .filter_map(|line| serde_json::from_str::<RunRecord>(line).ok())
        .collect())
}

/// The `batch-*` folder under `root` with the greatest name (names are
/// `batch-yyyyMMdd-HHmmss`, so name order is start order; file times are not),
/// or `None` when `root` is missing or holds none. Files named `batch-*` are
/// ignored. Any other I/O error is returned.
pub fn find_latest_batch(root: &Path) -> io::Result<Option<PathBuf>> {
    let entries = match fs::read_dir(root) {
        Ok(entries) => entries,
        Err(e) if matches!(e.kind(), io::ErrorKind::NotFound | io::ErrorKind::NotADirectory) => {
            return Ok(None)
        }
        Err(e) => return Err(e),
    };
    let mut best: Option<(String, PathBuf)> = None;
    for entry in entries {
        let entry = entry?;
        let name = entry.file_name().to_string_lossy().into_owned();
        if !name.to_ascii_lowercase().starts_with("batch-") || !entry.path().is_dir() {
            continue;
        }
        if best.as_ref().is_none_or(|(top, _)| name > *top) {
            best = Some((name, entry.path()));
        }
    }
    Ok(best.map(|(_, path)| path))
}
```

```rust
//! Writing a batch folder (used by `lab uat`, LX-15 to LX-17).

use std::fs;
use std::io::{self, Write};
use std::path::Path;
use std::sync::Mutex;

use super::{Plan, RunRecord};

static APPEND: Mutex<()> = Mutex::new(());

/// Writes `plan.json`.
pub fn write_plan(path: &Path, plan: &Plan) -> io::Result<()> {
    let text = serde_json::to_string_pretty(plan).map_err(io::Error::other)?;
    fs::write(path, text)
}

/// Appends one record to `batch.jsonl` as one line.
pub fn append_record(path: &Path, record: &RunRecord) -> io::Result<()> {
    let mut line = serde_json::to_string(record).map_err(io::Error::other)?;
    line.push('\n');
    let _guard = APPEND.lock().unwrap_or_else(|e| e.into_inner());
    let mut file = fs::OpenOptions::new().create(true).append(true).open(path)?;
    file.write_all(line.as_bytes())
}
```

```rust
//! The progress line `lab uat` prints for one finished run.

use super::RunRecord;

fn non_empty(text: &Option<String>) -> Option<&str> {
    text.as_deref().filter(|t| !t.is_empty())
}

/// `[<instance> <index>/<total>] <what> (<seconds>s)`, then `; golden: <g>`
/// when the record has a `Golden` value. `<what>` is `PASS`; `ERROR <error>`
/// for a run that threw; else `FAIL <row>: <reason>` plus ` [known <issue>]`
/// when the verdict has an issue.
pub fn format_progress_line(record: &RunRecord, total: u32) -> String {
    let what = if record.ok {
        "PASS".to_string()
    } else if let Some(error) = non_empty(&record.error) {
        format!("ERROR {error}")
    } else {
        let verdict = record.verdict.as_ref();
        let row = verdict.and_then(|v| non_empty(&v.row)).unwrap_or("");
        let reason = verdict.and_then(|v| non_empty(&v.reason)).unwrap_or("");
        let tag = verdict
            .and_then(|v| non_empty(&v.issue))
            .map(|issue| format!(" [known {issue}]"))
            .unwrap_or_default();
        format!("FAIL {row}: {reason}{tag}")
    };
    let mut line = format!(
        "[{} {}/{}] {} ({}s)",
        record.instance, record.index, total, what, record.seconds
    );
    if let Some(golden) = non_empty(&record.golden) {
        line.push_str(&format!("; golden: {golden}"));
    }
    line
}

#[cfg(test)]
mod tests {

        }
    }

    fn a_passing_run_reads_pass() {
        assert_eq!(format_progress_line(&record(true), 5), "[p2 3/5] PASS (41s)");
    }

    #[test]
    fn a_failing_run_names_the_row_the_reason_and_the_known_issue() {
        let mut r = record(false);
        r.verdict = Some(Verdict {
            ok: false,

        });
```

```rust
/// A JSON time as UTC. `Z` and offset forms parse; a time with no offset reads
/// as UTC. Anything else is `None`.
pub fn parse_time(value: &str) -> Option<DateTime<Utc>> {
    let text = value.trim();
    if text.is_empty() {
        return None;
    }
    if let Ok(t) = DateTime::parse_from_rfc3339(text) {
        return Some(t.with_timezone(&Utc));
    }
    for format in ["%Y-%m-%dT%H:%M:%S%.f", "%Y-%m-%d %H:%M:%S%.f"] {
        if let Ok(naive) = NaiveDateTime::parse_from_str(text, format) {
            return Some(Utc.from_utc_datetime(&naive));
        }
    }
    None
}

fn seconds_since(now: DateTime<Utc>, then: DateTime<Utc>) -> u64 {
    u64::try_from((now - then).num_seconds().max(0)).unwrap_or(0)
}

/// Reads a batch folder: `plan.json`, `batch.jsonl` and the state (D-LW5):
/// `Done` when `batch.json` exists, else `Unknown` without a plan, else `Live`
/// when the plan's pid passes `is_alive`, else `Interrupted`. A missing folder
/// reads as `Unknown` with no records; an unreadable `batch.jsonl` is an error.
pub fn read_batch(dir: &Path, is_alive: &dyn Fn(u32) -> bool) -> io::Result<BatchFiles> {
    let plan = read_plan(&dir.join(PLAN_JSON));
    let jsonl = dir.join(BATCH_JSONL);
    let records = read_records(&jsonl)
        .map_err(|e| io::Error::new(e.kind(), format!("{}: {e}", jsonl.display())))?;
    let state = if dir.join(BATCH_JSON).exists() {
        BatchState::Done
    } else {
        match &plan {
            None => BatchState::Unknown,
            Some(p) => match p.pid {
                Some(pid) if is_alive(pid) => BatchState::Live,
                _ => BatchState::Interrupted,
            },
        }
    };
    Ok(BatchFiles {
        name: dir
            .file_name()
            .map(|n| n.to_string_lossy().into_owned())
            .unwrap_or_default(),
        plan,
        records,
        state,
    })
}

/// A record's known issue from the one table: `known_issues::tag` first, the
/// record's own `Verdict.Issue` only as a fallback; `None` for an error record.
pub fn issue_of(record: &RunRecord) -> Option<String> {
    if record.error.as_deref().is_some_and(|e| !e.is_empty()) {
        return None;
    }
    let verdict = record.verdict.as_ref();
    let row = verdict.and_then(|v| v.row.as_deref()).unwrap_or("");
    let reason = verdict.and_then(|v| v.reason.as_deref()).unwrap_or("");
    if let Some(tag) = known_issues::tag(row, reason) {
        return Some(tag.to_string());
    }
    verdict
        .and_then(|v| v.issue.clone())
        .filter(|issue| !issue.is_empty())
}

/// A record as a failure line: `<row> <result>: <reason>` or `error: <error>`,
/// lease ids hidden, cut to 72 characters.
pub fn why_of(record: &RunRecord) -> String {
    let why = match record.error.as_deref().filter(|e| !e.is_empty()) {
        Some(error) => format!("error: {error}"),
        None => {
            let verdict = record.verdict.as_ref();
            let row = verdict.and_then(|v| v.row.as_deref()).unwrap_or("");
            let result = verdict.and_then(|v| v.result.as_deref()).unwrap_or("");
            let reason = verdict.and_then(|v| v.reason.as_deref()).unwrap_or("");
            format!("{row} {result}: {reason}")
        }
    };
    clip_text(&hide_lease_ids(&why), 72)
}

fn unknown_total_re() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| Regex::new(r"^\[(\S+) (\d+)/0\]").expect("progress prefix regex"))
}

/// A record's progress line as `lab uat` prints it, but tagged with the
/// re-classified issue so it agrees with the counts. Without a plan (`total`
/// 0) the runs per lease are unknown: `<i>/?`.
pub fn last_line(record: &RunRecord, total: u32) -> String {
    let mut copy = record.clone();
    let mut verdict = copy.verdict.take().unwrap_or_default();
    verdict.issue = issue_of(record);
    copy.verdict = Some(verdict);
    let mut line = format_progress_line(&copy, total);
    if total == 0 {
        line = unknown_total_re()
            .replace(&line, "[${1} ${2}/?]")
            .into_owned();
    }
    hide_lease_ids(&line)
}

fn lane_index(lanes: &mut Vec<LaneState>, name: &str, total: Option<u32>) -> usize {
    if let Some(i) = lanes.iter().position(|l| l.name == name) {
        return i;
    }
    lanes.push(LaneState {
        name: name.to_string(),
        count: 0,
        total,
        braked: false,
        last: None,
    });
    lanes.len() - 1
}

/// The batch's counts and lanes. Index 0 records are lane deaths: they brake
/// their lane and can be the last failure, but are not counted as runs.
/// Lanes are the plan's, then any instance only the records name.
pub fn panel(files: &BatchFiles) -> (BatchView, Vec<LaneState>) {
    let per_lane = files.plan.as_ref().map(|p| p.runs_per_lease);
    let mut lanes: Vec<LaneState> = Vec::new();
    if let Some(plan) = &files.plan {
        for lane in &plan.lanes {
            if !lane.instance.is_empty() {
                lane_index(&mut lanes, &lane.instance, per_lane);
            }
        }
    }
    let (mut runs, mut passed, mut failed, mut known) = (0u32, 0u32, 0u32, 0u32);
    let mut last_failure = None;
    for record in &files.records {
        let i = lane_index(&mut lanes, &record.instance, per_lane);
        if record.index > 0 {
            runs += 1;
            lanes[i].count += 1;
            if record.ok {
                passed += 1;
            } else {
                failed += 1;
                if issue_of(record).is_some() {
                    known += 1;
                }
            }
        }
        if record.braked {
            lanes[i].braked = true;
        }
        lanes[i].last = Some(record.clone());
        if !record.ok {
            last_failure = Some(LastFailure {
                run: format!("{}#{}", record.instance, record.index),
                why: why_of(record),
                issue: issue_of(record),
            });
        }
    }
    let section = files
        .plan
        .as_ref()
        .map(|p| p.section.clone())
        .filter(|s| !s.is_empty())
        .or_else(|| {
            files
                .records
                .first()
                .and_then(|r| r.rows.first())
                .and_then(|row| row.section.clone())
                .filter(|s| !s.is_empty())
        });
    let total = files
        .plan
        .as_ref()
        .map(|p| p.lanes.len() as u32 * p.runs_per_lease);
    let view = BatchView {
        name: files.name.clone(),
        state: files.state,
        section,
        runs,
        total,
        passed,
        failed,
        known,
        last_failure,
    };
    (view, lanes)
}

fn newest_png(dir: &Path) -> Option<SystemTime> {
    let mut newest: Option<SystemTime> = None;
    let mut pending = vec![dir.to_path_buf()];
    while let Some(current) = pending.pop() {
        let Ok(entries) = fs::read_dir(&current) else {
            continue;
        };
        for entry in entries.flatten() {
            let Ok(kind) = entry.file_type() else {
                continue;
            };
            let path = entry.path();
            if kind.is_dir() {
                pending.push(path);
            } else if kind.is_file()
                && path
                    .extension()
                    .is_some_and(|x| x.eq_ignore_ascii_case("png"))
            {
                if let Ok(modified) = entry.metadata().and_then(|m| m.modified()) {
                    if newest.is_none_or(|n| modified > n) {
                        newest = Some(modified);
                    }
                }
            }
        }
    }
    newest
}

/// Whole seconds since the newest `*.png` under `<run_dir>\rows`, or `None`
/// (an empty name, no `rows` folder, no screenshot). Errors under `rows` read
/// as no screenshot.
pub fn shot_age(run_dir: &str, now: DateTime<Utc>) -> Option<u64> {
    if run_dir.is_empty() {
        return None;
    }
    let rows = Path::new(run_dir).join("rows");
    if !rows.is_dir() {
        return None;
    }
    let modified = newest_png(&rows)?;
    Some(
        SystemTime::from(now)
            .duration_since(modified)
            .map(|d| d.as_secs())
            .unwrap_or(0),
    )
}

/// `2026-10-11T01:39:06.0000000Z`: the form of the `labd.log` stamps that
/// `log_counts` compares against (7 fractional digits, ordinal compare).
pub fn since_stamp(since: DateTime<Utc>) -> String {
    format!(
        "{}.{:07}Z",
        since.format("%Y-%m-%dT%H:%M:%S"),
        since.timestamp_subsec_nanos() / 100
    )
}

fn stamp_re() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| Regex::new(r"^(\d{4}-\d\d-\d\dT\S+)").expect("stamp regex"))
}

fn instance_re() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| Regex::new(r#"instance="([^"]+)""#).expect("instance span regex"))
}

/// Relaunches per instance and login retries in `labd.log` lines stamped at or
/// after `since` (an ordinal string compare of the ISO prefix; `""` keeps every
/// stamped line). A line without a leading stamp is skipped. A `relaunched
/// after crash` line counts for its first `instance="..."` span, or `default`
/// when it has none (a one-instance daemon); a `credential typing missed` line
/// is a login retry.
pub fn log_counts(lines: &[String], since: &str) -> LogCounts {
    let mut counts = LogCounts::default();
    for line in lines {
        let Some(caps) = stamp_re().captures(line) else {
            continue;
        };
        if !since.is_empty() && &caps[1] < since {
            continue;
        }
        if line.contains("relaunched after crash") {
            let instance = instance_re()
                .captures(line)
                .map(|c| c[1].to_string())
                .unwrap_or_else(|| "default".to_string());
            if let Some(entry) = counts.relaunches.iter_mut().find(|e| e.0 == instance) {
                entry.1 += 1;
            } else {
                counts.relaunches.push((instance, 1));
            }
        } else if line.contains("credential typing missed") {
            counts.login_retries += 1;
        }
    }
    counts
}

fn str_field<'a>(value: Option<&'a Value>, key: &str) -> &'a str {
    value
        .and_then(|v| v.get(key))
        .and_then(Value::as_str)
        .unwrap_or("")
}

/// An instance's LEASE cell: `free`, or `<owner> (<purpose>)` with the purpose
/// redacted and then cut to 40 characters (`...` after the cut). The time left
/// is never shown. Without a purpose, the owner alone.
pub fn lease_cell(lease: Option<&Value>) -> String {
    let held = lease
        .and_then(|l| l.get("held"))
        .and_then(Value::as_bool)
        .unwrap_or(false);
    if !held {
        return "free".to_string();
    }
    let inner = lease.and_then(|l| l.get("lease"));
    let owner = hide_lease_ids(str_field(inner, "owner"));
    let mut purpose = hide_lease_ids(str_field(inner, "purpose"));
    if purpose.chars().count() > 40 {
        purpose = format!("{}...", purpose.chars().take(40).collect::<String>());
    }
    if purpose.is_empty() {
        owner
    } else {
        format!("{owner} ({purpose})")
    }
}

/// One instance row. `row` is the instance's /status entry (`None` when /status
/// has none: the daemon is down, or the plan names a lane it lacks).
fn instance_view(
    name: &str,
    row: Option<&Value>,
    files: Option<&BatchFiles>,
    lanes: &[LaneState],
    now: DateTime<Utc>,
) -> InstanceView {
    let live = files.is_some_and(|f| f.state == BatchState::Live);
    let mut run = "-".to_string();
    let mut last = None;
    let mut run_dir: Option<String> = None;
    if let Some(lane) = lanes.iter().find(|l| l.name == name) {
        run = match lane.total {
            None => lane.count.to_string(),
            Some(total) if lane.braked => format!("{}/{total} braked", lane.count),
            Some(total) if live && lane.count < total => format!("{}/{total}", lane.count + 1),
            Some(total) => format!("{}/{total}", lane.count),
        };
        if let Some(record) = &lane.last {
            last = Some(last_line(record, lane.total.unwrap_or(0)));
            run_dir = record.run_dir.clone().filter(|d| !d.is_empty());
        }
    }
    let mut row_id = "-".to_string();
    let mut row_s = None;
    let mut client_pid = None;
    let mut lease = None;
    if let Some(r) = row {
        client_pid = r
            .get("client_pid")
            .and_then(Value::as_u64)
            .and_then(|p| u32::try_from(p).ok());
        lease = Some(lease_cell(r.get("lease")));
        match r.get("uat") {
            None => row_id = "?".to_string(),
            Some(uat) if uat.is_object() => {
                if let Some(current) = uat
                    .get("current_row")
                    .and_then(Value::as_str)
                    .filter(|c| !c.is_empty())
                {
                    row_id = hide_lease_ids(current);
                    row_s = uat
                        .get("row_started_at")
                        .and_then(Value::as_str)
                        .and_then(parse_time)
                        .map(|started| seconds_since(now, started));
                }
                if let Some(dir) = uat
                    .get("run_dir")
                    .and_then(Value::as_str)
                    .filter(|d| !d.is_empty())
                {
                    run_dir = Some(dir.to_string());
                }
            }
            Some(_) => {}
        }
    }
    InstanceView {
        instance: name.to_string(),
        client_pid,
        lease,
        run,
        row: row_id,
        row_s,
        shot_s: run_dir.and_then(|d| shot_age(&d, now)),
        last,
    }
}

/// The whole watch model. An error is an unreadable `batch.jsonl`.
pub fn build(inputs: &Inputs) -> io::Result<Snapshot> {
    let files = match inputs.batch_dir {
        Some(dir) => Some(read_batch(dir, inputs.is_alive)?),
        None => None,
    };
    let (batch, lanes) = match &files {
        Some(f) => {
            let (view, lanes) = panel(f);
            (Some(view), lanes)
        }
        None => (None, Vec::new()),
    };

    let rows: Vec<&Value> = inputs
        .status
        .and_then(|s| s.get("instances"))
        .and_then(Value::as_array)
        .map(|a| a.iter().filter(|v| !v.is_null()).collect())
        .unwrap_or_default();
    let mut instances = Vec::new();
    let mut seen: HashSet<String> = HashSet::new();
    for row in &rows {
        let name = str_field(Some(row), "instance").to_string();
        instances.push(instance_view(&name, Some(row), files.as_ref(), &lanes, inputs.now));
        seen.insert(name);
    }
    for lane in &lanes {
        if !seen.contains(&lane.name) {
            instances.push(instance_view(&lane.name, None, files.as_ref(), &lanes, inputs.now));
        }
    }

    let old_daemon = rows.iter().any(|row| row.get("uat").is_none());

    let mut orphans: Vec<u32> = Vec::new();
    if inputs.status.is_some() {
        let clients: HashSet<u32> = rows
            .iter()
            .filter_map(|r| r.get("client_pid").and_then(Value::as_u64))
            .filter_map(|p| u32::try_from(p).ok())
            .collect();
        for pid in inputs.sgw_pids {
            if !clients.contains(pid) && !orphans.contains(pid) {
                orphans.push(*pid);
            }
        }
    }

    let logs = files
        .as_ref()
        .and_then(|f| f.plan.as_ref())
        .map(|plan| {
            let since = parse_time(&plan.started).map(since_stamp).unwrap_or_default();
            log_counts(inputs.log_lines, &since)
        });

    Ok(Snapshot {
        daemon_up: inputs.status.is_some(),
        old_daemon,
        batch,
        instances,
        orphans,
        logs,
    })
}
```

```rust
pub fn copy_dir(from: &Path, to: &Path) {
    fs::create_dir_all(to).expect("mkdir");
    for entry in fs::read_dir(from).expect("read_dir") {
        let entry = entry.expect("entry");
        let dest = to.join(entry.file_name());
        if entry.file_type().expect("type").is_dir() {
            copy_dir(&entry.path(), &dest);
        } else {
            fs::copy(entry.path(), dest).expect("copy");
        }
    }
}

/// `<tmp>/<name>`: NEW_BATCH's plan.json, and its batch.jsonl with the
/// half-written last line cut off and `extra` appended as one more record.
pub fn batch_with(tmp: &Path, name: &str, extra: &RunRecord) -> PathBuf {
    let dir = tmp.join(name);
    fs::create_dir_all(&dir).expect("mkdir");
    fs::copy(tmp.join(NEW_BATCH).join("plan.json"), dir.join("plan.json")).expect("plan");
    let text = fs::read_to_string(tmp.join(NEW_BATCH).join("batch.jsonl")).expect("jsonl");

    let mut out = text[..cut].to_string();
    out.push_str(&serde_json::to_string(extra).expect("record"));
    out.push('\n');
    fs::write(dir.join("batch.jsonl"), out).expect("write");
    dir
}
```

```rust
//! Process checks for `lab watch`: which pids run `SGW.exe`, and whether the
//! process that wrote a batch's plan.json still runs.

/// Every running process as `(pid, image name)`. The name is the executable's
/// file name (`pwsh.exe`, `SGW.exe`); an unreadable list is empty.
#[cfg(windows)]
pub fn processes() -> Vec<(u32, String)> {
    use windows_sys::Win32::Foundation::{CloseHandle, INVALID_HANDLE_VALUE};
    use windows_sys::Win32::System::Diagnostics::ToolHelp::{
        CreateToolhelp32Snapshot, Process32FirstW, Process32NextW, PROCESSENTRY32W,
        TH32CS_SNAPPROCESS,
    };

    let mut found = Vec::new();
    // SAFETY: the snapshot handle is checked and closed; `entry` is a zeroed
    // PROCESSENTRY32W with `dwSize` set as the API requires, and outlives each call.
    unsafe {
        let snapshot = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
        if snapshot == INVALID_HANDLE_VALUE {
            return found;
        }
        let mut entry: PROCESSENTRY32W = std::mem::zeroed();
        entry.dwSize = std::mem::size_of::<PROCESSENTRY32W>() as u32;
        let mut more = Process32FirstW(snapshot, &mut entry);
        while more != 0 {
            let len = entry
                .szExeFile
                .iter()
                .position(|&c| c == 0)
                .unwrap_or(entry.szExeFile.len());
            found.push((
                entry.th32ProcessID,
                String::from_utf16_lossy(&entry.szExeFile[..len]),
            ));
            more = Process32NextW(snapshot, &mut entry);
        }
        CloseHandle(snapshot);
    }
    found
}

/// Every running process as `(pid, name)`, from `ps` (the name is the last path
/// segment of the command). No `ps` means an empty list.
#[cfg(not(windows))]
pub fn processes() -> Vec<(u32, String)> {
    let Ok(output) = std::process::Command::new("ps")
        .args(["-A", "-o", "pid=,comm="])
        .output()
    else {
        return Vec::new();
    };
    String::from_utf8_lossy(&output.stdout)
        .lines()
        .filter_map(|line| {
            let (pid, name) = line.trim_start().split_once(char::is_whitespace)?;
            let name = name.trim().rsplit(['/', '\\']).next().unwrap_or("").to_string();
            Some((pid.parse().ok()?, name))
        })
        .collect()
}

/// Whether a process image name is a lab batch's owner: `pwsh...` (a
/// PowerShell `lab uat`) or `lab` (`lab.exe`), ignoring case and `.exe`.
pub fn name_is_batch_owner(name: &str) -> bool {
    let lower = name.to_ascii_lowercase();
    let stem = lower.strip_suffix(".exe").unwrap_or(&lower);
    stem.starts_with("pwsh") || stem == "lab"
}
```

```rust
//! The files `lab watch` reads: a saved `/status` and the tail of `labd.log`.

use std::fs;
use std::io::{self, Read, Seek, SeekFrom};
use std::path::Path;

use serde_json::Value;

use crate::daemon::status::has_status_shape;

/// How many log lines a snapshot looks at.
pub const LOG_TAIL_LINES: usize = 4000;

/// How far back from the end of the log the tail reads (2 MiB holds far more
/// than 4000 lines).
const TAIL_WINDOW: u64 = 2 * 1024 * 1024;

/// A saved `/status`, held to the shape the daemon sends: anything else (a
/// missing or unreadable file, `null`, not JSON, another object) reads as the
/// daemon down, as it would live.
pub fn read_status_file(path: &Path) -> Option<Value> {
    let text = fs::read_to_string(path).ok()?;
    let body: Value = serde_json::from_str(text.trim_start_matches('\u{feff}')).ok()?;
    has_status_shape(&body).then_some(body)
}

/// The last `max_lines` lines of a file (it reads the last 2 MiB at most). An
/// error is an unreadable file (locked, denied).
pub fn read_tail_lines(path: &Path, max_lines: usize) -> io::Result<Vec<String>> {
    let mut file = fs::File::open(path)?;
    let len = file.metadata()?.len();
    let start = len.saturating_sub(TAIL_WINDOW);
    file.seek(SeekFrom::Start(start))?;
    let mut bytes = Vec::new();
    file.read_to_end(&mut bytes)?;
    let text = String::from_utf8_lossy(&bytes);
    let mut lines: Vec<&str> = text.lines().collect();
    if start > 0 && !lines.is_empty() {
        lines.remove(0); // the first line of a window cut mid-file is partial
    }
    let skip = lines.len().saturating_sub(max_lines);
    Ok(lines[skip..].iter().map(|l| (*l).to_string()).collect())
}
```

## Step B4: `commands/watch/text.rs` (new)

Port of `Format-LabWatchText`: plain text, one string per line, each at most `width`
characters, lease ids hidden. The columns are INSTANCE, CLIENT, LEASE, RUN, ROW, SHOT;
LAST (a whole progress line) goes on its own line, indented four spaces, under its
row. Column widths are the longest cell plus 2, for the first five columns; SHOT is
printed unpadded.

```rust
//! The plain-text frame of `lab watch`: the `--once` output.

use crate::clip::clip_text;
use crate::daemon::mcp::hide_lease_ids;

use super::snapshot::Snapshot;

const COLUMNS: [&str; 6] = ["INSTANCE", "CLIENT", "LEASE", "RUN", "ROW", "SHOT"];

/// The frame as lines, each at most `width` characters.
pub fn render_text(snapshot: &Snapshot, width: usize) -> Vec<String> {
    let mut lines: Vec<String> = Vec::new();

    match &snapshot.batch {
        None => lines.push(
            if snapshot.daemon_up { "no batch" } else { "no batch  daemon down" }.to_string(),
        ),
        Some(batch) => {
            let mut failed = format!("failed {}", batch.failed);
            if batch.known > 0 {
                failed.push_str(&format!(" ({} known)", batch.known));
            }
            let mut parts = vec![batch.name.clone()];
            if let Some(section) = &batch.section {
                parts.push(section.clone());
            }
            parts.push(batch.state.as_str().to_string());
            parts.push(format!("runs {}", batch.runs_text()));
            parts.push(format!("passed {}", batch.passed));
            parts.push(failed);
            lines.push(parts.join("  "));
            if let Some(last) = &batch.last_failure {
                let mut line = format!("last failure  {}  {}", last.run, last.why);
                if let Some(issue) = &last.issue {
                    line.push_str(&format!("  [{issue}]"));
                }
                lines.push(line);
            }
        }
    }
    lines.push(String::new());

    // The first six cells of a row are the columns; the seventh is LAST.
    let mut rows: Vec<[String; 7]> = vec![[
        COLUMNS[0].to_string(),
        COLUMNS[1].to_string(),
        COLUMNS[2].to_string(),
        COLUMNS[3].to_string(),
        COLUMNS[4].to_string(),
        COLUMNS[5].to_string(),
        "LAST".to_string(),
    ]];
    for i in &snapshot.instances {
        rows.push([
            i.instance.clone(),
            i.client_pid.map_or_else(|| "-".to_string(), |p| p.to_string()),
            i.lease.clone().filter(|l| !l.is_empty()).unwrap_or_else(|| "-".to_string()),
            i.run.clone(),
            i.row_cell(),
            i.shot_s.map_or_else(|| "-".to_string(), |s| format!("{s}s")),
            i.last.as_ref().map_or_else(String::new, |l| format!("    {l}")),
        ]);
    }
    let widths: Vec<usize> = (0..5)
        .map(|c| rows.iter().map(|r| r[c].chars().count()).max().unwrap_or(0) + 2)
        .collect();
    for (n, row) in rows.iter().enumerate() {
        let mut text = String::new();
        for c in 0..5 {
```

```rust
//! The one-line JSON of `lab watch --json`, and the usage-error object.

use crate::daemon::mcp::hide_lease_ids;

use super::snapshot::Snapshot;

/// A JSON object written key by key, in the order the keys are added.
struct Object {
    out: String,
    first: bool,
}

impl Object {
    fn new() -> Self {
        Object { out: String::from("{"), first: true }
    }

    fn raw(&mut self, key: &str, value: &str) {
        if !self.first {
            self.out.push(',');
        }
        self.first = false;
        self.out.push_str(&quote(key));
        self.out.push(':');
        self.out.push_str(value);
    }

    fn string(&mut self, key: &str, value: &str) {
        self.raw(key, &quote(value));
    }

    fn number(&mut self, key: &str, value: u64) {
        self.raw(key, &value.to_string());
    }

    fn is_empty(&self) -> bool {
        self.first
    }

    fn finish(mut self) -> String {
        self.out.push('}');
        self.out
    }
}

fn quote(text: &str) -> String {
    serde_json::to_string(text).expect("a string always serializes")
}

/// `{"ok":false,"exit":<code>,"error":"<message>"}`: a usage or read failure
/// under `--json` (lab uat's shape).
pub fn error_line(code: u8, message: &str) -> String {
    let mut o = Object::new();
    o.raw("ok", "false");
    o.number("exit", u64::from(code));
    o.string("error", message);
    o.finish()
}

/// The snapshot as one JSON line. Keys in order: `ok`, `daemon`, then with a
/// batch `batch`, `state`, `section`, `runs`, `passed`, `failed`, `known`,
/// `last_failure` (`run`, `why`, `issue`), `lanes` (instance to
/// `<run> [<row>] [shot <n>s]`, only instances in the batch), without one
/// `state: "none"`; then `orphans` (at most 3), `relaunches` (the total),
/// `login_retries`.
pub fn render_json(snapshot: &Snapshot) -> String {
    let mut o = Object::new();
    o.raw("ok", "true");
    o.string("daemon", if snapshot.daemon_up { "up" } else { "down" });
    match &snapshot.batch {
        None => o.string("state", "none"),
        Some(batch) => {
            o.string("batch", &batch.name);
            o.string("state", batch.state.as_str());
            if let Some(section) = &batch.section {
                o.string("section", section);
            }
            o.string("runs", &batch.runs_text());
            o.number("passed", u64::from(batch.passed));
            o.number("failed", u64::from(batch.failed));
            if batch.known > 0 {
                o.number("known", u64::from(batch.known));
            }
            if let Some(last) = &batch.last_failure {
                let mut f = Object::new();
                f.string("run", &last.run);
                f.string("why", &last.why);
                if let Some(issue) = &last.issue {
                    f.string("issue", issue);
                }
                o.raw("last_failure", &f.finish());
            }
            let mut lanes = Object::new();
            for i in &snapshot.instances {
                if i.run == "-" {
                    continue;
                }
                let mut parts = vec![i.run.clone()];
                let cell = i.row_cell();
                if cell != "-" {
                    parts.push(cell);
                }
                if let Some(s) = i.shot_s {
                    parts.push(format!("shot {s}s"));
                }
                lanes.string(&i.instance, &parts.join(" "));
            }
            if !lanes.is_empty() {
                o.raw("lanes", &lanes.finish());
            }
        }
    }
    if !snapshot.orphans.is_empty() {
        let shown: Vec<String> = snapshot.orphans.iter().take(3).map(|p| p.to_string()).collect();
        o.raw("orphans", &format!("[{}]", shown.join(",")));
    }
    if let Some(logs) = &snapshot.logs {
        let total: u32 = logs.relaunches.iter().map(|(_, c)| *c).sum();
        if total > 0 {
            o.number("relaunches", u64::from(total));
        }
        if logs.login_retries > 0 {
            o.number("login_retries", u64::from(logs.login_retries));
        }
    }
    hide_lease_ids(&o.finish())
}
```

```rust
/// Runs `lab watch`: checks the arguments, takes one snapshot and prints it as
/// text or one JSON line. Exit 0 a snapshot was printed; 1 the daemon is down
/// and there is no batch; 2 usage; 3 the lab state could not be read. Under
/// `--json`, 2 and 3 print `{"ok":false,"exit":N,"error":...}` on stdout.
pub fn run(args: &Args, ctx: &Ctx) -> u8 {
    let as_json = args.json;
    // Usage checks come first, so they exit 2 whatever the daemon's state.
    if !(1..=60).contains(&args.interval_seconds) {
        return fail(
            as_json,
            2,
            &format!("--interval-seconds must be 1 to 60 (got {})", args.interval_seconds),
        );
    }
    let batch_dir = match args.batch.as_deref() {
        None => None,
        Some(given) => {
            let path = Path::new(given);
            if !path.is_dir() {
                return fail(as_json, 2, &format!("no batch folder: {given}"));
            }
            if !path.join(PLAN_JSON).exists() && !path.join(BATCH_JSONL).exists() {
                return fail(
                    as_json,
                    2,
                    &format!("not a lab uat batch (no plan.json or batch.jsonl): {given}"),
                );
            }
            Some(std::path::absolute(path).unwrap_or_else(|_| path.to_path_buf()))
        }
    };
    if let Some(file) = args.status_file.as_deref() {
        if !Path::new(file).is_file() {
            return fail(as_json, 2, &format!("no status file: {file}"));
        }
    }

    // LX-14: the live view goes here, when neither --once nor --json is given
    // and stdout is a terminal. Until then every call prints one snapshot.
    let snapshot = match take_snapshot(args, batch_dir) {
        Ok(snapshot) => snapshot,
        Err(why) => {
            let first = why.lines().next().unwrap_or("");
            return fail(
                as_json,
                3,
                &format!("could not read the lab state: {}", hide_lease_ids(first)),
            );
        }
    };
    let mut out = std::io::stdout().lock();
    if as_json {
        let _ = writeln!(out, "{}", json::render_json(&snapshot));
    } else {
        for line in text::render_text(&snapshot, frame_width(&ctx.ui)) {
            let _ = writeln!(out, "{line}");
        }
    }
    if !snapshot.daemon_up && snapshot.batch.is_none() {
        1
    } else {
        0
    }
}

/// A usage or read failure: plain text on stderr, or under `--json` one object
/// on stdout. Returns the exit code.
fn fail(as_json: bool, code: u8, message: &str) -> u8 {
    if as_json {
        let _ = writeln!(std::io::stdout(), "{}", json::error_line(code, message));
    } else {
        eprintln!("{message}");
    }
    code
}

/// The terminal's width less one (a full-width line wraps on some hosts), or
/// 120 when there is no terminal to measure (stdout redirected) or it is
/// narrower than 41.
fn frame_width(ui: &Ui) -> usize {
    match ui.width {
        Some(w) if w >= 41 => w - 1,
        _ => 120,
    }
}

/// Everything one snapshot reads: /status (the file, or the daemon's `GET
/// /status`), the batch folder (`--batch`, else the newest under the UAT
/// root), the labd.log tail, the SGW pids and the clock. The only daemon call
/// is `GET /status`. An error is a batch folder or labd.log that cannot be read.
fn take_snapshot(args: &Args, batch_dir: Option<PathBuf>) -> Result<Snapshot, String> {
    let status_body = match args.status_file.as_deref() {
        Some(file) => source::read_status_file(Path::new(file)),
        None => status::fetch_value(&Endpoint::discover()),
    };
    // The newest batch is looked up on every snapshot: a batch started after
    // the watch began replaces the one on screen.
    let dir = match batch_dir {
        Some(dir) => Some(dir),
        None => {
            let root = paths::uat_root();
            find_latest_batch(&root).map_err(|e| format!("{}: {e}", root.display()))?
        }
    };
    let log_path = args
        .log_file
        .as_deref()
        .map(PathBuf::from)
        .unwrap_or_else(paths::labd_log_path);
    let log_lines = if log_path.is_file() {
        source::read_tail_lines(&log_path, source::LOG_TAIL_LINES)
            .map_err(|e| format!("{}: {e}", log_path.display()))?
    } else {
        Vec::new()
    };
    let sgw = process::sgw_pids();
    build(&Inputs {
        status: status_body.as_ref(),
        batch_dir: dir.as_deref(),
        log_lines: &log_lines,
        sgw_pids: &sgw,
        now: Utc::now(),
        is_alive: &process::batch_owner_alive,
    })
    .map_err(|e| e.to_string())
}
```
