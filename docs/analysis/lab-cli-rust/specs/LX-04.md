```rust
        pid: None,
        bind: DEFAULT_BIND.to_string(),

        return none;
    };

        .and_then(|pid| u32::try_from(pid).ok());

        .and_then(Value::as_str)
```

```rust
//! The labd.env file: its entries, read and written in place.

use std::io;
use std::path::{Path, PathBuf};
use std::sync::OnceLock;

use regex::Regex;

/// One line of labd.env.
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum EnvLine {
    /// `KEY=VALUE`: the key is trimmed, the value is everything after the
    /// first `=` of the trimmed line (it may itself contain `=`).
    Pair { key: String, value: String },
    /// A line whose first non-blank character is `#`, kept as written.
    Comment(String),
    /// An empty or whitespace-only line.
    Blank,
    /// Anything else (no `=` after the first character), kept as written.
    Unparsed(String),
}

/// Parses one line by the rules of `Read-LabdEnvFile`: trim, drop a leading
/// BOM, then blank, `#` comment, `KEY=VALUE` (the `=` at index 1 or later),
/// else unparsed.
pub fn parse_line(raw: &str) -> EnvLine {
    let trimmed = raw.trim().trim_start_matches('\u{feff}');
    if trimmed.is_empty() {
        return EnvLine::Blank;
    }
    if trimmed.starts_with('#') {
        return EnvLine::Comment(raw.to_string());
    }
    match trimmed.find('=') {
        Some(index) if index >= 1 => EnvLine::Pair {
            key: trimmed[..index].trim().to_string(),
            value: trimmed[index + 1..].to_string(),
        },
        _ => EnvLine::Unparsed(raw.to_string()),
    }
}

/// Splits `text` on `\r\n`, `\n` or `\r` (no empty last element for a
/// trailing newline, like `ReadAllLines`) and parses each line.
pub fn parse(text: &str) -> Vec<EnvLine> {
    let text = text.trim_start_matches('\u{feff}');
    text.replace("\r\n", "\n")
        .replace('\r', "\n")
        .lines()
        .map(parse_line)
        .collect()
}

/// Reads labd.env. A missing file is `Err` with `ErrorKind::NotFound`.
pub fn read(path: &Path) -> io::Result<Vec<EnvLine>> {
    let bytes = std::fs::read(path)?;
    Ok(parse(&String::from_utf8_lossy(&bytes)))
}

/// Reads labd.env; a missing or unreadable file is no lines (what
/// `Read-LabdEnvFile` returns for a missing file).
pub fn read_or_empty(path: &Path) -> Vec<EnvLine> {
    read(path).unwrap_or_default()
}

/// One line as it is written to the file.
pub fn render_line(line: &EnvLine) -> String {
    match line {
        EnvLine::Pair { key, value } => format!("{key}={value}"),
        EnvLine::Comment(text) | EnvLine::Unparsed(text) => text.clone(),
        EnvLine::Blank => String::new(),
    }
}

/// The file text: every line followed by CRLF, no BOM.
pub fn render(lines: &[EnvLine]) -> String {
    let mut text = String::new();
    for line in lines {
        text.push_str(&render_line(line));
        text.push_str("\r\n");
    }
    text
}

/// The value of `key` (matched ignoring ASCII case). The last line that sets
/// it wins, as in `Read-LabdEnvFile`'s map.
pub fn get<'a>(lines: &'a [EnvLine], key: &str) -> Option<&'a str> {
    lines.iter().rev().find_map(|line| match line {
        EnvLine::Pair { key: k, value } if k.eq_ignore_ascii_case(key) => Some(value.as_str()),
        _ => None,
    })
}

/// Sets `key`: the last line that sets it (ignoring case) is replaced in
/// place (with the new key spelling), else a new line is appended.
pub fn set(lines: &mut Vec<EnvLine>, key: &str, value: &str) {
    let new = EnvLine::Pair {
        key: key.to_string(),
        value: value.to_string(),
    };
    let last = lines
        .iter()
        .rposition(|line| matches!(line, EnvLine::Pair { key: k, .. } if k.eq_ignore_ascii_case(key)));
    match last {
        Some(index) => lines[index] = new,
        None => lines.push(new),
    }
}

/// Removes every line that sets `key` (ignoring case); comments and other
/// keys stay. Returns whether anything was removed.
pub fn unset(lines: &mut Vec<EnvLine>, key: &str) -> bool {
    let before = lines.len();
    lines.retain(
        |line| !matches!(line, EnvLine::Pair { key: k, .. } if k.eq_ignore_ascii_case(key)),
    );
    lines.len() != before
}

/// Whether `key` is a valid env key: `^[A-Z][A-Z0-9_]*$` (case sensitive).
/// Port of `Test-LabdEnvKey`.
pub fn is_valid_key(key: &str) -> bool {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| Regex::new(r"^[A-Z][A-Z0-9_]*$").expect("valid regex"))
        .is_match(key)
}

/// Copies an existing `path` to `<name>.bak-<yyyyMMdd-HHmmss>` beside it
/// (a `-2`, `-3`, ... suffix when that name is taken), then replaces `path`
/// with `lines` (UTF-8, CRLF, no BOM) by writing `<path>.tmp` and renaming it
/// over `path`. Creates the parent folder. Returns the backup path, `None`
/// when `path` did not exist. Port of `Save-LabdEnv`.
pub fn write_with_backup(path: &Path, lines: &[EnvLine]) -> io::Result<Option<PathBuf>> {
    let stamp = chrono::Local::now().format("%Y%m%d-%H%M%S").to_string();
    write_with_backup_at(path, lines, &stamp)
}

/// [`write_with_backup`] with the timestamp given (so tests can force a
/// same-second collision).
pub fn write_with_backup_at(
    path: &Path,
    lines: &[EnvLine],
    stamp: &str,
) -> io::Result<Option<PathBuf>> {
    let dir = match path.parent() {
        Some(parent) if !parent.as_os_str().is_empty() => parent.to_path_buf(),
        _ => PathBuf::from("."),
    };
    std::fs::create_dir_all(&dir)?;
    let mut backup = None;
    if path.exists() {
        let name = path
            .file_name()
            .map(|n| n.to_string_lossy().into_owned())
            .unwrap_or_else(|| "labd.env".to_string());
        let mut candidate = dir.join(format!("{name}.bak-{stamp}"));
        let mut n = 2;
        while candidate.exists() {
            candidate = dir.join(format!("{name}.bak-{stamp}-{n}"));
            n += 1;
        }
        std::fs::copy(path, &candidate)?;
        backup = Some(candidate);
    }
    let mut tmp_name = path.as_os_str().to_owned();
    tmp_name.push(".tmp");
    let tmp = PathBuf::from(tmp_name);
    std::fs::write(&tmp, render(lines).as_bytes())?;
    std::fs::rename(&tmp, path)?;
    Ok(backup)
}

fn secret_assignment() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| {
        Regex::new(r"(?i)((?:TOKEN|SECRET|PASSWORD|KEY|AUTH|CREDENTIAL)\w*\s*=)\S*")
            .expect("valid regex")
    })
}

fn bearer() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| Regex::new(r"(?i)bearer\s+\S+").expect("valid regex"))
}

fn hex64() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| Regex::new(r"(?i)\b[0-9a-f]{64}\b").expect("valid regex"))
}

fn url_host() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| {
        Regex::new(r"(?i)([a-z][a-z0-9+.-]*://)(?:\S*@)?[^/\s:?#@]+").expect("valid regex")
    })
}

fn secret_key() -> &'static Regex {
    static RE: OnceLock<Regex> = OnceLock::new();
    RE.get_or_init(|| {
        Regex::new(r"(?i)TOKEN|SECRET|PASSWORD|KEY|AUTH|CREDENTIAL").expect("valid regex")
    })
}

/// Masks secret-looking text for display (port of `Hide-LabdEnvText`): a
/// secret-named assignment loses its value, a bearer token and a 64-hex token
/// become `<redacted>`, and a URL's userinfo and host become `<host>` (the
/// port and path stay).
pub fn hide_text(text: &str) -> String {
    let t = secret_assignment().replace_all(text, "${1}<redacted>");
    let t = bearer().replace_all(&t, "<redacted>");
    let t = hex64().replace_all(&t, "<redacted>");
    url_host().replace_all(&t, "${1}<host>").into_owned()
}

/// A value for display (port of `Get-MaskedEnvValue`): `<redacted>` when the
/// key name contains TOKEN, SECRET, PASSWORD, KEY, AUTH or CREDENTIAL
/// (ignoring case), else [`hide_text`] of the value.
pub fn mask(key: &str, value: &str) -> String {
    if secret_key().is_match(key) {
        "<redacted>".to_string()
    } else {
        hide_text(value)
    }
}

/// One line for display (port of `Get-MaskedEnvLine`): `KEY=<masked value>`;
/// a comment is masked as text; a blank line is empty; an unparsed line is
/// not shown (`<unparsed line>`).
pub fn mask_line(line: &EnvLine) -> String {
    match line {
        EnvLine::Pair { key, value } => format!("{key}={}", mask(key, value)),
        EnvLine::Comment(text) => hide_text(text),
        EnvLine::Blank => String::new(),
        EnvLine::Unparsed(_) => "<unparsed line>".to_string(),
    }
}
```

```rust
//! The daemon bearer token that gates its loopback endpoint.

/// The variable the daemon's bearer token lives in.
pub const TOKEN_VAR: &str = "CIMMERIA_LAB_DAEMON_TOKEN";

/// The bearer token: the user environment (`HKCU\Environment`, Windows
/// only), else the process environment. An empty value is `None`. Never
/// print the result. Port of `Get-LabToken`.
pub fn daemon_token() -> Option<String> {
    pick_token(user_env(TOKEN_VAR), std::env::var(TOKEN_VAR).ok())
}

/// The first non-empty of the user value and the process value.
pub fn pick_token(user: Option<String>, process: Option<String>) -> Option<String> {
    [user, process]
        .into_iter()
        .flatten()
        .find(|token| !token.is_empty())
}

/// `HKCU\Environment\<name>` (REG_SZ or REG_EXPAND_SZ, expanded), or `None`
/// when it is absent, empty or unreadable.
#[cfg(windows)]
fn user_env(name: &str) -> Option<String> {
    use std::ptr::null_mut;
    use windows_sys::Win32::Foundation::{ERROR_MORE_DATA, ERROR_SUCCESS};
    use windows_sys::Win32::System::Registry::{
        RegGetValueW, HKEY_CURRENT_USER, RRF_RT_REG_EXPAND_SZ, RRF_RT_REG_SZ,
    };

    let subkey: Vec<u16> = "Environment\0".encode_utf16().collect();
    let value: Vec<u16> = name.encode_utf16().chain(std::iter::once(0)).collect();
    let flags = RRF_RT_REG_SZ | RRF_RT_REG_EXPAND_SZ;

    // First call: ask for the size in bytes (includes the terminating NUL).
    let mut size: u32 = 0;
    // SAFETY: `subkey` and `value` are NUL-terminated UTF-16 that outlive the
    // call; a null data pointer with a size pointer only queries the size.
    let status = unsafe {
        RegGetValueW(
            HKEY_CURRENT_USER,
            subkey.as_ptr(),
            value.as_ptr(),
            flags,
            null_mut(),
            null_mut(),
            &mut size,
        )
    };
    if status != ERROR_SUCCESS || size == 0 {
        return None;
    }

    // Second call: read into a buffer; retry a few times if the value grew.
    for _ in 0..3 {
        let mut buffer: Vec<u16> = vec![0; (size as usize) / 2 + 2];
        let mut bytes = (buffer.len() * 2) as u32;
        // SAFETY: `buffer` holds `bytes` bytes and outlives the call.
        let status = unsafe {
            RegGetValueW(
                HKEY_CURRENT_USER,
                subkey.as_ptr(),
                value.as_ptr(),
                flags,
                null_mut(),
                buffer.as_mut_ptr().cast(),
                &mut bytes,
            )
        };
        if status == ERROR_MORE_DATA {
            size = bytes;
            continue;
        }
        if status != ERROR_SUCCESS {
            return None;
        }
        let units = ((bytes as usize) / 2).min(buffer.len());
        let text = String::from_utf16_lossy(&buffer[..units]);
        let text = text.trim_end_matches('\0').to_string();
        return if text.is_empty() { None } else { Some(text) };
    }
    None
}

/// No user environment off Windows: only the process environment counts.
#[cfg(not(windows))]
fn user_env(_name: &str) -> Option<String> {
    None
}
```

```rust
//! Shared test support: the environment lock and guard (Part A); the stub
//! HTTP server is appended in LX-04b.
#![allow(dead_code)]

use std::ffi::OsString;
use std::sync::{Mutex, MutexGuard};

/// Tests that change the process environment hold this lock (nextest gives
/// each test its own process, but `cargo test` runs them as threads of one).
pub static ENV_LOCK: Mutex<()> = Mutex::new(());

/// Takes [`ENV_LOCK`], ignoring poison from an earlier failed test.
pub fn env_lock() -> MutexGuard<'static, ()> {
    ENV_LOCK.lock().unwrap_or_else(|e| e.into_inner())
}

/// Sets or removes environment variables and restores them when dropped.
pub struct EnvGuard {
    saved: Vec<(String, Option<OsString>)>,
}

impl EnvGuard {
    /// `Some(value)` sets the variable, `None` removes it.
    pub fn set(vars: &[(&str, Option<&str>)]) -> Self {
        let mut saved = Vec::new();
        for (key, value) in vars {
            saved.push((key.to_string(), std::env::var_os(key)));
            match value {
                Some(value) => std::env::set_var(key, value),
                None => std::env::remove_var(key),
            }
        }
        EnvGuard { saved }
    }
}

impl Drop for EnvGuard {
    fn drop(&mut self) {
        for (key, old) in self.saved.iter().rev() {
            match old {
                Some(old) => std::env::set_var(key, old),
                None => std::env::remove_var(key),
            }
        }
    }
}
```

```rust
/// The token every stub [`Endpoint`] carries.
pub const STUB_TOKEN: &str = "test-token-not-real";

/// One request the stub received. Header names are lower-cased.
#[derive(Debug, Clone)]
pub struct RecordedRequest {
    pub method: String,
    /// The path without its query string.
    pub path: String,
    pub headers: Vec<(String, String)>,
    pub body: String,
}

impl RecordedRequest {
    /// A header value by name (any case).
    pub fn header(&self, name: &str) -> Option<&str> {
        let name = name.to_ascii_lowercase();
        self.headers
            .iter()
            .find(|(key, _)| *key == name)
            .map(|(_, value)| value.as_str())
    }
}

struct Route {
    method: String,
    path: String,
    status: u16,
    body: String,
}

/// A stub HTTP server on `127.0.0.1:0`: one request per connection, answered
/// from `routes`, every request recorded. Stops when dropped.
pub struct StubServer {
    addr: SocketAddr,
    requests: Arc<Mutex<Vec<RecordedRequest>>>,
    stop: Arc<AtomicBool>,
    handle: Option<JoinHandle<()>>,
}

impl StubServer {
    /// `routes` are `(method, path, status, body)`; a request that matches
    /// none gets `404 not found`. The returned [`Endpoint`] points at the
    /// stub and carries [`STUB_TOKEN`].
    pub fn start(routes: Vec<(&str, &str, u16, &str)>) -> (StubServer, Endpoint) {
        let routes: Vec<Route> = routes
            .into_iter()
            .map(|(method, path, status, body)| Route {
                method: method.to_string(),
                path: path.to_string(),
                status,
                body: body.to_string(),
            })
            .collect();
        let listener = TcpListener::bind("127.0.0.1:0").expect("bind the stub server");
        let addr = listener.local_addr().expect("stub address");
        let requests = Arc::new(Mutex::new(Vec::new()));
        let stop = Arc::new(AtomicBool::new(false));
        let handle = {
            let requests = Arc::clone(&requests);
            let stop = Arc::clone(&stop);
            std::thread::spawn(move || {
                for stream in listener.incoming() {
                    if stop.load(Ordering::SeqCst) {
                        break;
                    }
                    let Ok(stream) = stream else { continue };
                    handle_connection(stream, &routes, &requests);
                }
            })
        };
        let endpoint = Endpoint {
            base_url: format!("http://{addr}"),
            token: Some(STUB_TOKEN.to_string()),
        };
        (
            StubServer {
                addr,
                requests,
                stop,
                handle: Some(handle),
            },
            endpoint,
        )
    }

    /// Every request received so far, in order.
    pub fn requests(&self) -> Vec<RecordedRequest> {
        self.requests.lock().expect("requests lock").clone()
    }
}

impl Drop for StubServer {
    fn drop(&mut self) {
        self.stop.store(true, Ordering::SeqCst);
        // Wake the accept loop so it sees the stop flag.
        let _ = TcpStream::connect(self.addr);
        if let Some(handle) = self.handle.take() {
            let _ = handle.join();
        }
    }
}

fn reason(status: u16) -> &'static str {
    match status {
        200 => "OK",
        401 => "Unauthorized",
        404 => "Not Found",
        500 => "Internal Server Error",
        _ => "Status",
    }
}

fn handle_connection(mut stream: TcpStream, routes: &[Route], log: &Mutex<Vec<RecordedRequest>>) {
    let _ = stream.set_read_timeout(Some(Duration::from_secs(2)));
    let mut buf: Vec<u8> = Vec::new();
    let mut chunk = [0u8; 1024];
    let head_end = loop {
        if let Some(pos) = buf.windows(4).position(|w| w == b"\r\n\r\n") {
            break pos;
        }
        match stream.read(&mut chunk) {
            Ok(0) | Err(_) => return,
            Ok(n) => buf.extend_from_slice(&chunk[..n]),
        }
    };
    let head = String::from_utf8_lossy(&buf[..head_end]).into_owned();
    let mut head_lines = head.split("\r\n");
    let request_line = head_lines.next().unwrap_or_default();
    let mut parts = request_line.split_whitespace();
    let method = parts.next().unwrap_or_default().to_string();
    let target = parts.next().unwrap_or_default().to_string();
    let headers: Vec<(String, String)> = head_lines
        .filter_map(|line| line.split_once(':'))
        .map(|(key, value)| (key.trim().to_ascii_lowercase(), value.trim().to_string()))
        .collect();
    let length: usize = headers
        .iter()
        .find(|(key, _)| key == "content-length")
        .and_then(|(_, value)| value.parse().ok())
        .unwrap_or(0);
    let mut body = buf[head_end + 4..].to_vec();
    while body.len() < length {
        match stream.read(&mut chunk) {
            Ok(0) | Err(_) => break,
            Ok(n) => body.extend_from_slice(&chunk[..n]),
        }
    }
    let path = target.split('?').next().unwrap_or_default().to_string();
    log.lock().expect("requests lock").push(RecordedRequest {
        method: method.clone(),
        path: path.clone(),
        headers,
        body: String::from_utf8_lossy(&body).into_owned(),
    });
    let (status, text) = routes
        .iter()
        .find(|r| r.method.eq_ignore_ascii_case(&method) && r.path == path)
        .map(|r| (r.status, r.body.clone()))
        .unwrap_or((404, "not found".to_string()));
    let response = format!(
        "HTTP/1.1 {status} {}\r\nContent-Type: application/json\r\nContent-Length: {}\r\nConnection: close\r\n\r\n{text}",
        reason(status),
        text.len()
    );
    let _ = stream.write_all(response.as_bytes());
    let _ = stream.flush();
    let _ = stream.shutdown(Shutdown::Both);
}

/// An [`Endpoint`] on a loopback port that nothing listens on: a listener is
/// bound to get a free port, then dropped.
pub fn closed_port_endpoint() -> Endpoint {
    let listener = TcpListener::bind("127.0.0.1:0").expect("bind");
    let port = listener.local_addr().expect("addr").port();
    drop(listener);
    Endpoint {
        base_url: format!("http://127.0.0.1:{port}"),
        token: Some(STUB_TOKEN.to_string()),
    }
}
```
