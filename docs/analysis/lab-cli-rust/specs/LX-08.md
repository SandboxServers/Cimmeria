## Step 2: `doctor/checks.rs` (write the whole file)

Ports `Get-TaskCheck` .. `Get-VersionCheck`. Every check is a pure function
of `Inputs`, which `system.rs` fills. Every PASS, WARN and FAIL text is copied
from the PowerShell; a text that ended in `; run lab xyz` is split into
`detail` and `hint` (plain mode prints `detail; hint`, fancy mode prints the
hint on its own line).

One deliberate change: the binary check says `same bytes (N bytes)` where the
PowerShell said `same SHA-256 (...)`, because the crate adds no hashing
dependency (the packet says to compare bytes).

```rust
//! The ten `lab doctor` checks, ported from tools/lab/cli/doctor.ps1.
//! Every check is a pure function of [`Inputs`], which `system.rs` gathers.

use std::path::{Path, PathBuf};

use crate::daemon::status::StatusReply;

/// How a check came out.
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum Status {
    Pass,
    Warn,
    Fail,
}

impl Status {
    /// The word plain mode prints.
    pub fn label(self) -> &'static str {
        match self {
            Status::Pass => "PASS",
            Status::Warn => "WARN",
            Status::Fail => "FAIL",
        }
    }
}

/// One check's result.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Check {
    pub status: Status,
    pub name: String,
    pub detail: String,
    /// What to do about it, e.g. `run lab install`; plain mode appends it to
    /// the detail after `; `, fancy mode prints it as its own `hint:` line.
    pub hint: Option<String>,
}

impl Check {
    fn new(status: Status, name: &str, detail: impl Into<String>, hint: Option<&str>) -> Self {
        Check {
            status,
            name: name.to_string(),
            detail: detail.into(),
            hint: hint.map(str::to_string),
        }
    }

    /// The detail as the PowerShell printed it: `detail; hint`.
    pub fn plain_detail(&self) -> String {
        match &self.hint {
            Some(hint) => format!("{}; {hint}", self.detail),
            None => self.detail.clone(),
        }
    }
}

/// What the gathering found about one instance.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct InstanceFacts {
    pub label: String,
    /// The instance's account file exists under the install dir.
    pub account_file: bool,
    /// The instance's profile folder is seeded.
    pub seeded: bool,
}

/// How the installed exe and the daemon's copy compare.
#[derive(Debug, Clone, PartialEq, Eq)]
pub enum BinaryFacts {
    /// This file (the first of the two that is missing or unreadable).
    Missing(PathBuf),
    /// Both exist and their bytes differ.
    Differ,
    /// Both exist and are identical; the length in bytes.
    Same(u64),
}

/// `git diff --quiet <sha> origin/main -- tools/lab` came out as:
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum ContentCheck {
    /// Exit 0: no difference.
    Same,
    /// Exit 1: tools/lab differs.
    Differs,
    /// Any other exit (an unknown sha, no origin/main).
    Unknown,
}

/// Everything read from the system, so the checks stay pure.
#[derive(Debug, Clone)]
pub struct Inputs {
    pub task_installed: bool,
    /// `GET /status`, or `None` when the daemon is down or refuses.
    pub status: Option<StatusReply>,
    pub token_set: bool,
    /// `CIMMERIA_LAB_INSTALL_DIR` from labd.env, trimmed; `None` when blank.
    pub install_dir: Option<String>,
    /// `<install dir>\Binaries\SGW.exe` exists.
    pub sgw_exe_exists: bool,
    /// The profile root as text.
    pub profile_root: String,
    pub instances: Vec<InstanceFacts>,
    /// Pids of the running `SGW.exe` processes.
    pub running_sgw: Vec<u32>,
    /// `client_pid` of every instance `/status` lists.
    pub listed_sgw: Vec<u32>,
    pub binaries: BinaryFacts,
    /// The first line of `<lab home>\cli\cli\VERSION`, trimmed; `None` when absent or blank.
    pub version_sha: Option<String>,
    /// The Cimmeria checkout the command runs in, if any.
    pub checkout: Option<PathBuf>,
    /// The tools/lab content comparison; `None` when it was skipped (no
    /// checkout or no `version_sha`).
    pub content: Option<ContentCheck>,
}

/// `<install dir>\Binaries\SGW.exe`.
pub fn sgw_exe_path(install_dir: &str) -> PathBuf {
    Path::new(install_dir).join("Binaries").join("SGW.exe")
}

/// All ten checks, in the PowerShell's order.
pub fn run_all(i: &Inputs) -> Vec<Check> {
    vec![
        task_check(i),
        daemon_check(i),
        token_check(i),
        install_check(i),
        profile_root_check(i),
        account_check(i),
        seed_check(i),
        stray_sgw_check(i),
        binary_check(i),
        version_check(i),
    ]
}

/// `Get-TaskCheck`.
pub fn task_check(i: &Inputs) -> Check {
    let name = "scheduled task CimmeriaLabDaemon exists";
    if i.task_installed {
        Check::new(Status::Pass, name, "installed", None)
    } else {
        Check::new(Status::Fail, name, "not installed", Some("run lab install"))
    }
}

/// `Get-DaemonCheck`.
pub fn daemon_check(i: &Inputs) -> Check {
    let name = "daemon answers /status";
    match &i.status {
        None => Check::new(
            Status::Fail,
            name,
            "not running, or the status endpoint is down",
            Some("run lab start"),
        ),
        Some(status) => Check::new(
            Status::Pass,
            name,
            format!("pid {}, version {}", status.daemon.pid, status.daemon.version),
            None,
        ),
    }
}

/// `Get-TokenCheck`. The token itself is never reported.
pub fn token_check(i: &Inputs) -> Check {
    let name = "CIMMERIA_LAB_DAEMON_TOKEN is set";
    if i.token_set {
        Check::new(Status::Pass, name, "set", None)
    } else {
        Check::new(Status::Fail, name, "not set", Some("run lab install"))
    }
}

/// `Get-InstallCheck`.
pub fn install_check(i: &Inputs) -> Check {
    let name = "labd.env has CIMMERIA_LAB_INSTALL_DIR and SGW.exe is under it";
    let Some(dir) = i.install_dir.as_deref() else {
        return Check::new(
            Status::Fail,
            name,
            "CIMMERIA_LAB_INSTALL_DIR is not set in labd.env",
            None,
        );
    };
    let exe = sgw_exe_path(dir);
    if !i.sgw_exe_exists {
        return Check::new(Status::Fail, name, format!("no {}", exe.display()), None);
    }
    Check::new(Status::Pass, name, exe.display().to_string(), None)
}

/// `Get-ProfileRootCheck`: the root must be absolute and outside the install
/// dir. A path that only shares a name prefix with it (`C:\Games\SGWX` against
/// `C:\Games\SGW`) is outside.
pub fn profile_root_check(i: &Inputs) -> Check {
    let name = "profile root is absolute and not inside the install dir";
    let root = i.profile_root.as_str();
    if root.is_empty() || !is_fully_qualified(root) {
        return Check::new(
            Status::Fail,
            name,
            format!("not an absolute path: {root}"),
            None,
        );
    }
    let full = full_path(root);
    if let Some(dir) = i.install_dir.as_deref() {
        let install = full_path_lossy(dir);
        let under = format!("{install}\\");
        if full.eq_ignore_ascii_case(&install) || starts_with_ignore_case(&full, &under) {
            return Check::new(
                Status::Fail,
                name,
                format!("inside the install dir: {full}"),
                None,
            );
        }
    }
    Check::new(Status::Pass, name, full, None)
}

/// `Get-AccountCheck`.
pub fn account_check(i: &Inputs) -> Check {
    let name = "each instance's account file exists";
    if i.install_dir.is_none() {
        return Check::new(
            Status::Warn,
            name,
            "no install dir; account files not checked",
            None,
        );
    }
    let missing: Vec<&str> = i
        .instances
        .iter()
        .filter(|f| !f.account_file)
        .map(|f| f.label.as_str())
        .collect();
    if !missing.is_empty() {
        return Check::new(
            Status::Warn,
            name,
            format!("missing: {}", missing.join(", ")),
            Some("run lab instances init"),
        );
    }
    Check::new(
        Status::Pass,
        name,
        format!("{} present", i.instances.len()),
        None,
    )
}

/// `Get-SeedCheck`.
pub fn seed_check(i: &Inputs) -> Check {
    let name = "each instance's profile is seeded";
    let unseeded: Vec<&str> = i
        .instances
        .iter()
        .filter(|f| !f.seeded)
        .map(|f| f.label.as_str())
        .collect();
    if !unseeded.is_empty() {
        return Check::new(
            Status::Warn,
            name,
            format!("not seeded: {}", unseeded.join(", ")),
            None,
        );
    }
    Check::new(
        Status::Pass,
        name,
        format!("{} seeded", i.instances.len()),
        None,
    )
}

/// `Get-StraySgwCheck`: running `SGW.exe` pids that `/status` does not list.
pub fn stray_sgw_check(i: &Inputs) -> Check {
    let name = "no SGW.exe runs that /status does not list";
    let stray: Vec<String> = i
        .running_sgw
        .iter()
        .filter(|pid| !i.listed_sgw.contains(pid))
        .map(|pid| pid.to_string())
        .collect();
    if !stray.is_empty() && i.status.is_none() {
        return Check::new(
            Status::Warn,
            name,
            format!(
                "daemon down; running SGW pid {} cannot be matched to an instance",
                stray.join(", ")
            ),
            None,
        );
    }
    if !stray.is_empty() {
        return Check::new(
            Status::Warn,
            name,
            format!("not listed by /status: pid {}", stray.join(", ")),
            None,
        );
    }
    Check::new(Status::Pass, name, "none", None)
}

/// `Get-BinaryCheck`: the installed exe and the daemon's copy are one file.
pub fn binary_check(i: &Inputs) -> Check {
    let name = "installed cimmeria-lab.exe is the daemon's copy";
    match &i.binaries {
        BinaryFacts::Missing(path) => Check::new(
            Status::Warn,
            name,
            format!("missing: {}", path.display()),
            Some("run lab install"),
        ),
        BinaryFacts::Differ => Check::new(
            Status::Warn,
            name,
            "bin and labd copies differ",
            Some("run lab restart"),
        ),
        BinaryFacts::Same(len) => {
            Check::new(Status::Pass, name, format!("same bytes ({len} bytes)"), None)
        }
    }
}

/// `Get-VersionCheck`: the CLI copy has origin/main's tools/lab (content, not
/// ancestry: PRs are squash-merged, so an installed worktree sha never
/// becomes an ancestor of origin/main).
pub fn version_check(i: &Inputs) -> Check {
    let name = "CLI copy matches origin/main (as of the last fetch)";
    let Some(root) = i.checkout.as_deref() else {
        return Check::new(
            Status::Pass,
            name,
            "not in a Cimmeria checkout; skipped",
            None,
        );
    };
    let Some(sha) = i.version_sha.as_deref() else {
        return Check::new(
            Status::Warn,
            name,
            "no cli\\cli\\VERSION",
            Some("run lab setup"),
        );
    };
    match i.content {
        Some(ContentCheck::Same) => Check::new(
            Status::Pass,
            name,
            format!("{sha} has origin/main's tools/lab"),
            None,
        ),
        Some(ContentCheck::Differs) => Check::new(
            Status::Warn,
            name,
            format!("{sha} differs from origin/main in tools/lab"),
            Some("run lab setup"),
        ),
        Some(ContentCheck::Unknown) | None => Check::new(
            Status::Warn,
            name,
            format!("{sha} is unknown in {}", root.display()),
            Some("git fetch, or run lab setup"),
        ),
    }
}

/// .NET's `Path.IsPathFullyQualified` for Windows paths: `X:\` or `X:/`, or a
/// leading pair of separators (UNC and `\\?\`). Text-only, so it behaves the
/// same on every platform.
fn is_fully_qualified(path: &str) -> bool {
    let b = path.as_bytes();
    let sep = |c: u8| c == b'\\' || c == b'/';
    if b.len() >= 2 && sep(b[0]) && sep(b[1]) {
        return true;
    }
    b.len() >= 3 && b[0].is_ascii_alphabetic() && b[1] == b':' && sep(b[2])
}

/// `Path.GetFullPath(...).TrimEnd('\')` for a fully qualified path: `/`
/// becomes `\`, `.` and empty segments go, `..` removes the segment before it.
fn full_path(path: &str) -> String {
    let s = path.replace('/', "\\");
    let (prefix, rest) = (s[..2].to_string(), &s[2..]);
    let mut parts: Vec<&str> = Vec::new();
    for segment in rest.split('\\') {
        match segment {
            "" | "." => {}
            ".." => {
                parts.pop();
            }
            other => parts.push(other),
        }
    }
    format!("{prefix}\\{}", parts.join("\\"))
        .trim_end_matches('\\')
        .to_string()
}

/// [`full_path`] for a fully qualified path; for anything else the text with
/// `/` turned into `\` and no trailing `\`.
fn full_path_lossy(path: &str) -> String {
    if is_fully_qualified(path) {
        full_path(path)
    } else {
        path.replace('/', "\\").trim_end_matches('\\').to_string()
    }
}

fn starts_with_ignore_case(text: &str, prefix: &str) -> bool {
    text.len() >= prefix.len()
        && text.as_bytes()[..prefix.len()].eq_ignore_ascii_case(prefix.as_bytes())
}
```

Check `full_path("C:\\")`: `s[..2]` is `C:`, rest is `\`, parts empty, the format gives `C:\`,
trimmed to `C:`. For `\\server\share\x`: prefix `\\`, parts `server, share, x`,
result `\\\server\share\x`: three leading backslashes. Avoid that: in
`full_path`, build with `if prefix == "\\\\" { format!("{prefix}{}", parts.join("\\")) } else { format!("{prefix}\\{}", ...) }`.
Write it that way (replace the `format!` line above with):

```rust
    let joined = parts.join("\\");
    let full = if prefix == "\\\\" {
        format!("{prefix}{joined}")
    } else {
        format!("{prefix}\\{joined}")
    };
    full.trim_end_matches('\\').to_string()
```

## Step 3: `doctor/system.rs` (write the whole file)

Gathers `Inputs`. Rules, from the packet and the PowerShell:

- Scheduled task: `schtasks /Query /TN CimmeriaLabDaemon`, exit code only
  (stdin, stdout, stderr null). A spawn failure (no `schtasks`, as on Linux)
  is "not installed".
- Daemon: `status::fetch(&Endpoint::discover())`.
- Token: `token::daemon_token().is_some()`.
- Instances: with a daemon, the `instance` of each `/status` row (zero rows
  gives none); without one, `CIMMERIA_LAB_INSTANCES` from labd.env split on
  `,`, each trimmed, blanks dropped, and `["default"]` when nothing is left.
- Account file: `<install dir>\Binaries\sessions\lab-account.json` for
  `default`, else `lab-account.<label>.json`.
- Seed: `<profile root>\<label>\profile\Documents\My Games\Firesky\SGWGame`
  exists.
- `SGW.exe` pids: Windows: `CreateToolhelp32Snapshot` over the process list,
  image name `SGW.exe` ignoring case; elsewhere none. `sgw_pids` is
  `pub(crate)` because LX-15 reuses it.
- Binaries: `<lab home>\bin\cimmeria-lab.exe` against
  `<lab home>\labd\cimmeria-lab.exe`: a file that is absent or will not open
  is `Missing` (the `bin` one first), different lengths is `Differ`, equal
  lengths are compared chunk by chunk (64 KiB), no hashing.
- Version: the first line of `<lab home>\cli\cli\VERSION`
  (`installed_cli_dir().join("cli").join("VERSION")`), trimmed, a BOM
  dropped, blank is none.
- Checkout: `git -C <cwd> rev-parse --show-toplevel` succeeds with non-empty
  output, and `<top>\tools\lab\lab.ps1` exists. (LX-18 deletes `lab.ps1`; it
  changes this marker then.) No `git` on the PATH is no checkout.
- Content check, run only when both the checkout and a version sha exist
  (otherwise `None`, and git is never started):
  `git -C <root> diff --quiet <sha> origin/main -- tools/lab`; exit 0 is
  `Same`, exit 1 `Differs`, anything else `Unknown`. A sha that starts with
  `-` is `Unknown` without running git.

```rust
//! What `lab doctor` reads from the machine: the scheduled task, the daemon,
//! labd.env, account and seed folders, running `SGW.exe` processes, the two
//! installed exes and the tools/lab content check.

use std::fs::File;
use std::io::{self, Read};
use std::path::{Path, PathBuf};
use std::process::{Command, Stdio};

use super::checks::{sgw_exe_path, BinaryFacts, ContentCheck, InstanceFacts, Inputs};
use crate::daemon::env_file::{self, EnvLine};
use crate::daemon::status::{self, Endpoint, StatusReply};
use crate::daemon::{paths, token};

/// Gathers every fact the checks need.
pub fn gather() -> Inputs {
    let env = env_file::read_or_empty(&paths::labd_env_path());
    let status = status::fetch(&Endpoint::discover());
    let install_dir = env_file::get(&env, "CIMMERIA_LAB_INSTALL_DIR")
        .map(str::trim)
        .filter(|dir| !dir.is_empty())
        .map(str::to_string);
    let profile_root = paths::profile_root();
    let instances = labels_for(status.as_ref(), &env)
        .into_iter()
        .map(|label| {
            let account_file = install_dir
                .as_deref()
                .is_some_and(|dir| account_path(dir, &label).exists());
            let seeded = seed_path(&profile_root, &label).exists();
            InstanceFacts {
                label,
                account_file,
                seeded,
            }
        })
        .collect();
    let home = paths::lab_home();
    let version_sha = read_version_sha(&paths::installed_cli_dir().join("cli").join("VERSION"));
    let checkout = std::env::current_dir()
        .ok()
        .and_then(|dir| checkout_root(&dir));
    let content = content_check(checkout.as_deref(), version_sha.as_deref());
    Inputs {
        task_installed: task_installed(),
        listed_sgw: listed_pids(status.as_ref()),
        status,
        token_set: token::daemon_token().is_some(),
        sgw_exe_exists: install_dir
            .as_deref()
            .is_some_and(|dir| sgw_exe_path(dir).exists()),
        install_dir,
        profile_root: profile_root.to_string_lossy().into_owned(),
        instances,
        running_sgw: sgw_pids(),
        binaries: compare_binaries(
            &home.join("bin").join("cimmeria-lab.exe"),
            &home.join("labd").join("cimmeria-lab.exe"),
        ),
        version_sha,
        checkout,
        content,
    }
}

/// Whether `schtasks /Query /TN CimmeriaLabDaemon` exits 0.
pub fn task_installed() -> bool {
    Command::new("schtasks")
        .args(["/Query", "/TN", "CimmeriaLabDaemon"])
        .stdin(Stdio::null())
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .status()
        .map(|status| status.success())
        .unwrap_or(false)
}

/// The instances the checks cover: the daemon's, else labd.env's
/// `CIMMERIA_LAB_INSTANCES`, else `default`.
pub fn labels_for(status: Option<&StatusReply>, env: &[EnvLine]) -> Vec<String> {
    if let Some(status) = status {
        return status.instances.iter().map(|row| row.instance.clone()).collect();
    }
    let listed: Vec<String> = env_file::get(env, "CIMMERIA_LAB_INSTANCES")
        .unwrap_or("")
        .split(',')
        .map(|label| label.trim().to_string())
        .filter(|label| !label.is_empty())
        .collect();

/// The first line of the VERSION file, trimmed; `None` when absent or blank.
pub fn read_version_sha(path: &Path) -> Option<String> {
    let text = std::fs::read_to_string(path).ok()?;
    let line = text
        .trim_start_matches('\u{feff}')
        .lines()
        .next()?
        .trim()
        .to_string();
    (!line.is_empty()).then_some(line)
}

/// Compares the installed exe with the daemon's copy.
pub fn compare_binaries(bin: &Path, daemon: &Path) -> BinaryFacts {
    let mut files = Vec::new();
    for exe in [bin, daemon] {
        match File::open(exe) {
            Ok(file) if exe.is_file() => files.push(file),
            _ => return BinaryFacts::Missing(exe.to_path_buf()),
        }
    }
    let (mut a, mut b) = (files.remove(0), files.remove(0));
    let (Ok(meta_a), Ok(meta_b)) = (a.metadata(), b.metadata()) else {
        return BinaryFacts::Differ;
    };
    if meta_a.len() != meta_b.len() {
        return BinaryFacts::Differ;
    }
    let mut buf_a = vec![0u8; 64 * 1024];
    let mut buf_b = vec![0u8; 64 * 1024];
    loop {
        let (Ok(n_a), Ok(n_b)) = (fill(&mut a, &mut buf_a), fill(&mut b, &mut buf_b)) else {
            return BinaryFacts::Differ;
        };
        if n_a != n_b || buf_a[..n_a] != buf_b[..n_b] {
            return BinaryFacts::Differ;
        }
        if n_a == 0 {
            return BinaryFacts::Same(meta_a.len());
        }
    }
}

/// Reads until `buf` is full or the file ends.
fn fill(reader: &mut impl Read, buf: &mut [u8]) -> io::Result<usize> {
    let mut filled = 0;
    while filled < buf.len() {
        let n = reader.read(&mut buf[filled..])?;
        if n == 0 {
            break;
        }
        filled += n;
    }
    Ok(filled)
}

/// The top of the Cimmeria checkout that `start` is in (it has
/// `tools\lab\lab.ps1`), else `None`; also `None` when git is missing.
pub fn checkout_root(start: &Path) -> Option<PathBuf> {
    let output = Command::new("git")
        .arg("-C")
        .arg(start)
        .args(["rev-parse", "--show-toplevel"])
        .stdin(Stdio::null())
        .stderr(Stdio::null())
        .output()
        .ok()?;
    if !output.status.success() {
        return None;
    }
    let top = String::from_utf8_lossy(&output.stdout).trim().to_string();
    if top.is_empty() {
        return None;
    }
    let top = PathBuf::from(top);
    top.join("tools")
        .join("lab")
        .join("lab.ps1")
        .exists()
        .then_some(top)
}

/// `git -C <root> diff --quiet <sha> origin/main -- tools/lab`, run only when
/// both the checkout and the sha exist (else `None`, and git is not started).
pub fn content_check(root: Option<&Path>, sha: Option<&str>) -> Option<ContentCheck> {
    let (root, sha) = (root?, sha?);
    if sha.starts_with('-') {
        return Some(ContentCheck::Unknown);
    }
    let code = Command::new("git")
        .arg("-C")
        .arg(root)
        .args(["diff", "--quiet", sha, "origin/main", "--", "tools/lab"])
        .stdin(Stdio::null())
        .stdout(Stdio::null())
        .stderr(Stdio::null())
        .status()
        .ok()
        .and_then(|status| status.code());
    Some(match code {
        Some(0) => ContentCheck::Same,
        Some(1) => ContentCheck::Differs,
        _ => ContentCheck::Unknown,
    })
}

/// Pids of every running `SGW.exe`. LX-15 reuses this.
#[cfg(windows)]
pub(crate) fn sgw_pids() -> Vec<u32> {
    use std::mem::{size_of, zeroed};
    use windows_sys::Win32::Foundation::{CloseHandle, INVALID_HANDLE_VALUE};
    use windows_sys::Win32::System::Diagnostics::ToolHelp::{
        CreateToolhelp32Snapshot, Process32FirstW, Process32NextW, PROCESSENTRY32W,
        TH32CS_SNAPPROCESS,
    };

    let mut pids = Vec::new();
    // SAFETY: plain Win32 calls. `entry` is zeroed and its `dwSize` is set as
    // the API requires; the snapshot handle is closed before returning.
    unsafe {
        let snapshot = CreateToolhelp32Snapshot(TH32CS_SNAPPROCESS, 0);
        if snapshot == INVALID_HANDLE_VALUE {
            return pids;
        }
        let mut entry: PROCESSENTRY32W = zeroed();
        entry.dwSize = size_of::<PROCESSENTRY32W>() as u32;
        let mut more = Process32FirstW(snapshot, &mut entry) != 0;
        while more {
            let len = entry
                .szExeFile
                .iter()
                .position(|&unit| unit == 0)
                .unwrap_or(entry.szExeFile.len());
            let name = String::from_utf16_lossy(&entry.szExeFile[..len]);
            if is_sgw_image(&name) {
                pids.push(entry.th32ProcessID);
            }
            more = Process32NextW(snapshot, &mut entry) != 0;
        }
        CloseHandle(snapshot);
    }
    pids
}
```

```rust
/// 1 when any check is a FAIL, else 0.
pub fn exit_code(results: &[Check]) -> u8 {
    u8::from(results.iter().any(|check| check.status == Status::Fail))
}

/// The text `lab doctor` prints.
///
/// Plain mode (what the PowerShell printed): one line per check,
/// `<PASS|WARN|FAIL>  <check>  <detail>` with two spaces between the parts
/// and the hint after `; ` in the detail, and nothing else.
///
/// Fancy mode: a glyph and the check name; under a warning or failure the
/// detail, then the hint, each indented four spaces and dim; then a count
/// line, `N checks · P passed · W warning(s) · F failed`.
pub fn render(ui: &Ui, results: &[Check]) -> String {
    let mut out = String::new();
    for check in results {
        if !ui.fancy {
            out.push_str(&format!(
                "{}  {}  {}\n",
                check.status.label(),
                check.name,
                check.plain_detail()
            ));
            continue;
        }
        let (mark, tone) = match check.status {
            Status::Pass => (Mark::Pass, Tone::Ok),
            Status::Warn => (Mark::Warn, Tone::Warn),
            Status::Fail => (Mark::Fail, Tone::Fail),
        };
        out.push_str(&format!("{} {}\n", ui.paint(tone, ui.mark(mark)), check.name));
        if check.status != Status::Pass {
            out.push_str(&format!("    {}\n", ui.paint(Tone::Dim, &check.detail)));
            if let Some(hint) = &check.hint {
                out.push_str(&format!(
                    "    {}\n",
                    ui.paint(Tone::Dim, &format!("hint: {hint}"))
                ));
            }
        }
    }
    if ui.fancy {
        let count = |status: Status| results.iter().filter(|c| c.status == status).count();
        let warnings = count(Status::Warn);
        let noun = if warnings == 1 { "warning" } else { "warnings" };
        let line = format!(
            "{} checks · {} passed · {warnings} {noun} · {} failed",
            results.len(),
            count(Status::Pass),
            count(Status::Fail)
        );
        out.push_str(&format!("{}\n", ui.paint(Tone::Dim, &line)));
    }
    out
}
```
