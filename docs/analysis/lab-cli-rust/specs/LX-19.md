````powershell
@'
Set-StrictMode -Version Latest
$ErrorActionPreference = 'Stop'

function Get-FullPath([string]$Path) { [IO.Path]::GetFullPath($Path, (Get-Location).Path) }

# Replaces $Old with $New in the file; $Old must occur exactly once. Both may span lines
# (written as here-strings with LF); the file's own line ending is used.
function Edit-Text([string]$Path, [string]$Old, [string]$New) {
    $Path = Get-FullPath $Path
    $t = [IO.File]::ReadAllText($Path)
    $nl = if ($t.Contains("`r`n")) { "`r`n" } else { "`n" }
    $o = $Old -replace '\r?\n', $nl
    $n = $New -replace '\r?\n', $nl
    $i = $t.IndexOf($o, [StringComparison]::Ordinal)
    if ($i -lt 0) { throw "old text not found in ${Path}:`n$o" }
    if ($t.IndexOf($o, $i + 1, [StringComparison]::Ordinal) -ge 0) { throw "old text is not unique in ${Path}:`n$o" }
    [IO.File]::WriteAllText($Path, $t.Substring(0, $i) + $n + $t.Substring($i + $o.Length), [Text.UTF8Encoding]::new($false))
    Write-Host "edited $(Split-Path $Path -Leaf)"
}

# Same, for text that must occur exactly $Count times; every occurrence is replaced.
function Edit-TextAll([string]$Path, [string]$Old, [string]$New, [int]$Count) {
    $Path = Get-FullPath $Path
    $t = [IO.File]::ReadAllText($Path)
    $nl = if ($t.Contains("`r`n")) { "`r`n" } else { "`n" }
    $o = $Old -replace '\r?\n', $nl
    $n = $New -replace '\r?\n', $nl
    $found = ([regex]::Matches($t, [regex]::Escape($o))).Count
    if ($found -ne $Count) { throw "expected $Count occurrences, found $found in ${Path}:`n$o" }
    [IO.File]::WriteAllText($Path, $t.Replace($o, $n), [Text.UTF8Encoding]::new($false))
    Write-Host "edited $(Split-Path $Path -Leaf) ($Count places)"
}

# Replaces the lines from the line equal to $StartLine through the next line equal to $EndLine
# (both included) with $New. $Expect is how many lines that must be.
function Edit-Region([string]$Path, [string]$StartLine, [string]$EndLine, [int]$Expect, [string]$New) {
    $Path = Get-FullPath $Path
    $t = [IO.File]::ReadAllText($Path)
    $nl = if ($t.Contains("`r`n")) { "`r`n" } else { "`n" }
    $lines = [System.Collections.Generic.List[string]]::new([string[]][regex]::Split($t, '\r?\n'))
    $starts = @(0..($lines.Count - 1) | Where-Object { $lines[$_] -ceq $StartLine })
    if ($starts.Count -ne 1) { throw "start line matched $($starts.Count) times in ${Path}: $StartLine" }
    $s = $starts[0]
    $e = -1
    for ($k = $s; $k -lt $lines.Count; $k++) { if ($lines[$k] -ceq $EndLine) { $e = $k; break } }
    if ($e -lt 0) { throw "end line not found after the start in ${Path}: $EndLine" }
    $count = $e - $s + 1
    if ($count -ne $Expect) { throw "the region is $count lines, expected $Expect, in ${Path}" }
    $lines.RemoveRange($s, $count)
    $lines.InsertRange($s, [string[]][regex]::Split($New, '\r?\n'))
    [IO.File]::WriteAllText($Path, ($lines -join $nl), [Text.UTF8Encoding]::new($false))
    Write-Host "replaced $count lines in $(Split-Path $Path -Leaf)"
}

# On the one line that contains $Anchor, rewrites every PowerShell-style lab flag (-Rows,
# -PlanOnly, -RunsPerLease, ...) as its --flag form (--rows, --plan-only, --runs-per-lease).
function Edit-Flags([string]$Path, [string]$Anchor) {
    $Path = Get-FullPath $Path
    $t = [IO.File]::ReadAllText($Path)
    $nl = if ($t.Contains("`r`n")) { "`r`n" } else { "`n" }
    $lines = [regex]::Split($t, '\r?\n')
    $hits = @(0..($lines.Count - 1) | Where-Object { $lines[$_].Contains($Anchor) })
    if ($hits.Count -ne 1) { throw "anchor matched $($hits.Count) lines in ${Path}: $Anchor" }
    $names = 'Rows','Leases','RunsPerLease','Runs','Json','Heal','DiffGolden','PlanOnly','Chaos','Fixtures','Pose','Prepare','Apply','Quiet','BudgetMinutes','Instance','CharacterExpr','Character','MaxMinutes','SpecsDir','NoDraft','Profile','ServerVersion','Rebless','Once','Force','From','SkipBuild'
    $re = '(?<![\w-])-(' + ($names -join '|') + ')(?![\w-])'
    $old = $lines[$hits[0]]
    $new = [regex]::Replace($old, $re, { param($m) '--' + ([regex]::Replace($m.Groups[1].Value, '(?<=[a-z])(?=[A-Z])', '-')).ToLower() })
    if ($new -ceq $old) { throw "no PowerShell-style flag on the line in ${Path}: $Anchor" }
    $lines[$hits[0]] = $new
    [IO.File]::WriteAllText($Path, ($lines -join $nl), [Text.UTF8Encoding]::new($false))
    Write-Host "flags rewritten in $(Split-Path $Path -Leaf)"
}

# Writes a new file with CRLF line endings (no BOM). $Path must not exist.
function New-CrlfFile([string]$Path, [string]$Text) {
    $Path = Get-FullPath $Path
    if (Test-Path -LiteralPath $Path) { throw "$Path already exists" }
    [IO.File]::WriteAllText($Path, (([regex]::Replace($Text.TrimEnd("`r", "`n"), '\r?\n', "`r`n")) + "`r`n"), [Text.UTF8Encoding]::new($false))
    Write-Host "created $(Split-Path $Path -Leaf)"
}
'@ | Set-Content -LiteralPath (Join-Path $env:TEMP 'lx19-edit.ps1') -Encoding utf8
. (Join-Path $env:TEMP 'lx19-edit.ps1')
Get-Command Edit-Text, Edit-TextAll, Edit-Region, Edit-Flags, New-CrlfFile | Select-Object -ExpandProperty Name
````

## The `lab` command

`lab` is one program, `lab.exe`, in front of the lab and the [shared daemon](#the-shared-daemon-cimmeria-lab---http): its status, its log, `labd.env`, installs, instances and clients, and UAT runs, recordings, goldens and calibration. It is Rust ([`crates/lab-cli`](../../crates/lab-cli/)) and stands alone: no PowerShell starts for any command except `start`, `stop`, `restart`, `instances`, `install`, `setup` and `local-setup`, which run a daemon or installer script (see "Where things live" below). It runs from an installed copy, never from a checkout, so a checkout on an old branch can't drive the lab (D-LC4 in the [lab CLI ledger](../analysis/lab-cli/README.md)). It replaced a PowerShell dispatcher in the [lab CLI in Rust](../analysis/lab-cli-rust/README.md) campaign (D-LX1).

### Set up the `lab` command

`lab.exe` is `%LOCALAPPDATA%\cimmeria-lab\bin\lab.exe`, next to `cimmeria-lab.exe`. There is no shim: `lab` is found through your user `PATH`. On a machine that has no `lab.exe` yet, bootstrap from a checkout on a current `main`:

```powershell
pwsh tools/lab/install.ps1      # builds and installs cimmeria-lab.exe, lab.exe and the other lab binaries
pwsh tools/lab/cli/setup.ps1    # copies the scripts lab still runs; offers to put bin on your PATH
```

`setup.ps1` does three things:

1. Copies the PowerShell the CLI still runs into `%LOCALAPPDATA%\cimmeria-lab\cli\`: the scripts in `cli\*.ps1` (`install`, `setup`, `local-setup` and their libraries), and `daemon.ps1`, `instances.ps1` and `labd-lib.ps1` beside them. Tests are not copied, and a script the checkout no longer has is removed. `cli\cli\VERSION`, the checkout's short commit, is written last, so a copy that failed part way has no `VERSION`.
2. Deletes `%LOCALAPPDATA%\cimmeria-lab\bin\lab.cmd` when an older setup left it: it ran a PowerShell dispatcher that no longer exists, and `lab.exe` is the command now.
3. Asks `Add ...\cimmeria-lab\bin to your user PATH? [y/N]` when the
   folder isn't on it yet. `-Yes` adds it without asking, and `-NoPath`
   leaves the `PATH` alone. The edit goes through `HKCU\Environment`
   directly, so a `REG_EXPAND_SZ` `PATH` keeps its type, and setup
   broadcasts the change. Open a new terminal to see it.

From then on `lab` does the same itself. `lab setup --from C:\src\Cimmeria` refreshes the copied scripts (`--no-path` leaves the `PATH` alone, `--yes` adds it without asking); from the installed copy there is no checkout to copy from, so `lab setup` needs `--from`. `lab install --from <worktree>` builds and installs the lab, `lab.exe` included, and refreshes the scripts from that worktree. Each replaced file is kept as `<name>.<yyyymmdd>.old`, and Windows lets `lab install` rename the running `lab.exe`, so the command can replace itself. `lab version` shows which commit your `lab.exe` was built from.

### Help, flags and output

- **Help.** `lab` or `lab help` prints the commands in four groups: Run and record UAT (`uat`, `watch`, `record`, `calibrate`, `golden`), Daemon (`status`, `start`, `stop`, `restart`, `logs`, `doctor`), Instances and clients (`instances`, `clients`), and Install and configure (`env`, `install`, `setup`, `local-setup`, `version`, `completions`), with a line for each, then examples. `lab help <command>` (or `lab <command> -h`) prints that command's page: its arguments, a longer description and examples. A mistyped command exits 2 and names the nearest commands (`lab stauts` suggests `status`).
- **Flags.** Flags are `--long-flags`, which is how this guide writes them. The PowerShell spellings still work: `-Rows`, `-PlanOnly` and `-Value:-x` run the same as `--rows`, `--plan-only` and `--value=-x`, because `lab` rewrites them before it parses anything. Commands, scripts and agent briefs written either way keep working.
- **Colour and plain output.** On a terminal, `lab` prints colour, glyphs and tables. When stdout is redirected (a pipe or a file), when `NO_COLOR` is set to a non-empty value, or when `TERM` is `dumb`, it prints the plain text the PowerShell command printed, with no escape codes. `--json` output and exit codes are the same either way.
- **Completions.** `lab completions powershell | Out-String | Invoke-Expression` loads tab completion for the current session. `lab completions powershell --install` adds that line to `$PROFILE.CurrentUserAllHosts` once (it looks for the line first, so running it again changes nothing) and says what it did; open a new terminal afterwards, and `lab u` then Tab completes to `lab uat`. `lab completions bash`, `zsh` and `fish` print their scripts for you to load; `--install` is for PowerShell only.
